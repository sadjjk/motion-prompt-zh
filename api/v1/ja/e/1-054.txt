// 思考オーブ・検索中 / Thinking orb: searching
// パック: aiux (グロー・リキッド) · element 06 · カテゴリ: 読み込み・思考中
// 何が起きるか: 点でできた地球が回り、縦の光の線がその上をなでるように動いて、触れた点が明るくなる。
// 動きの仕様:
//   - 球は半径 58px、点 240 個。1 周期で 1 回自転
//   - 走査する経線は 1 周期で 2 周（自転の 2 倍の速さ）。経線は不透明度 0.35、太さ 1px
//   - 経線に近い点ほど明るい：ずれの角度の 2 乗に対して exp(−Δ²/0.05)、最大 +0.75
//   - 右に「Searching the web…」15px
// 使いどころ: Web検索・資料の検索・データの照会など、何かを探している間に。
// タグ: thinking-orb, searching, search, retrieval, loading, globe
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#1-054
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 06
// 06 思考球·搜索（state="searching"）：点阵地球自转，一条亮经线扫过，扫到的点变亮
CELLS[5] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC, sweep = 2 * TAU * u / CYC;   // 自转 1 圈、扫描 2 圈 / 周期
  for (const p of SPHERE) {
    const q = rot(p, spin, 0), lon = Math.atan2(q[0], q[2]);
    const dd = wrapPi(lon - sweep), boost = 0.75 * Math.exp(-(dd * dd) / 0.05);
    dot(k, rot(p, spin, ORB.tilt), 0.7, boost);
  }
  k.strokeStyle = 'rgba(236,236,241,0.35)'; k.lineWidth = 1; k.beginPath();
  for (let i = 0; i <= 40; i++) {
    const lat = -Math.PI / 2 + Math.PI * i / 40;
    const q = rot([Math.cos(lat) * Math.sin(sweep), Math.sin(lat), Math.cos(lat) * Math.cos(sweep)], 0, ORB.tilt);
    const X = ORB.x + q[0] * ORB.r * 1.02, Y = ORB.y - q[1] * ORB.r * 1.02;
    if (i) k.lineTo(X, Y); else k.moveTo(X, Y);
  }
  k.stroke();
  orbLabel(k, 'Searching the web…');
};
