// パック: aiux (AIインターフェースの動き) · element 04 · カテゴリ: 読み込み・思考中
// 何が起きるか: 暗い入力欄のまわりに虹色の輪が何重にも広がっては縮み、欄そのものはくっきりしたまま。
// 動きの仕様:
//   - 入力欄 380×64px、角丸 16px。光は欄の外側だけに描く
//   - 呼吸 2.4 秒（1 周期に 2 回）。外への広がり 4→18px
//   - 4 重の輪（太さ 3px・ぼかし 18px）を外ほど薄く重ねる。不透明度 0.25→0.7
//   - 虹色のグラデーションは 1 周期で 1 回転
// 使いどころ: 空の状態やトップページの主な入力欄、おすすめのプランカードを目立たせたいときに。
// タグ: pulse, input, highlight, glow, landing, featured
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#1-052
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 04
// 04 外扩呼吸光晕（border-beam size="pulse-outside"）：彩色光晕在输入框外侧向外扩散再收回
CELLS[3] = (k, w, h, u) => {
  bg(k);
  const x = 79.5, y = 72, W = 380, H = 64, R = 16;
  const s = 0.5 - 0.5 * Math.cos(TAU * u / 2.4);
  const e = 4 + 14 * s, rot0 = TAU * u / CYC;
  for (let i = 4; i >= 1; i--) {
    const ex = e * i / 4;
    k.strokeStyle = conic(k, x + W / 2, y + H / 2, rot0, (0.25 + 0.45 * s) * (1.15 - i / 4));
    k.lineWidth = 3; k.shadowColor = hsla(200 + 360 * u / CYC, 100, 60, 0.5 * s); k.shadowBlur = 18;
    rr(k, x - ex, y - ex, W + ex * 2, H + ex * 2, R + ex); k.stroke();
  }
  k.shadowBlur = 0; k.shadowColor = 'transparent';
  field(k, x, y, W, H, R, 'Describe what you want to build…');
};
