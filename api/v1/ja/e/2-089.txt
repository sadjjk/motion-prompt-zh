// 文字トンネル / Text tunnel
// パック: motion2 (モーショングラフィックスの演出 第2弾) · element 50 · カテゴリ: 文字・図形の演出
// 何が起きるか: 「穿越」の層が中心の小さな点から次々と拡大して画面の外へ抜け、白とコーラルの層が交互に流れ、中心は暗く沈んでいる。
// 動きの仕様:
//   - 同じ語8層が中心で0.04倍から9倍まで指数的に拡大する
//   - 1層が通り抜けるのに4.7秒、各層の位相は1/8ずつずれる
//   - 前半25%で現れ、後半40%で消える
//   - 隣の層は白とコーラルが交互で、中心半径70pxに暗い光
// 使いどころ: オープニング、ワープ・突入の演出、場面転換に。
// タグ: tunnel, zoom, warp, depth, loop, intro
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#2-089
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 50
// 改写自 KittenJam Motion Library #55（Hannah）· 文字隧道：8 层「穿越」从中心 0.04 倍按指数放大到 9 倍冲出画面，每层走完需 4.7 秒、相位依次错开 1/8；前 25% 淡入、后 40% 淡出，相邻层白色与珊瑚色交替，中心有暗色光晕。
const TUNL_N = 8;
CELLS[49] = (k, w, h, u) => {
  u = snapU(u);
  stage(k, '#0A1230');
  const L = [];
  for (let i = 0; i < TUNL_N; i++) L.push([mod(u / CYC + i / TUNL_N, 1), i]);
  L.sort((a, b) => a[0] - b[0]).forEach(([p, i]) => {
    const s = 0.04 * Math.pow(9 / 0.04, p), a = P(p, 0, 0.25) * (1 - P(p, 0.6, 1));
    if (a <= 0) return;
    k.save(); k.globalAlpha = a; k.translate(w / 2, h / 2); k.scale(s, s);
    txt(k, '穿越', 0, 0, `900 44px ${JP}`, i % 2 ? C.coral : '#fff', 'center', 'middle');
    k.restore();
  });
  const g = k.createRadialGradient(w / 2, h / 2, 0, w / 2, h / 2, 70);
  g.addColorStop(0, 'rgba(5,9,26,.75)'); g.addColorStop(1, 'rgba(5,9,26,0)');
  k.fillStyle = g; k.fillRect(0, 0, w, h);
};
