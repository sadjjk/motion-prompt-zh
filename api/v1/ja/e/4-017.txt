// リソグラフ / Risograph
// パック: scenes (スタイルシーン) · element 17 · カテゴリ: 文様・グラフィック
// 何が起きるか: 生成りの紙に、ピンクとティールの大きな円・四角・太字が重なって第3の色を生み、版ずれが小さく行き来する。
// 動きの仕様:
//   - 画面 539 × 208 px、生成りの地 #F4F0E6。蛍光ピンク #FF48B0 とティール #00A69C の2版を乗算で重ねて新たな色を作る
//   - ピンク版：半径 78px の円、84 × 84 px の四角、100px の極太 RISO。ティール版：半径 66px の円、64px の極太 PRINT、太さ 4px の横線 8 本
//   - 2版のずれは 3 ± 2px（1〜5px）、正弦で 6.4 秒に 3 往復し、ピンク版は右へ、ティール版は上へずれる
//   - 固定のかすれ点 1500 個と、不透明度 0.1 の乗算グレイン
// 使いどころ: インディー誌や音楽イベントのポスター・カバー、クリエイティブ系ブランドのファーストビューにレトロな印刷感を出したいとき。
// タグ: risograph, print, poster, overprint, retro, grain, typography
// サイズ: 539x208 px · 1周: 6.4 s
// Needs: kit scenes -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/scenes.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#4-017
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 17
// 17 孔版印刷：荧光粉与青绿两版正片叠底，大圆、方块、粗体 RISO/PRINT 叠出第三色；两版错位按正弦每周期往复 3 次
CELLS[16] = (k, w, h, u) => {
  k.fillStyle = '#F4F0E6'; k.fillRect(0, 0, CW, CH);
  const off = 3 + Math.sin(TAU * u / CYC * 3) * 2;           // 套印错位 1–5px
  k.save(); k.globalCompositeOperation = 'multiply'; k.letterSpacing = '-4px';
  k.fillStyle = '#FF48B0';
  k.beginPath(); k.arc(300 + off, 92, 78, 0, TAU); k.fill();
  txt(k, 'RISO', 14 + off, 196, `900 100px ${DISP}`, '#FF48B0');
  k.fillRect(440 + off, 14, 84, 84);
  k.fillStyle = '#00A69C';
  k.beginPath(); k.arc(372, 128 - off, 66, 0, TAU); k.fill();
  txt(k, 'PRINT', 16, 86 - off, `900 64px ${DISP}`, '#00A69C');
  for (let i = 0; i < 8; i++) k.fillRect(440, 118 + i * 11, 90, 4);
  k.restore();
  // 漏印斑点：1500 个固定浅色点（原作 22000 个，按面积缩减），缓存成图层
  k.drawImage(layer('RISO_speck', g => { for (let i = 0; i < 1500; i++) { g.fillStyle = `rgba(244,240,230,${hash(i * 3 + 5) * 0.55})`; g.fillRect(hash(i * 3 + 6) * CW, hash(i * 3 + 7) * CH, 1.2, 1.2); } }), 0, 0);
  grain(k, 0.1, 'multiply');
};
