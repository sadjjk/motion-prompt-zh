// フレームインフレーム / Frame within a frame
// パック: photo (写真の表現・光と画風) · element 30 · カテゴリ: 構図
// 何が起きるか: 手前の暗いドア枠が現れ、奥に立つ人物をもう一度囲む。内側の縁に明るい線があり、厚みがあると分かる。
// 動きの仕様:
//   - 人物は中央 x 270 に立ち、身長56px。背景の左に木、右に建物
//   - ドア口: 幅168px、y 30 から下端まで、左右中央。外側は暗い #33405A で塗る
//   - ドア枠の内側に6pxの明るい縁 #CFCECB を付けて厚みを示す。破線で開口を描き、上に「框中框」
//   - 補助線: 琥珀色 #F5A524 の破線、太さ1.6px。効果の量に合わせて現れる
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、イーズインアウト）
// 使いどころ: 視線を被写体へ集めたいとき、層を増やしたいとき、のぞき見や隔たりを表したいとき。
// タグ: frame within a frame, framing, doorway, window, composition, depth
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#3-062
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 30
// 框中框: 近处的门框淡入，把远处的主体再框一次；门框内沿留一条浅色边，表示它有厚度（不是后期加的装饰边框）
CELLS[29] = cell((k, u) => {
  const a = env(u), D = field(k, BASE, 150);
  tree(D, 120, 150, 14, BASE); building(D, 400, 150, 44, 50, BASE, 4);
  figure(D, CW / 2, 162, 56, BASE.person);
  const hx = CW / 2 - 84, hy = 30, hw = 168, hh = CH - 30;
  k.save(); k.globalAlpha = a;
  k.fillStyle = BASE.shade; k.beginPath(); k.rect(0, 0, CW, CH); k.rect(hx, hy, hw, hh); k.fill('evenodd');
  k.fillStyle = BASE.edge; k.fillRect(hx - 6, hy - 6, hw + 12, 6); k.fillRect(hx - 6, hy, 6, hh); k.fillRect(hx + hw, hy, 6, hh);
  k.restore();
  guide(k, [[hx, hy], [hx + hw, hy], [hx + hw, CH], [hx, CH], [hx, hy]], a);
  tag(k, '框中框', CW / 2, 16, a);
});
