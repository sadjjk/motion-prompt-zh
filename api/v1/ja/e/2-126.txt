// 積み上がるコイン / Growing coin stacks
// 分類: モーショングラフィックス / 図形・オブジェクト / 物・機械
// 何が起きるか: 地面の線の上に金貨の山が 5 つ。左の山から順に積み始め、新しい一枚は少し上から落ちて乗る。3.6 秒で 4、7、10、13、16 枚の山がそろう。
// 動きの仕様:
//   - 金貨の山 5 つが 80 px 間隔で地面の線の上に立つ。1 枚は 60 × 16 px の楕円で、上面は金色、側面は濃い金色。段の間隔は 9 px
//   - c 番目の山は最大 4 + 3·c 枚（4、7、10、13、16）。0.3 + 0.3·c 秒に始まり、一定の速さで増えて 3.6 秒で満杯になる
//   - 新しい 1 枚は 20 px 上から減速しながら乗る。各山のいちばん上の 1 枚には短い白いハイライト
//   - 5.4–6.0 秒に金貨がフェードアウト。0 秒では地面の線だけなので、ループがつながる
// 使いどころ: 貯金、収入、ユーザー数などが段階的に増えていく様子を、グラフより親しみやすく見せたいとき。
// タグ: coin-stack, coins, growth, money, bar-chart, savings, short-video
// サイズ: 539x208 px · 1周: 6.4 s
// 依存する共通コード: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/vibefx.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#2-126
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここからこの効果のコード。CELLS や定数・補助関数は依存する共通コードにある。

// @element 03
// 硬币堆高：5 列金币从左到右依次加高（4、7、10、13、16 枚），像柱状图一样长起来
CELLS[2] = (k, w, h, u) => {
  bg(k);
  const F = fadeIn(u), BASE = 184, STEP = 9;
  k.strokeStyle = '#3A3D48'; k.lineWidth = 2;
  k.beginPath(); k.moveTo(50, BASE + 10); k.lineTo(490, BASE + 10); k.stroke();
  k.globalAlpha = F;
  for (let c = 0; c < 5; c++) {
    const x = 110 + 80 * c, max = 4 + 3 * c, p = seg(u, 0.3 + 0.3 * c, 3.6) * max;
    const n = Math.floor(p + 1e-9), frac = p - n;
    const total = Math.min(max, n + (frac > 0 ? 1 : 0));
    for (let i = 0; i < total; i++) {
      let y = BASE - STEP * i;
      if (i === n) y -= 20 * (1 - out3(clamp(frac / 0.4)));
      k.fillStyle = GOLD_D; k.beginPath(); k.ellipse(x, y + 4, 30, 8, 0, 0, TAU); k.fill();
      k.fillStyle = GOLD; k.beginPath(); k.ellipse(x, y, 30, 8, 0, 0, TAU); k.fill();
      if (i === total - 1) {
        k.strokeStyle = 'rgba(255,255,255,0.6)'; k.lineWidth = 2;
        k.beginPath(); k.ellipse(x, y, 20, 4, 0, Math.PI * 1.1, Math.PI * 1.6); k.stroke();
      }
    }
  }
  k.globalAlpha = 1;
};
