// 思考オーブ・聞き取り中 / Thinking orb: listening
// パック: aiux (グロー・リキッド) · element 07 · カテゴリ: 読み込み・思考中
// 何が起きるか: 点を並べた緯線の輪が上から下へ順にふくらんではしぼみ、波が球の上を転がっていくように見える。
// 動きの仕様:
//   - 球は半径 58px。緯線の輪 11 本（−75°〜75°、15° おき）、輪ごとの点の数は緯度に応じて 8〜34 個
//   - 各輪の半径が最大 ±10%（約 6px）ふくらむ。1 周期に 2 波、隣の輪とは 0.7 rad ずれる
//   - ふくらんだ輪ほど点が明るい。球は 1 周期で 1 回自転
//   - 右に「Listening…」15px
// 使いどころ: 音声入力・音声モード・ディクテーションでマイクが開いている間に。
// タグ: thinking-orb, listening, voice, microphone, dictation, wave
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#1-055
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 07
// 07 思考球·聆听（state="listening"）：纬线环上的点随声波起伏，波沿纬度方向滚动
CELLS[6] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC;
  for (let i = 0; i < 11; i++) {
    const lat = (-75 + 15 * i) * Math.PI / 180, cl = Math.cos(lat);
    const amp = 0.1 * Math.sin(2 * TAU * u / CYC - i * 0.7);   // 振幅 ≈ 6 px，2 个波 / 周期
    const n = Math.max(8, Math.round(34 * cl));
    for (let j = 0; j < n; j++) {
      const lon = j * TAU / n, rr2 = cl * (1 + amp);
      dot(k, rot([rr2 * Math.cos(lon), Math.sin(lat), rr2 * Math.sin(lon)], spin, ORB.tilt), 0.8, Math.max(0, amp) * 4);
    }
  }
  orbLabel(k, 'Listening…');
};
