// 分類: モーショングラフィックス / 図形・オブジェクト / 情報・テック
// 何が起きるか: 下の金色の「主题」から 3 本の枝が伸びて「概念」「方法」「案例」が現れ、さらに上へ 6 つの金色の点が出る。ノードは弾むように現れ、ツリー全体がゆっくり上下に揺れる。
// 動きの仕様:
//   - 3 層。下の中央に根のボックス「主题」、中段に 160 px 間隔で「概念」「方法」「案例」の 3 つのボックス、上段に半径 7 px の金色の点 6 つ（各ボックスの上に 2 つ）
//   - d 層目は 0.3 + 0.9·d 秒（0.3、1.2、2.1 秒）に現れる。そのノードへの S 字の光る線が直前の 0.5 秒で描かれ、続いてノードが減衰バネで弾んで出る
//   - 中段への線は金色で幅 3 px、上段への線は濃い金色で幅 2 px。ツリー全体が 1 周期に 2 回、2 px 上下に揺れる
//   - 5.4–6.0 秒にツリーがフェードアウト。0 秒では背景だけなので、ループがつながる
// 使いどころ: 講座の目次、知識の体系、テーマの分解など、全体の構造を最初に見せたいとき。
// タグ: tree-diagram, knowledge-map, hierarchy, nodes, branching, spring-pop, short-video
// サイズ: 539x208 px · 1周: 6.4 s
// 依存する共通コード: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/vibefx.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#2-129
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここからこの効果のコード。CELLS や定数・補助関数は依存する共通コードにある。

// @element 06
// 知识树：从底部根节点「主题」向上长出 3 个分支和 6 个末梢，节点逐层弹出，整棵树轻轻浮动
CELLS[5] = (k, w, h, u) => {
  bg(k);
  const F = fadeIn(u), bob = 2 * Math.sin(u * TAU / CYC * 2);
  const L1 = ['概念', '方法', '案例'], X1 = [110, 270, 430];
  const nodes = [{ x: 270, y: 180, d: 0, label: '主题', p: -1 }];
  X1.forEach((x, i) => nodes.push({ x, y: 110, d: 1, label: L1[i], p: 0 }));
  X1.forEach((x, i) => { nodes.push({ x: x - 40, y: 40, d: 2, p: 1 + i }); nodes.push({ x: x + 40, y: 40, d: 2, p: 1 + i }); });
  k.globalAlpha = F;
  const T0 = d => 0.3 + 0.9 * d;
  nodes.forEach(nd => {
    if (nd.p < 0) return;
    const pa = nodes[nd.p];
    glowLine(k, pa.x, pa.y + bob, nd.x, nd.y + bob, seg(u, T0(nd.d) - 0.5, T0(nd.d)), nd.d === 1 ? GOLD : GOLD_D, nd.d === 1 ? 3 : 2);
  });
  nodes.forEach(nd => {
    const s = spring(u - T0(nd.d));
    if (s <= 0) return;
    const y = nd.y + bob;
    if (nd.label) {
      const bw = nd.d ? 64 : 76, bh = nd.d ? 30 : 34;
      k.save(); k.translate(nd.x, y); k.scale(s, s);
      rr(k, -bw / 2, -bh / 2, bw, bh, 8); k.fillStyle = nd.d ? '#22252E' : GOLD; k.fill();
      k.strokeStyle = GOLD; k.lineWidth = 1.5; k.stroke();
      txt(k, nd.label, 0, 1, nd.d ? 15 : 16, nd.d ? INK : BG, 'center', 700);
      k.restore();
    } else {
      k.fillStyle = GOLD; k.beginPath(); k.arc(nd.x, y, 7 * s, 0, TAU); k.fill();
    }
  });
  k.globalAlpha = 1;
};
