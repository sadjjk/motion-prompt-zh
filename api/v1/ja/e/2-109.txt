// ピクセレート / Pixelate
// 分類: モーショングラフィックス / エフェクト・トランジション / トランジション
// 何が起きるか: 画面がだんだん粗いモザイクになり、いちばん粗いところで切り替わって細かく戻る。
// 動きの仕様:
//   - 野原の画面 → 街の画面 → 野原を2.35秒ずつ。0.2〜1.9秒（1.7秒）に、モザイクの1マスの大きさを 1・2・3・4・6・8・11・15・20・27・36px と段階で上げて下げる（中間の1.05秒がいちばん粗い）
//   - 前半は前の画面、後半は次の画面（いちばん粗いところで差し替える）
//   - 作り方: 画面を 幅÷マス の大きさに縮めて（なめらか補間あり）、補間を切って元の大きさに引き伸ばす
// 使いどころ: ゲーム風・レトロ・デジタルな場面転換に。隠したい物をぼかして切り替えたいときにも。
// タグ: pixelate, mosaic, pixel, transition, retro, 8-bit, scene-change
// サイズ: 539x208 px · 1周: 4.7 s
// 依存する共通コード: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#2-109
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここからこの効果のコード。CELLS や定数・補助関数は依存する共通コードにある。

// @element 21
// 21 ピクセレート（画面がだんだん粗いモザイクになり、いちばん粗いところで切り替わって細かく戻る）
const SMC = mkCanvas(CW, CH), SMK = SMC.getContext('2d');
const BLV = [1, 2, 3, 4, 6, 8, 11, 15, 20, 27, 36];
CELLS[20] = abCell((k, w, h, u, a, b) => {
  const S = [scField, scCity], p = P(u, TA - 0.3, TB + 0.4), tri = 1 - Math.abs(2 * p - 1), bs = BLV[Math.round(tri * (BLV.length - 1))];
  const scr = p < 0.5 ? a : b;
  if (bs <= 1) return S[scr](k, w, h);
  TK.setTransform(1, 0, 0, 1, 0, 0); TK.globalCompositeOperation = 'source-over'; TK.clearRect(0, 0, w, h); S[scr](TK, w, h);
  const sw = Math.ceil(w / bs), sh = Math.ceil(h / bs);
  SMK.setTransform(1, 0, 0, 1, 0, 0); SMK.clearRect(0, 0, CW, CH); SMK.imageSmoothingEnabled = true; SMK.imageSmoothingQuality = 'high'; SMK.drawImage(TMP, 0, 0, sw, sh);
  k.imageSmoothingEnabled = false; k.drawImage(SMC, 0, 0, sw, sh, 0, 0, sw * bs, sh * bs); k.imageSmoothingEnabled = true;
});
