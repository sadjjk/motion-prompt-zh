// 思考オーブ・作業中 / Thinking orb: working
// パック: aiux (AIインターフェースの動き) · element 05 · カテゴリ: 読み込み・思考中
// 何が起きるか: ゆっくり回る点の球のまわりを、傾いた3本の軌道に沿って尾を引く粒がそれぞれの速さで回る。
// 動きの仕様:
//   - 球は半径 58px、点 240 個（フィボナッチ球面配置）。手前ほど点が大きく明るい。1 周期で 1 回自転、俯角 0.32 rad
//   - 軌道は 3 本（傾き 0.35・1.05・−0.6 rad）、半径は球の 1.14 倍。各軌道に粒 8 個
//   - 1 周期に 1 周・2 周・逆向き 1 周
//   - 粒は 5 つの残像で尾を引く。右に「Working…」15px
// 使いどころ: エージェントの作業やツールの実行など、汎用の「AIが作業中」を示したいときに。2秒未満の待ちには使わない。
// タグ: thinking-orb, working, agent, loading, ai, orbit, particles
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#1-053
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 05
// 05 思考球·运转（thinking-orbs state="working"）：粒子在三条倾斜轨道上绕点阵球运行
CELLS[4] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC;
  for (const p of SPHERE) dot(k, rot(p, spin, ORB.tilt), 0.45);
  const orbits = [[0.35, 0, 1], [1.05, 2.1, 2], [-0.6, 4.2, -1]];   // [倾角, 偏航, 每周期圈数]
  for (const [tilt, yaw, sp] of orbits) {
    for (let j = 0; j < 8; j++) {
      for (let tr = 5; tr >= 0; tr--) {
        const th = sp * TAU * u / CYC + j * TAU / 8 - Math.sign(sp) * tr * 0.07;
        let p = [Math.cos(th) * 1.14, 0, Math.sin(th) * 1.14];
        p = rot([p[0], p[1] * Math.cos(tilt) - p[2] * Math.sin(tilt), p[1] * Math.sin(tilt) + p[2] * Math.cos(tilt)], yaw, ORB.tilt);
        dot(k, p, tr ? 0.5 * (1 - tr / 6) : 1, tr ? 0 : 0.35);
      }
    }
  }
  orbLabel(k, 'Working…');
};
