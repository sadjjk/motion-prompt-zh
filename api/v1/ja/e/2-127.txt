// 複利カーブ / Compound growth curve
// 分類: モーショングラフィックス / 図形・オブジェクト / 情報・テック
// 何が起きるか: 左と下の座標軸に、1 期 10% で 40 期分の成長カーブが原点から描かれる。先端の白い点が右へ進み、カーブは終盤で急に立ち上がり、右上に「×45」が浮かぶ。
// 動きの仕様:
//   - 灰色の座標軸で 460 × 150 px の描画域を囲む。カーブは y = (1.1^(40x) − 1) / (1.1^40 − 1)。1 期 10% で 40 期分の成長を描画域に合わせて縮めたもの
//   - 0.3–3.6 秒に、幅 3 px で柔らかく光る金色の線がイーズインアウトで左から右へ描かれる。先端に半径 5 px の白い点
//   - 3.6–4.0 秒に右上へ 30 px の太字の金色で「×45」がフェードイン。1.1^40 ≈ 45 のため
//   - 5.4–6.0 秒にカーブとラベルがフェードアウト。0 秒では座標軸だけなので、ループがつながる
// 使いどころ: 複利、習慣、学習など、少しずつの積み重ねが後で大きな差になることを伝えたいとき。
// タグ: compound-interest, exponential, growth-curve, line-chart, finance, data-visualization, short-video
// サイズ: 539x208 px · 1周: 6.4 s
// 依存する共通コード: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/vibefx.js
// 動く見本: https://sadjjk.github.io/motion-prompt-zh/?lang=ja#2-127
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここからこの効果のコード。CELLS や定数・補助関数は依存する共通コードにある。

// @element 04
// 复利曲线：每期 +10%、共 40 期的增长曲线从坐标原点画出，先平后陡，亮点沿线走，最后标出「×45」
CELLS[3] = (k, w, h, u) => {
  bg(k);
  const F = fadeIn(u), OX = 40, OY = 184, W = 460, H = 150, G = Math.pow(1.1, 40);
  const f = x => (Math.pow(1.1, 40 * x) - 1) / (G - 1);
  k.strokeStyle = '#3A3D48'; k.lineWidth = 2;
  k.beginPath(); k.moveTo(OX, OY - H - 10); k.lineTo(OX, OY); k.lineTo(OX + W + 10, OY); k.stroke();
  k.globalAlpha = F;
  const p = inOut3(seg(u, 0.3, 3.6)), n = 80;
  if (p > 0) {
    k.save(); k.strokeStyle = GOLD; k.lineWidth = 3; k.lineJoin = 'round'; k.shadowColor = GOLD; k.shadowBlur = 8;
    k.beginPath();
    for (let i = 0; i <= n; i++) { const x = p * i / n; const X = OX + W * x, Y = OY - H * f(x); if (i) k.lineTo(X, Y); else k.moveTo(X, Y); }
    k.stroke(); k.restore();
    const X = OX + W * p, Y = OY - H * f(p);
    k.fillStyle = '#FFFFFF'; k.beginPath(); k.arc(X, Y, 5, 0, TAU); k.fill();
  }
  const L = seg(u, 3.6, 4.0);
  if (L > 0) { k.globalAlpha = F * L; txt(k, '×45', OX + W - 8, OY - H + 6, 30, GOLD, 'right', 800); }
  k.globalAlpha = 1;
};
