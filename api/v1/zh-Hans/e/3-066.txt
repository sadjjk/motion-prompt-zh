// 动作衔接 / Match on action
// 分类包: camera (镜头运动与剪辑) · element 35 · 子分类: 剪辑与转场
// 动画表现: 人物匀速横穿画面，走到一半从正面机位硬切到侧前方机位；位置和步伐在切点前后连贯，脚下闪一下蓝色虚线表示是同一个动作。
// 运动规格:
//   - 人物：0 秒至 4.2 秒从 x −3.2 m 匀速走到 3.2 m，共 6.4 m，步伐连续。
//   - 机位 1：正面，高 1.4 m，距人物 8 m。
//   - 机位 2：2.3 秒切换，位于右侧 1.4 m、高 1.2 m、距人物 3 m，转向 0.35 rad。
//   - 切换后 0.35 秒内，脚下显示蓝色虚线短标并淡出；右上角显示“机位 1”/“机位 2 · 动作延续”。
//   - 最后 0.5 秒交叉淡化回第一个镜头，每轮 4.7 秒。
// 适用场景: 想换角度又不打断动作流畅感时，比如开门、坐下、递东西、出拳。
// 标签: match on action, cutting on action, match cut, action continuity, cut, editing, continuity
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/camera.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#3-066
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 35
// 35 动作衔接：A 匀速横穿，走到一半从正面机位切到侧前方机位；走位与步态都由 u 连续算出，切点前后动作不重来
const walkX = u => lerp(-3.2, 3.2, Math.min(1, u / 4.2));
const walker = u => ({ person: { x: walkX(u), z: 8, walk: u * 7, col: C.blue } });
CELLS[34] = shot(cutCell([
  [0, (k, w, h, u) => world(k, w, h, cam({ y: 1.4 }), walker(u)), '机位 1'],
  [2.3, (k, w, h, u) => world(k, w, h, cam({ x: 1.4, y: 1.2, z: 5, yaw: -0.35 }), Object.assign(walker(u), {
    after: (kk, cc) => {   // 切点后 0.35 秒：脚下的蓝色虚线短标，说明是同一个动作
      const g = 1 - P(u, 2.3, 2.65); if (g <= 0) return;
      const p = proj(cc, walkX(u), 0, 8);
      kk.save(); kk.globalAlpha = g; kk.strokeStyle = C.blue; kk.lineWidth = 2; kk.setLineDash([5, 4]);
      kk.beginPath(); kk.moveTo(p.x - 40, p.y + 6); kk.lineTo(p.x + 40, p.y + 6); kk.stroke(); kk.restore();
    }
  })), '机位 2 · 动作延续']
]));
