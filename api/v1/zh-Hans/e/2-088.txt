// 钟摆摇晃 / Swing letters
// 分类包: motion2 (动态图形进阶) · element 49 · 子分类: 文字与图形
// 动画表现: 4 个字用细线挂在横杆下，以上方黄色支点为中心缓缓左右摆动，相邻字相位略有错开。
// 运动规格:
//   - 每个字吊在上方支点，摆幅 ±14°
//   - 4.7 秒摆 3 个来回（每个约 1.57 秒），余弦曲线
//   - 相邻字相位错开 0.27 秒
//   - 偶数位字为珊瑚色，其余白色
// 适用场景: 适合慵懒可爱的 Logo、角色介绍或待机画面。
// 标签: swing, pendulum, sway, hanging, cute, idle
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#2-088
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 49
// 改写自 KittenJam Motion Library #26（Hannah）· 钟摆摇晃：每个字用细线挂在横杆下，以字上方的小圆点为支点 ±14° 左右摆，4.7 秒摆 3 个来回（ease-in-out 的余弦），相邻字相位错开 0.27 秒，偶数位字用珊瑚色。
const SWNG_T = CYC / 3;
CELLS[48] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '悠然自得', F = `900 56px ${JP}`, RAIL = 34, PIV = 50;
  k.fillStyle = C.navy3; rr(k, 70, RAIL - 2, w - 140, 4, 2); k.fill();
  chars(k, S, F, 34).forEach((c, i) => {
    const a = 14 * Math.PI / 180 * Math.cos(Math.PI * 2 * (u + 0.27 * i) / SWNG_T), x = w / 2 + c.x;
    k.strokeStyle = C.dim; k.lineWidth = 1.5; k.beginPath(); k.moveTo(x, RAIL); k.lineTo(x, PIV); k.stroke();
    k.save(); k.translate(x, PIV); k.rotate(a);
    k.beginPath(); k.moveTo(0, 0); k.lineTo(0, 30); k.stroke();
    txt(k, c.ch, 0, 62, F, i % 2 ? C.coral : '#fff', 'center', 'middle');
    k.restore();
    k.fillStyle = C.yel; k.beginPath(); k.arc(x, PIV, 4.5, 0, 7); k.fill();
  });
};
