// 思考球·聆听 / Thinking orb: listening
// 分类包: aiux (光效与液态) · element 07 · 子分类: 加载与思考
// 动画表现: 一圈圈点阵纬线从上到下依次鼓起又收回，像一道波在球面上滚过。
// 运动规格:
//   - 球半径 58 px，11 条纬线环（−75° 到 75°，每 15° 一条），每环 8–34 个点，按纬度增减。
//   - 各环半径最多起伏 ±10%（约 6 px）；每周期 2 个波，相邻两环相位差 0.7 rad。
//   - 鼓起的环更亮；球每周期自转一圈。
//   - 右侧文字“Listening…”15 px。
// 适用场景: 适合语音输入、语音模式、听写等开麦阶段。
// 标签: thinking-orb, listening, voice, microphone, dictation, wave
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#1-055
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 07
// 07 思考球·聆听（state="listening"）：纬线环上的点随声波起伏，波沿纬度方向滚动
CELLS[6] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC;
  for (let i = 0; i < 11; i++) {
    const lat = (-75 + 15 * i) * Math.PI / 180, cl = Math.cos(lat);
    const amp = 0.1 * Math.sin(2 * TAU * u / CYC - i * 0.7);   // 振幅 ≈ 6 px，2 个波 / 周期
    const n = Math.max(8, Math.round(34 * cl));
    for (let j = 0; j < n; j++) {
      const lon = j * TAU / n, rr2 = cl * (1 + amp);
      dot(k, rot([rr2 * Math.cos(lon), Math.sin(lat), rr2 * Math.sin(lon)], spin, ORB.tilt), 0.8, Math.max(0, amp) * 4);
    }
  }
  orbLabel(k, 'Listening…');
};
