// 外扩呼吸光晕 / Outer pulse
// 分类包: aiux (光效与液态) · element 04 · 子分类: 加载与思考
// 动画表现: 深色输入框周围，多层彩虹光圈向外扩散再收回，输入框本身保持清晰。
// 运动规格:
//   - 输入框 380 × 64 px，圆角 16 px；光只画在输入框外侧。
//   - 一次呼吸 2.4 秒（每周期 2 次），外扩距离 4→18 px。
//   - 叠 4 层光圈（3 px 宽、模糊 18 px），越往外越淡；不透明度 0.25→0.7。
//   - 彩虹渐变每周期旋转一圈。
// 适用场景: 适合空状态或首页的主输入框，也可用在推荐的价格方案卡片上。
// 标签: pulse, input, highlight, glow, landing, featured
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#1-052
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 04
// 04 外扩呼吸光晕（border-beam size="pulse-outside"）：彩色光晕在输入框外侧向外扩散再收回
CELLS[3] = (k, w, h, u) => {
  bg(k);
  const x = 79.5, y = 72, W = 380, H = 64, R = 16;
  const s = 0.5 - 0.5 * Math.cos(TAU * u / 2.4);
  const e = 4 + 14 * s, rot0 = TAU * u / CYC;
  for (let i = 4; i >= 1; i--) {
    const ex = e * i / 4;
    k.strokeStyle = conic(k, x + W / 2, y + H / 2, rot0, (0.25 + 0.45 * s) * (1.15 - i / 4));
    k.lineWidth = 3; k.shadowColor = hsla(200 + 360 * u / CYC, 100, 60, 0.5 * s); k.shadowBlur = 18;
    rr(k, x - ex, y - ex, W + ex * 2, H + ex * 2, R + ex); k.stroke();
  }
  k.shadowBlur = 0; k.shadowColor = 'transparent';
  field(k, x, y, W, H, R, 'Describe what you want to build…');
};
