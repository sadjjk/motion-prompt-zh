// 有机流动色块 / Organic blob
// 分类包: aiux (光效与液态) · element 15 · 子分类: 液态交互
// 动画表现: 深色背景中央，一团粉到紫渐变的色块轮廓不停蠕动，后面的淡影子按另一种节奏变形。
// 运动规格:
//   - 轮廓用极坐标描出：半径 = 基准 62 px + 3 组正弦波（2、3、5 瓣，振幅 6、5、3 px）。
//   - 这些正弦波每周期分别转 1、2、3 圈，方向交替，循环无缝。
//   - 后方影子放大 1.15 倍、错开 (10, 8) px，用另外 3 组正弦波（2、3、4 瓣，振幅 7、4、4 px）变形，不透明度 0.22。
//   - 色块是色相 330° 到 265° 的斜向渐变，中央放 22 px 文字。
// 适用场景: 适合 Hero 装饰、需要强调的标签背景、俏皮的点缀，想加一点柔软动感时。
// 标签: blob, organic, morph, liquid, wobble, shape, decoration
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#1-177
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 15
// 15 有机流动色块（改写自 KittenJam #326，Hannah，MIT）：极坐标半径 = 基准 62px + 3 组正弦谐波（各自整数圈速），
// 色块持续蠕动；后方一层放大 1.15 倍、错开 (10, 8) px 的淡影子用另一组谐波变形，节奏不同。
const OB_MAIN = [[2, 6, 1, 0.0], [3, 5, -2, 1.3], [5, 3, 3, 2.1]];     // [瓣数, 振幅 px, 每周期圈数, 相位]
const OB_SHADOW = [[2, 7, -1, 0.7], [3, 4, 2, 2.5], [4, 4, -3, 0.4]];
const OB_R = 62, OB_CX = 269.5, OB_CY = 104, OB_N = 96;
function obPath(k, cx, cy, R, waves, ph) {
  k.beginPath();
  for (let i = 0; i <= OB_N; i++) {
    const th = TAU * i / OB_N;
    let r = R;
    for (const [n, a, sp, p0] of waves) r += a * Math.sin(n * th + sp * ph + p0);
    const x = cx + r * Math.cos(th), y = cy + r * Math.sin(th);
    if (i) k.lineTo(x, y); else k.moveTo(x, y);
  }
  k.closePath();
}
CELLS[14] = (k, w, h, u) => {
  bg(k);
  const ph = TAU * u / CYC;
  k.fillStyle = hsla(300, 80, 62, 0.22);
  obPath(k, OB_CX + 10, OB_CY + 8, OB_R * 1.15, OB_SHADOW, ph); k.fill();
  const g = k.createLinearGradient(OB_CX - 70, OB_CY - 70, OB_CX + 70, OB_CY + 70);
  g.addColorStop(0, hsla(330, 90, 66, 1)); g.addColorStop(1, hsla(265, 85, 62, 1));
  k.fillStyle = g; obPath(k, OB_CX, OB_CY, OB_R, OB_MAIN, ph); k.fill();
  txt(k, '灵感', OB_CX, OB_CY + 1, `700 22px ${FONT}`, '#FFFFFF', 'center');
};
