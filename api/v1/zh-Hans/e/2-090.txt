// 立体字 / Extruded text
// 分类包: motion2 (动态图形进阶) · element 51 · 子分类: 文字与图形
// 动画表现: 白色正面、后面叠着渐暗珊瑚色层做出厚度的「立体」绕竖轴缓慢转一圈，转到侧面能看到厚度。
// 运动规格:
//   - 正面白色，后面 14 层珊瑚色阶（每层深 2.6 px，越深越暗）
//   - 每 4.7 秒绕竖轴匀速转一圈（scaleX = cos）
//   - 12° 俯角让各层略向上错位
//   - 转到背面时调换绘制顺序，最深一层在前
// 适用场景: 适合 Logo、标题或游戏风格的大字。
// 标签: extrude, 3d, depth, rotate, title, logo
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#2-090
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 51
// 改写自 KittenJam Motion Library #61（Hannah）· 立体字：「立体」由 14 层珊瑚色阶（每层深 2.6px，越深越暗）叠出厚度，正面白色；整组绕竖轴每 4.7 秒匀速转一圈（scaleX = cos），带 12° 俯角的上下错位，转到侧面能看到厚度。
CELLS[50] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '立体', F = `900 96px ${JP}`, th = Math.PI * 2 * u / CYC, c = Math.cos(th), s = Math.sin(th), TX = Math.tan(12 * Math.PI / 180);
  const sx = Math.abs(c) < 0.02 ? (c < 0 ? -0.02 : 0.02) : c;
  k.fillStyle = 'rgba(0,0,0,.3)'; k.beginPath(); k.ellipse(w / 2, 172, 130 * Math.max(0.3, Math.abs(c)), 7, 0, 0, 7); k.fill();
  const ids = []; for (let i = 14; i >= 0; i--) ids.push(i);   // 0 = 正面
  if (c < 0) ids.reverse();                                     // 转到背面时近处是最深一层
  ids.forEach(i => {
    const d = i * 2.6;
    k.save(); k.translate(w / 2 - d * s, h / 2 - 4 - d * TX); k.scale(sx, 1);
    txt(k, S, 0, 0, F, i ? `hsl(6,72%,${54 - i * 2}%)` : '#fff', 'center', 'middle');
    k.restore();
  });
};
