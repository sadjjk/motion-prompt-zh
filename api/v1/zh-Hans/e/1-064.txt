// 滚动吸附 / Scroll snap
// 分类: 界面与网页 / 页面 / 滚动
// 动画表现: 轻微滚动后，页面自动补完剩余距离，将下一分区顶边与视口顶部对齐。
// 运动规格:
//   - 每个分区与视口等高，为 168 px；第一屏浅蓝，第二屏深蓝。
//   - 每 2.4 秒吸附一次；0.45–0.85 秒用户仅滚动 34 px，采用三次缓出。
//   - 0.9–1.5 秒页面自动补完剩余 134 px，对齐下一分区顶部，采用三次缓出。
//   - 第二个 2.4 秒同样向上返回首屏，对应 scroll-snap-type: y mandatory。
// 适用场景: 适合像幻灯片一样逐屏阅读的页面。
// 标签: scroll-snap, snap, full-page, sections, css, mandatory, scroll
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖的公共代码: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#1-064
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为本效果代码，CELLS、常量和辅助函数位于依赖的公共代码中。

// @element 06
// 06 スクロールスナップ（少しスクロールすると、次のセクションの頭までピタッと送られる）
CELLS[5] = (k, w, h, u) => {
  frame(k);
  const T = CYC / 2, lu = mod(u, T), dir = u < T ? 1 : -1, from = u < T ? 0 : VH, DRIFT = 34;
  const y = from + dir * (DRIFT * eOut(P(lu, 0.45, 0.85)) + (VH - DRIFT) * eOut(P(lu, 0.9, 1.5)));
  view(k, () => {
    k.save(); k.translate(0, -y);
    for (let i = 0; i < 2; i++) {
      const y0 = i * VH, dark = i === 1;
      k.fillStyle = dark ? C.ink : C.blueL; k.fillRect(0, y0, VW, VH);
      txt(k, '0' + (i + 1), 34, y0 + 70, `800 46px ${EN}`, C.acc);
      txt(k, dark ? '上手试用' : '了解产品', 34, y0 + 102, `800 18px ${JP}`, dark ? '#fff' : C.ink);
      bars(k, 34, y0 + 118, 170, 2, dark ? 'rgba(255,255,255,.3)' : '#C3CDEA');
      pic(k, 262, y0 + 24, 220, 120, dark ? 2 : 0, 8);
    }
    k.restore();
    sbar(k, y / VH, 0.5);
  });
};
