// 思考球·搜索 / Thinking orb: searching
// 分类包: aiux (AI 界面动效) · element 06 · 子分类: 加载与思考
// 动画表现: 点阵地球自转，一条竖向光线在表面扫过，经过的点随之变亮。
// 运动规格:
//   - 球半径 58 px，240 个点；每周期自转一圈。
//   - 扫描经线每周期转 2 圈（是自转的 2 倍速），线宽 1 px、不透明度 0.35。
//   - 离经线越近的点越亮，按角度差 Δ 以 exp(−Δ²/0.05) 衰减，最多加亮 0.75。
//   - 右侧文字“Searching the web…”15 px。
// 适用场景: 适合联网搜索、检索资料、查询数据等“正在找”的阶段。
// 标签: thinking-orb, searching, search, retrieval, loading, globe
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// 动画演示: https://sadjjk.github.io/motion-prompt-zh/?lang=zh-Hans#1-054
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 06
// 06 思考球·搜索（state="searching"）：点阵地球自转，一条亮经线扫过，扫到的点变亮
CELLS[5] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC, sweep = 2 * TAU * u / CYC;   // 自转 1 圈、扫描 2 圈 / 周期
  for (const p of SPHERE) {
    const q = rot(p, spin, 0), lon = Math.atan2(q[0], q[2]);
    const dd = wrapPi(lon - sweep), boost = 0.75 * Math.exp(-(dd * dd) / 0.05);
    dot(k, rot(p, spin, ORB.tilt), 0.7, boost);
  }
  k.strokeStyle = 'rgba(236,236,241,0.35)'; k.lineWidth = 1; k.beginPath();
  for (let i = 0; i <= 40; i++) {
    const lat = -Math.PI / 2 + Math.PI * i / 40;
    const q = rot([Math.cos(lat) * Math.sin(sweep), Math.sin(lat), Math.cos(lat) * Math.cos(sweep)], 0, ORB.tilt);
    const X = ORB.x + q[0] * ORB.r * 1.02, Y = ORB.y - q[1] * ORB.r * 1.02;
    if (i) k.lineTo(X, Y); else k.moveTo(X, Y);
  }
  k.stroke();
  orbLabel(k, 'Searching the web…');
};
