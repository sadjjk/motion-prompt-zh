// Photo stack shuffle / フォトスタックシャッフル
// Pack: lp (Landing page & web effects) · element 33 · Category: Hover & click interactions
// What happens: Each time the cursor clicks a pile of 4 tilted polaroids, the top one slides out to the right and goes to the bottom while the rest move up a place.
// Motion spec:
//   - A pile of 4 tilted polaroids (96×112) whose slots are angled −3°, 4°, −7° and 6°.
//   - The cursor clicks the top card at 0.6, 1.5, 2.4 and 3.3 s; each shuffle starts 0.05 s after the click and lasts 0.75 s.
//   - The top card slides 170px to the right and turns 10° more with easeOut power 3 over the first 0.36 s, then tucks into the bottom slot over the remaining 0.39 s (easeInOut power 3).
//   - The other 3 cards each move up one slot at the same time (easeInOut power 3); after 4 shuffles the order is back to the start, matching the first frame.
// When to use it: Use it in travel stories, albums or portfolios to flip through photos one at a time.
// Tags: photo-stack, polaroid, shuffle, cards, gallery, click
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-125
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 33
// 改写自 KittenJam Motion Library #270 拍立得洗牌：一叠 4 张歪斜的拍立得（96×112，各槽位角度 −3°/4°/−7°/6°），
// 光标在 0.6、1.5、2.4、3.3 秒点顶张：顶张前 0.36 秒 easeOut 向右抽出 170px 并多转 10°，后 0.39 秒插回最底层槽位；
// 其余 3 张同时 easeInOut 3 次方各上移一个槽位。4 次后顺序回到初始，周期首尾一致。
const PSTACK_SLOT = [[0, 0, -3], [7, 4, 4], [-6, 7, -7], [4, 10, 6]];
const PSTACK_CLICKS = [0.6, 1.5, 2.4, 3.3], PSTACK_DUR = 0.75;
CELLS[32] = (k, w, h, u) => {
  frame(k, 'example.com/memories');
  const [cx, cy] = path([[0, 430, 150], [0.45, 205, 70], [3.5, 205, 70], [4.2, 430, 150], [4.8, 430, 150]], u);
  let n = 0, p = -1;
  for (const tc of PSTACK_CLICKS) if (u >= tc + 0.05) { if (u < tc + 0.05 + PSTACK_DUR) p = (u - tc - 0.05) / PSTACK_DUR; else n++; }
  const ox = 190, oy = 84;
  const pola = (card, x, y, deg) => {
    k.save(); k.translate(ox + x, oy + y); k.rotate(deg * Math.PI / 180);
    k.shadowColor = 'rgba(28,35,51,.2)'; k.shadowBlur = 10; k.shadowOffsetY = 3;
    k.fillStyle = '#fff'; k.fillRect(-48, -56, 96, 112); k.shadowColor = 'transparent';
    pic(k, -42, -50, 84, 80, card, 2);
    bars(k, -38, 38, 50, 1, C.faint, 5);
    txt(k, `0${card + 1}`, 34, 42, `700 9px ${EN}`, C.sub, 'right', 'middle');
    k.restore();
  };
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    txt(k, '旅行回忆', 330, 70, `800 20px ${JP}`, C.ink); bars(k, 330, 86, 140, 2);
    // 槽位 j 上的卡 = (j + n) % 4；动画中槽位 0 的卡在抽出
    const draws = [];
    for (let j = 3; j >= 0; j--) {
      const card = (j + n) % 4;
      if (p < 0) { draws.push([card, ...PSTACK_SLOT[j]]); continue; }
      if (j === 0) continue;
      const f = eIO(clamp(p / 0.9)), A = PSTACK_SLOT[j], B = PSTACK_SLOT[j - 1];
      draws.push([card, lerp(A[0], B[0], f), lerp(A[1], B[1], f), lerp(A[2], B[2], f)]);
    }
    if (p >= 0) {
      const top = n % 4, S0 = PSTACK_SLOT[0], S3 = PSTACK_SLOT[3];
      if (p < 0.48) { const f = eOut(p / 0.48); draws.push([top, S0[0] + 170 * f, S0[1], S0[2] + 10 * f]); }
      else { const f = eIO((p - 0.48) / 0.52); draws.unshift([top, lerp(S0[0] + 170, S3[0], f), S3[1], lerp(S0[2] + 10, S3[2], f)]); }
    }
    draws.forEach(d => pola(...d));
    click(k, cx, cy, u, PSTACK_CLICKS);
  });
};
