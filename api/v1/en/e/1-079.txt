// Dot grid ripple / ドットグリッドの波紋
// Pack: uxmotion (Interaction & data motion) · element 19 · Category: Hero backgrounds
// What happens: As the pointer crosses the calm grey grid, rings spread from its trail; dots on the wave swell, lift and shift from blue-violet to teal.
// Motion spec:
//   - A dot grid at 17 px spacing; at rest the dots are grey at 55% opacity.
//   - The pointer drops a small ripple every 110 ms (amplitude .55) and a click a big wave (amplitude 1.6); up to 14 run at once.
//   - Wavefronts spread at 170 px/s with shape sin(k/14)·e^(k/60), decaying by e^(−1.3·age) and gone after 2.6 s.
//   - Raised dots grow to radius 1.6 + m × 3.4, lift m × 3 px and shift hue 230 − m × 110 from blue-violet to teal; demo clicks at 2.4 s.
// When to use it: Use it as a hero or section background on product sites to add mouse-responsive feedback to a calm page.
// Tags: dot-grid, ripple, wave, hero, background, cursor, interactive
// Size: 539x208 px · Cycle: 6.4 s
// Needs: kit uxmotion -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/uxmotion.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-079
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 点阵涟漪：17px 间距的点阵（静止时 ink-3、55% 透明）；指针划过时每 110ms 在经过处投下一圈小涟漪（振幅 .55），2.4 秒按下投下大波（振幅 1.6）；
// 波前以 170 px/s 扩散，波形 sin(k/14)·e^(k/60)，按 e^(−1.3·年龄) 衰减、2.6 秒消失，同时最多 14 道；隆起的点变大（1.6+m×3.4）、上浮 m×3 并由蓝紫转到青绿（色相 230−m×110）。上一周期的余波计入，循环无缝
const DOTW_GAP = 17;
const DOTW_DOTS = (() => { const d = []; for (let y = 12; y < CH - 4; y += DOTW_GAP) for (let x = 14; x < CW - 4; x += DOTW_GAP) d.push([x, y]); return d; })();
const DOTW_ptr = u => UXA_path([[0, 590, 240], [0.4, 520, 170], [1.4, 360, 60], [2.4, 270, 110], [2.6, 270, 110], [3.6, 130, 150], [4.4, 60, 60], [5.0, -40, -40], [6.4, -40, -40]], u);
const DOTW_WAVES = (() => {
  const W = [];
  for (let t = 0.45; t < 4.95; t += 0.11) { const [x, y] = DOTW_ptr(t); if (x > 0 && x < CW && y > 0 && y < CH) W.push([t, x, y, 0.55]); }
  W.push([2.4, 270, 110, 1.6]);
  return W.sort((a, b) => a[0] - b[0]);
})();
CELLS[18] = (k, w, h, u) => {
  k.fillStyle = UX.surface; k.fillRect(0, 0, CW, CH);
  // 当前存活的波（含上一周期，年龄 < 2.6 秒），只保留最近 14 道
  const live = [];
  for (const wv of DOTW_WAVES) for (const off of [0, CYC]) { const age = u - wv[0] + off; if (age >= 0 && age < 2.6) live.push([age, wv[1], wv[2], wv[3]]); }
  live.sort((a, b) => a[0] - b[0]); live.length = Math.min(live.length, 14);
  for (const d of DOTW_DOTS) {
    let hsum = 0;
    for (const [age, x, y, a] of live) {
      const kk = Math.hypot(d[0] - x, d[1] - y) - age * 170;
      if (kk > 0 || kk < -90) continue;
      hsum += Math.sin(kk / 14) * Math.exp(kk / 60) * a * Math.exp(-age * 1.3);
    }
    const m = clamp(hsum, -1, 1.6), calm = Math.abs(m) < 0.04;
    k.globalAlpha = calm ? 0.55 : 1;
    k.fillStyle = calm ? UX.ink3 : `hsl(${(230 - Math.max(0, m) * 110).toFixed(0)} 85% ${(58 - Math.min(0, m) * 10).toFixed(0)}%)`;
    k.beginPath(); k.arc(d[0], d[1] - m * 3, 1.6 + Math.max(0, m) * 3.4, 0, TAU); k.fill();
  }
  k.globalAlpha = 1;
  const [px, py] = DOTW_ptr(u);
  cursor(k, px, py, pingPong(u, 2.3, 2.4, 2.42, 2.56));
};
