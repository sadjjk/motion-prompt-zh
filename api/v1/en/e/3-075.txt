// Pack: photo (Photo looks, lighting & styles) · element 39 · Category: Lighting & time of day
// What happens: The room darkens and a torch in front of the chest lights the person from below; the head is bright at the bottom and darker towards the top, and a large shadow stretches up the wall behind.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure (head centre x 270, y 78, radius 29 px); the background turns to a dark scene and wall #34363E.
//   - Light: radial from under the chin at (270, 129); #FFE6B0 at the centre → #FFC75A → shadow #15161B (radius about 75 px).
//   - Wall shadow #1C1D22: the figure shape scaled 1.45× wide and 1.75× tall, stretching upwards.
//   - A torch (10 px wide) in front of the chest; the plan shows the light in front and below; top right reads "0° · ↓60°".
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, ease-in-out).
// When to use it: For horror, ghost stories or Halloween when something should feel eerie or wrong.
// Tags: under lighting, monster lighting, horror, flashlight, uplight, lighting
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-075
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 39
// 底光（アンダーライト）: 周りが暗くなり、胸の前の懐中電灯があごの下から上へ照らす。頭は下半分が明るく上ほど暗い。
// 後ろの壁に、上へ伸びて大きくなった影が出る
CELLS[38] = cell((k, u) => {
  const a = env(u), D = pen(k);
  stagePortrait(k, pal(BASE, DARK, a));
  // 壁と影: 暗い壁を重ね、胸像の形を上へ大きく伸ばした影を落とす
  k.save(); k.globalAlpha = a; D.rect(0, 0, CW, CH, '#34363E');
  k.translate(PX, PY + 70); k.scale(1.45, 1.75); k.translate(-PX, -(PY + 70) - 14);
  bustPath(k, PX, PY, PR); k.fillStyle = '#1C1D22'; k.fill();
  k.restore();
  const lit = BASE.lit, sh = mixc(BASE.person, '#15161B', a), FY = PY + PR * 1.75;
  bust(k, PX, PY, PR, mixc(BASE.person, sh, 0.6 * a), kk => {
    // 光の中心はあごの下（懐中電灯の位置）。上へ行くほど暗くなる
    const g = kk.createRadialGradient(PX, FY, 4, PX, FY, PR * 2.6);
    g.addColorStop(0, mixc(BASE.person, '#FFE6B0', a)); g.addColorStop(0.45, mixc(BASE.person, lit, a)); g.addColorStop(1, sh);
    kk.fillStyle = g; kk.fillRect(0, 0, CW, CH);
  });
  // 懐中電灯と光の点
  k.save(); k.globalAlpha = a;
  D.rr(PX - 5, FY + 2, 10, 26, 3, '#2A2C33'); D.circ(PX, FY + 2, 5.5, '#FFF4D6');
  k.globalCompositeOperation = 'lighter'; k.globalAlpha = 0.35 * a; D.circ(PX, FY, 14, '#FFE3A3'); k.restore();
  plan(k, [[0, a]]);
  hud(k, '0° · ↓60°');
});
