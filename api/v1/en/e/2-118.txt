// Delay / ディレイ
// Category: Motion graphics / Motion principles / Entrances & easing
// What happens: Four bars wait 0, 0.15, 0.3 and 0.45 s before they start; the squares on the right slide in in the same order.
// Motion spec:
//   - One 4.7 s cycle. The timeline starts at 0.5 s; a white playhead moves at 1 s = 170 px.
//   - The four rows have delays of 0, 0.15, 0.3 and 0.45 s: a dotted line while waiting, then a 0.5 s bar.
//   - Each square on the right (110×30 px) slides 30 px from the left and fades in over 0.5 s (cubic ease-out).
//   - Everything fades out at 3.5–3.9 s before the next cycle.
// When to use it: Use it to bring in lists or cards one after another instead of all at once.
// Tags: delay, stagger, transition-delay, sequence, timing, offset, list
// Size: 539x208 px · Cycle: 4.7 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-118
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 67
// 67 动画延迟（4 个方块各等 0 / 0.15 / 0.3 / 0.45 秒再开始，每个都动 0.5 秒）
const DL_S = 0.5, DL_DL = [0, 0.15, 0.3, 0.45], DL_D = 0.5, DL_OUT = [3.5, 3.9], DL_SC = 170;
CELLS[66] = (k, w, h, u) => {
  stage(k);
  const t = u - DL_S, gx = 132, fade = 1 - P(u, DL_OUT[0], DL_OUT[1]);
  // 时间轴刻度（1 秒 = 170px）
  [0, 0.5, 1].forEach(s => { const x = gx + s * DL_SC; k.fillStyle = C.navy2; k.fillRect(x - 0.5, 18, 1, 166); txt(k, s + ' 秒', x, 198, `500 11px ${JP}`, C.dim, 'center', 'middle'); });
  DL_DL.forEach((dl, i) => {
    const y = 34 + i * 44, x = gx + dl * DL_SC, p = P(t, dl, dl + DL_D);
    txt(k, dl + ' 秒', 26, y + 1, `700 18px ${JP}`, PAL[i], 'left', 'middle');
    // 等待段（虚线）+ 动画段（条）
    if (dl > 0) { k.strokeStyle = C.navy3; k.lineWidth = 2; k.setLineDash([4, 5]); k.beginPath(); k.moveTo(gx, y); k.lineTo(x, y); k.stroke(); k.setLineDash([]); }
    k.fillStyle = C.navy2; rr(k, x, y - 7, DL_D * DL_SC, 14, 7); k.fill();
    if (p > 0 && fade > 0) { k.globalAlpha = fade; k.fillStyle = PAL[i]; rr(k, x, y - 7, Math.max(14, DL_D * DL_SC * p), 14, 7); k.fill(); k.globalAlpha = 1; }
    // 右侧方块：向右滑入 30px 并淡入（ease-out）
    const e = eOut(p);
    if (e * fade > 0) { k.globalAlpha = e * fade; k.fillStyle = PAL[i]; rr(k, 400 - 30 * (1 - e), y - 15, 110, 30, 8); k.fill(); k.globalAlpha = 1; }
  });
  // 播放头
  const end = DL_DL[3] + DL_D, pa = t < 0 ? 0 : 1 - P(t, end + 0.2, end + 0.5);
  if (pa > 0) { const x = gx + Math.min(t, end) * DL_SC; k.globalAlpha = pa; k.fillStyle = C.white; k.fillRect(x - 1, 14, 2, 172); k.beginPath(); k.arc(x, 14, 4, 0, 7); k.fill(); k.globalAlpha = 1; }
};
