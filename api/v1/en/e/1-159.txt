// Direction-aware fill button / 方向認識フィルボタン
// Pack: lp (Landing page & web effects) · element 44 · Category: Hover & click interactions
// What happens: On a dark page the cursor enters an outlined button from the lower left and an orange circle spreads from that point to fill it; when the cursor leaves upward, the circle shrinks away toward the exit point.
// Motion spec:
//   - A 200×52 pill button with an accent outline on a dark page.
//   - The cursor enters from the lower left; an accent circle centred on the entry point grows from 0 to the radius that covers the farthest corner over 0.55 s (easeInOut power 3), and the label turns from accent to white.
//   - After 3.0 s the cursor leaves upward; the circle re-centres on the exit point and shrinks away over 0.55 s, so the colour drains toward the exit.
// When to use it: Use it on contact or sign-up CTAs where the button should visibly respond to the cursor's direction.
// Tags: button, direction-aware, fill, hover, circle, cta
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-159
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 44
// 改写自 KittenJam Motion Library #103 光标扩散填满按钮：深色页面上 200×52 朱色描边胶囊按钮。光标从左下方进入按钮，
// 朱色圆以「进入点」为圆心、0.55 秒 easeInOut 3 次方从 0 放大到能盖住最远角的半径，文字由朱色变白；
// 3.0 秒后光标往上方离开，圆改以「离开点」为圆心 0.55 秒缩小消失，颜色往出口方向收回。
const DFILL_W = 200, DFILL_H = 52;
CELLS[43] = (k, w, h, u) => {
  frame(k, 'example.com/contact');
  const bx = (VW - DFILL_W) / 2, by = (VH - DFILL_H) / 2;
  const cur = t => path([[0, 40, 150], [0.9, 250, 86], [3.0, 262, 84], [3.5, 320, 16], [4.2, 40, 16], [4.8, 40, 150]], t);
  const on = t => { const [a, b] = cur(t); return a > bx && a < bx + DFILL_W && b > by && b < by + DFILL_H; };
  // 圆心 = 最后一次进入或离开按钮那一刻的光标位置
  const q = trans(on, u, 0.55, eIO), ft = flipAt(on, u);
  const [ox, oy] = ft === null ? [bx, by] : cur(ft);
  const R = Math.max(...[[bx, by], [bx + DFILL_W, by], [bx, by + DFILL_H], [bx + DFILL_W, by + DFILL_H]].map(([a, b]) => Math.hypot(a - ox, b - oy)));
  view(k, () => {
    k.fillStyle = '#151A26'; k.fillRect(0, 0, VW, VH);
    if (q > 0) { k.save(); rr(k, bx, by, DFILL_W, DFILL_H, DFILL_H / 2); k.clip(); k.fillStyle = C.acc; k.beginPath(); k.arc(ox, oy, R * q, 0, 7); k.fill(); k.restore(); }
    k.strokeStyle = C.acc; k.lineWidth = 2; rr(k, bx, by, DFILL_W, DFILL_H, DFILL_H / 2); k.stroke();
    txt(k, '立即联系', VW / 2, VH / 2 + 1, `700 16px ${JP}`, mixHex(C.acc, '#FFFFFF', q), 'center', 'middle', '2px');
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
