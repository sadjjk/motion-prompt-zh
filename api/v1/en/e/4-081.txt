// Stippling / 点描
// Category: Visual styles / Craft & patterns / Print & handmade
// What happens: An egg and its shadow on cream paper. Dots go in from the darkest areas to the lightest, with the pen tip following the latest dot and fading away once it is done.
// Motion spec:
//   - Cream paper. An egg 108 × 140 px, narrower at the top, centred at (270, 96), lit from the upper left, with an elliptical shadow on the ground to the lower right.
//   - The dots are worked out once in advance: a 2.2 px grid with ±0.9 px jitter over x 150–390 and y 18–204 keeps each dot with a probability of tone^1.2 × 0.9; the egg's rim is darkened so its outline reads even on the lit side.
//   - 0.3–4.8 s: the dots, 0.85 px in radius, appear in 6 tone bands from darkest to lightest, each band swept back and forth in rows 10 px tall, so the shadow side builds up first and the pen moves in patches.
//   - A pen tip sits on the latest dot while it works and fades out at 4.8–5.1 s.
//   - At 5.4–6.0 s the drawing fades back to blank paper; at 0 s and 6.4 s the card is blank paper, so the loop restarts cleanly.
// When to use it: Use it for scientific, natural-history or medical illustration, or to show patience and attention to detail.
// Tags: drawing, stipple, dots, pen, tone, process
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/drawstyles.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-081
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 10
// 点描：一支针管笔，从最暗处开始一点点点出一只蛋和它的投影；明暗只靠点的疏密，最亮处几乎不落点
{
  const INK = '#20201E', CX = 270, CY = 96, RX = 54, RY = 70;
  // 蛋形：上窄下宽，横半轴随纵向位置变化
  // 边缘再加深一圈（q>0.8 起），否则亮面一侧几乎没有点、蛋形认不出来
  const eggTone = (x, y) => {
    const ny = (y - CY) / RY, rx = RX * (1 + 0.12 * ny), t = sphereTone(x, y, CX, CY, rx, RY);
    if (t < 0) return t;
    const nx = (x - CX) / rx, q = nx * nx + ny * ny;
    return Math.min(1, Math.max(t, 0.14) + 0.45 * seg(q, 0.8, 1));
  };
  const tone = (x, y) => {
    const t = eggTone(x, y);
    if (t >= 0) return t;
    const dx = (x - 304) / 84, dy = (y - 180) / 13, d2 = dx * dx + dy * dy;
    return d2 < 1 ? 0.7 * (1 - d2) : 0;
  };
  // 点阵在定义时算好：2.2 px 网格加抖动，按色调概率保留，再按从暗到亮排序（计划为 3 px，截图后加密）
  const DOTS = [];
  let id = 0;
  for (let y = 18; y <= 204; y += 2.2) for (let x = 150; x <= 390; x += 2.2, id++) {
    const px = x + (rnd(id, 1) - 0.5) * 1.8, py = y + (rnd(id, 2) - 0.5) * 1.8, t = tone(px, py);
    if (t > 0 && rnd(id, 7) < Math.pow(t, 1.2) * 0.9) DOTS.push([px, py, t, id]);
  }
  // 只按色调排序时，相邻两点常在蛋的两头，笔尖每帧乱跳；
  // 改为色调分 6 档从暗到亮，每档内按 10 px 行蛇形扫过，笔尖成片移动
  const band = d => Math.min(5, Math.floor(d[2] * 6)), row = d => Math.floor(d[1] / 10);
  DOTS.sort((a, b) => band(b) - band(a) || row(a) - row(b) || (row(a) % 2 ? b[0] - a[0] : a[0] - b[0]));
  CELLS[9] = (k, w, h, u) => {
    paper(k, '#F6F2EA', 'rgba(60,50,30,.05)');
    const a = vis(u), n = Math.floor(seg(u, 0.3, 4.8) * DOTS.length);
    k.globalAlpha = a; k.fillStyle = INK;
    for (let i = 0; i < n; i++) { k.beginPath(); k.arc(DOTS[i][0], DOTS[i][1], 0.85, 0, TAU); k.fill(); }
    // 针管笔：停在刚点下的那个点上，画完后淡出
    const pa = (1 - seg(u, 4.8, 5.1)) * (n > 0 ? 1 : 0);
    if (pa > 0) {
      const [px, py] = DOTS[Math.min(n, DOTS.length - 1)];
      k.globalAlpha = a * pa * 0.8; k.strokeStyle = INK; k.lineCap = 'round';
      k.lineWidth = 2; k.beginPath(); k.moveTo(px + 2, py - 2); k.lineTo(px + 26, py - 26); k.stroke();
      k.lineWidth = 5; k.beginPath(); k.moveTo(px + 14, py - 14); k.lineTo(px + 26, py - 26); k.stroke();
    }
    k.globalAlpha = 1;
  };
}
