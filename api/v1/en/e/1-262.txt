// Grain gradient / 粒子グラデーション
// Category: UI & Web / Pages / Hero backgrounds
// What happens: Colour bands shading from dark purple through magenta and orange to cream keep changing shape as they drift. A fine grain covers everything, and the colours speckle into each other where the bands meet.
// Motion spec:
//   - No simulation: the band coordinate is a diagonal ramp (0.26 per unit across, −0.3 per unit down) plus 0.9 × and 0.5 × two layers of value-noise fbm.
//   - The first noise layer's sample centre moves once around a circle of radius 0.9 per 6.4 s cycle; the second moves twice the other way around a circle of radius 0.6 and is warped again by the first, so the loop joins up.
//   - Warping is computed on a 135×52 grid and scaled up; grain is per pixel.
//   - Grain: the band coordinate gets ±0.05 of fixed noise before the colour lookup, then brightness is multiplied by 1 ± 9%.
//   - Colours #0D0A1C, #2B1660, #6A2AA8, #D2508F, #FF9A6B, #FFE2B0, interpolated in OKLab.
// When to use it: Calm landing-page heroes, product introductions or portfolio backgrounds.
// Tags: gradient, grain, noise, domain-warp, hero-background, shader
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/webfx.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-262
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 38
// 38 颗粒渐变：几条色带被缓慢流动的噪声扭曲，整面带磨砂相纸一样的颗粒，色带交界处的颗粒让两种颜色互相渗点。没有模拟，画面直接由 u 算出。
// 色带坐标 = 沿斜向的位置 + 两层扭曲噪声；第一层噪声的取样中心沿半径 0.9 的圆每周期转一圈，第二层沿半径 0.6 的圆反向转两圈并被第一层再扭一次，
// 所以画面一直在流、周期结束回到原样。扭曲在 135×52 的粗网格上算再放大；颗粒按像素：色带坐标加 ±0.05 的固定噪点后查 OKLab 色表，再乘 ±9% 的亮度噪点。
const GG_W = 135, GG_H = 52;
const GG_PAL = ['#0D0A1C', '#2B1660', '#6A2AA8', '#D2508F', '#FF9A6B', '#FFE2B0'];
const GG_MIX = 0.1, GG_GRAIN = 0.18;                  // 色带坐标噪点幅度、亮度噪点幅度（峰峰值）
let GG = null;
CELLS[37] = (k, w, h, u) => {
  if (!GG) GG = { ramp: labRamp(GG_PAL), n1: grainTab(17.3), n2: grainTab(91.7), g: new Float32Array(GG_W * GG_H), up: new Float32Array(CW * CH), px: pixels() };
  const { ramp, n1, n2, g, up } = GG, out = GG.px.buf, a = TAU * u / CYC;
  const ax = 0.9 * Math.cos(a), ay = 0.9 * Math.sin(a), bx = 0.6 * Math.cos(1 - 2 * a), by = 0.6 * Math.sin(1 - 2 * a);
  for (let j = 0; j < GG_H; j++) for (let i = 0; i < GG_W; i++) {
    const x = i / (GG_H - 1), y = j / (GG_H - 1);       // 以画面高度为 1
    const w1 = fbm(1.2 * x + ax + 3.1, 1.2 * y + ay + 7.7), w2 = fbm(2.3 * x + bx + 2 * w1, 2.3 * y + by + 2 * w1);
    g[j * GG_W + i] = 0.5 + 0.26 * (x - 1.3) - 0.3 * (y - 0.5) + 0.9 * (w2 - 0.5) + 0.5 * (w1 - 0.5);
  }
  upGrid(g, GG_W, GG_H, 1, up);
  for (let p = 0; p < CW * CH; p++) {
    const t = clamp(up[p] + GG_MIX * (n1[p] - 0.5)), q = 3 * Math.round(t * 255), b = 1 + GG_GRAIN * (n2[p] - 0.5);
    out[3 * p] = ramp[q] * b; out[3 * p + 1] = ramp[q + 1] * b; out[3 * p + 2] = ramp[q + 2] * b;
  }
  GG.px.show(k);
};
