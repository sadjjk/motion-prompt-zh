// Jelly button / ゼリーボタン
// Pack: lp (Landing page & web effects) · element 31 · Category: Hover & click interactions
// What happens: The cursor presses the button twice; each time it squashes wide, stretches tall and wobbles back, while the caption counts the items added.
// Motion spec:
//   - A 160×50 vermilion pill button; the cursor clicks it at 1.0 s and 2.8 s.
//   - After each click the scale goes (1, 1) → (1.18, 0.82) at 0.12 s → (0.85, 1.15) at 0.27 s → (1.05, 0.95) at 0.42 s → back to (1, 1) at 0.6 s, each step easeInOut power 3.
//   - The caption under the button counts the clicks ("Added 1 item", "Added 2 items"); squashing wide before stretching tall keeps the bounce readable.
// When to use it: Use it on add-to-cart, like or send buttons where a press should feel physical.
// Tags: jelly, squash-stretch, button, click, bounce, feedback
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-123
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 31
// 改写自 KittenJam Motion Library #233 果冻弹跳按钮：160×50 朱色胶囊按钮，光标在 1.0、2.8 秒各点一次。
// 每次点击后按关键帧缩放 (1,1) → 0.12 秒 (1.18,0.82) → 0.15 秒 (0.85,1.15) → 0.15 秒 (1.05,0.95) → 0.18 秒回到 (1,1)，每段 easeInOut 3 次方，像果冻被压扁后回弹。
const JELLY_KEYS = [[0, 1, 1], [0.12, 1.18, 0.82], [0.27, 0.85, 1.15], [0.42, 1.05, 0.95], [0.6, 1, 1]];
const JELLY_AT = (d) => {
  if (d <= 0 || d >= 0.6) return [1, 1];
  for (let i = 1; i < JELLY_KEYS.length; i++) if (d < JELLY_KEYS[i][0]) { const a = JELLY_KEYS[i - 1], b = JELLY_KEYS[i], f = eIO((d - a[0]) / (b[0] - a[0])); return [lerp(a[1], b[1], f), lerp(a[2], b[2], f)]; }
  return [1, 1];
};
CELLS[30] = (k, w, h, u) => {
  frame(k, 'example.com/shop');
  const [cx, cy] = path([[0, 430, 150], [0.7, 262, 88], [3.4, 262, 88], [4.2, 430, 150], [4.8, 430, 150]], u);
  let sx = 1, sy = 1;
  for (const tc of [1.0, 2.8]) { const [a, b] = JELLY_AT(u - tc); sx *= a; sy *= b; }
  const n = (u >= 1.0 ? 1 : 0) + (u >= 2.8 ? 1 : 0);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(VW / 2, VH / 2); k.scale(sx, sy);
    k.save(); k.shadowColor = 'rgba(255,90,44,.35)'; k.shadowBlur = 14; k.shadowOffsetY = 5; k.fillStyle = C.acc; rr(k, -80, -25, 160, 50, 25); k.fill(); k.restore();
    txt(k, '加入购物车', 0, 1, `800 15px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    txt(k, n ? `已加入 ${n} 件` : '点一下试试', VW / 2, VH / 2 + 50, `600 11px ${JP}`, n ? C.acc : C.sub, 'center', 'middle');
    click(k, cx, cy, u, [1.0, 2.8]);
  });
};
