// Analog clock / アナログ時計
// Pack: lp (Landing page & web effects) · element 47 · Category: Loading, navigation & ambient loops
// What happens: On a white face with 60 ticks, the hour and minute hands stay at 10:10 while the orange second hand jumps from major tick to major tick, overshooting slightly each time.
// Motion spec:
//   - A 140px clock face with 60 ticks (every 5th thicker and longer); the hour and minute hands rest at a sample time of 10:10 instead of reading the real time.
//   - The accent second hand jumps to the next major tick (30°) every 0.4 s.
//   - Each jump takes 0.15 s with back-out (overshoot 1.7), passing slightly beyond the tick and settling back.
//   - 12 jumps per cycle make exactly one turn, so the loop is seamless.
// When to use it: Use it as an accent on screens about time, such as opening hours, time zones or countdowns.
// Tags: clock, analog, second-hand, tick, overshoot, loop
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-162
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 47
// 改写自 KittenJam Motion Library #323 指针时钟：直径 140 的表盘，60 个刻度（每 5 格较粗较长）；时针、分针停在示意时间 10:10（不读真实时间）。
// 朱色秒针每 0.4 秒跳到下一个粗刻度（30°），每跳 0.15 秒 back-out（过冲 1.7）略超过再弹回；一周期跳 12 次刚好一圈，首尾无缝。
const CLK_R = 70, CLK_STEP = 0.4;
const CLK_BACK = x => { const s = 1.7; return 1 + (s + 1) * Math.pow(x - 1, 3) + s * Math.pow(x - 1, 2); };
CELLS[46] = (k, w, h, u) => {
  frame(k, 'example.com/clock');
  const n = Math.floor(u / CLK_STEP), f = clamp((u - n * CLK_STEP) / 0.15);
  const sa = (n + CLK_BACK(f)) * Math.PI / 6;   // 秒针角度（0 = 12 点方向）
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    const ox = VW / 2, oy = VH / 2;
    k.save(); k.shadowColor = 'rgba(28,35,51,.16)'; k.shadowBlur = 14; k.shadowOffsetY = 4; k.fillStyle = '#fff'; k.beginPath(); k.arc(ox, oy, CLK_R, 0, 7); k.fill(); k.restore();
    k.strokeStyle = C.ink; k.lineWidth = 3; k.beginPath(); k.arc(ox, oy, CLK_R, 0, 7); k.stroke();
    for (let i = 0; i < 60; i++) {
      const a = i * Math.PI / 30, big = i % 5 === 0, r0 = CLK_R - (big ? 13 : 8);
      k.strokeStyle = big ? C.ink : C.sub; k.lineWidth = big ? 2.4 : 1;
      k.beginPath(); k.moveTo(ox + Math.sin(a) * r0, oy - Math.cos(a) * r0); k.lineTo(ox + Math.sin(a) * (CLK_R - 5), oy - Math.cos(a) * (CLK_R - 5)); k.stroke();
    }
    const hand = (a, len, tail, lw, col) => {
      k.strokeStyle = col; k.lineWidth = lw; k.lineCap = 'round';
      k.beginPath(); k.moveTo(ox - Math.sin(a) * tail, oy + Math.cos(a) * tail); k.lineTo(ox + Math.sin(a) * len, oy - Math.cos(a) * len); k.stroke();
    };
    hand((10 + 10 / 60) * Math.PI / 6, 36, 6, 5, C.ink);   // 时针 10:10
    hand(10 * Math.PI / 30, 52, 8, 3.5, C.ink);             // 分针 10 分
    hand(sa, 58, 14, 1.8, C.acc);
    k.lineCap = 'butt';
    k.fillStyle = C.acc; k.beginPath(); k.arc(ox, oy, 4.5, 0, 7); k.fill();
  });
};
