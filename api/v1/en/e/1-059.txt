// Gooey trail / リキッドの尾
// Pack: aiux (Glow & liquid) · element 11 · Category: Liquid interactions
// What happens: A white indicator moves between three options; the faster it goes, the more it stretches and the bigger the droplet clinging behind it.
// Motion spec:
//   - Segmented control 288 × 44 px with three options 96 px apart; the indicator is 84 × 34 px.
//   - It moves every 1.2 s, starting at 0.1, 1.3, 2.5 and 3.7 s, 0.45 s per move (cubic ease-in-out): 1 → 2 → 3 → 2 → 1.
//   - With speed it widens by up to 25% and flattens by up to 12% (full at 320 px/s).
//   - The tail sits where the indicator was 0.09 s earlier; its blob grows from 6 to 16 px radius, joined by a neck. Labels invert over the indicator.
// When to use it: Use it for the sliding indicator of a tab bar or segmented control, or for a slider thumb.
// Tags: gooey, liquid, tabs, segmented-control, indicator, trail, slider
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-059
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 11
// 11 液态拖尾（effect="move"）：分段控件的指示块移动时拉长，身后拖出液滴尾巴
CELLS[10] = (k, w, h, u) => {
  bg(k); grid(k);
  const tx = 125.5, ty = 82, TW = 288, TH = 44, step = 96, x0 = tx + 48;
  const MOVES = [[0.1, 0.55, 0, 1], [1.3, 1.75, 1, 2], [2.5, 2.95, 2, 1], [3.7, 4.15, 1, 0]];   // 每 1.2 秒移动一次，位移 96 px
  const pos = t => { t = ((t % CYC) + CYC) % CYC; let s = 0; for (const [a, b, f, g] of MOVES) { if (t >= a) s = mix(f, g, inOut3(seg(t, a, b))); } return s; };
  const s = pos(u), hx = x0 + step * s;
  const v = (pos(u + 0.01) - pos(u - 0.01)) / 0.02 * step;   // px/s
  const sp = clamp(Math.abs(v) / 320), dir = Math.sign(v) || 1;
  k.fillStyle = C.panel; rr(k, tx, ty, TW, TH, TH / 2); k.fill();
  k.strokeStyle = C.line; k.lineWidth = 1; rr(k, tx + 0.5, ty + 0.5, TW - 1, TH - 1, TH / 2); k.stroke();
  const PW = 84 * (1 + 0.25 * sp), PH = 34 * (1 - 0.12 * sp);   // 速度越快越扁长（stretch）
  const lag = x0 + step * pos(u - 0.09);                       // 尾巴滞后 0.09 秒（trail）
  const tailR = 6 + 10 * sp, tailX = lag - dir * (PW / 2 - 10);
  const endX = hx - dir * (PW / 2 - PH / 2);
  k.fillStyle = C.goo;
  rr(k, hx - PW / 2, ty + TH / 2 - PH / 2, PW, PH, PH / 2); k.fill();
  if (sp > 0.02) { neck(k, endX, ty + TH / 2, PH / 2, tailX, ty + TH / 2, tailR, PH / 2 + tailR + 40); disc(k, tailX, ty + TH / 2, tailR); }
  ['Chat', 'Search', 'Agents'].forEach((lab, i) => {
    const lx = x0 + step * i, on = clamp(1 - Math.abs(lx - hx) / 46);
    txt(k, lab, lx, ty + TH / 2 + 1, `600 14px ${FONT}`, on > 0.5 ? C.ink : C.muted, 'center');
  });
};
