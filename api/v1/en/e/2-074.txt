// Ink bleed / インクブリード
// Pack: motion2 (Motion graphics moves, part 2) · element 35 · Category: Text & shapes
// What happens: On rice-paper beige, four characters in turn tighten from soft ink blots into crisp shapes, hold, then spread and vanish.
// Motion spec:
//   - 4 characters (68 px) starting at 0.3 s, 0.25 s apart; one character's cycle is 3.6 s.
//   - 0–55%: blur from 16 px to 0 px and scale from 1.3x to 1x (smoothstep), appearing over the first 15%.
//   - 55–80% holds sharp; 80–100% spreads to 16 px blur and 1.2x while fading.
//   - The blurred ink is thresholded with contrast(16) so blots merge at the edges, then multiplied onto the paper.
// When to use it: Use it for titles with an East Asian ink, calligraphy, tea or sake feel.
// Tags: ink, bleed, blur, calligraphy, gooey, text
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-074
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 35
// 改写自 KittenJam Motion Library #36（Hannah）· 墨水晕开：宣纸色底，每个字（间隔 0.25 秒）先是模糊 16px、1.3 倍的墨团，收敛成清晰字形，停留后再晕开到 1.2 倍消失；离屏画模糊墨字后用 contrast(16) 阈值化做出墨团粘连的边缘，再以 multiply 叠到纸上。
const INKB_C = mkCanvas(CW, CH), INKB_G = INKB_C.getContext('2d');
const INKB_FIB = (() => { const r = rand(3636); return Array.from({ length: 40 }, () => [r() * CW, r() * CH, 6 + r() * 18, r() * Math.PI]); })();
CELLS[34] = (k, w, h, u) => {
  u = snapU(u);
  stage(k, '#F2EBDD');
  k.strokeStyle = 'rgba(150,120,80,.12)'; k.lineWidth = 1;
  INKB_FIB.forEach(([x, y, l, a]) => { k.beginPath(); k.moveTo(x, y); k.lineTo(x + Math.cos(a) * l, y + Math.sin(a) * l); k.stroke(); });
  const S = '墨染江南', F = `900 68px ${JP}`, g = INKB_G;
  g.save(); g.filter = 'none'; g.globalAlpha = 1; g.fillStyle = '#fff'; g.fillRect(0, 0, CW, CH);
  g.font = F; g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillStyle = '#000';
  chars(g, S, F, 14).forEach((c, i) => {
    const t = (u - 0.3 - 0.25 * i) / 3.6; if (t <= 0 || t >= 1) return;
    let b, op, sc;
    if (t < 0.55) { const q = smooth(t / 0.55); b = lerp(16, 0, q); sc = lerp(1.3, 1, q); op = clamp(t / 0.15); }
    else if (t < 0.8) { b = 0; sc = 1; op = 1; }
    else { const q = smooth((t - 0.8) / 0.2); b = lerp(0, 16, q); sc = lerp(1, 1.2, q); op = 1 - q; }
    g.filter = `blur(${b.toFixed(2)}px)`; g.globalAlpha = op;
    g.save(); g.translate(CW / 2 + c.x, CH / 2); g.scale(sc, sc); g.fillText(c.ch, 0, 0); g.restore();
  });
  g.restore();
  k.save(); k.filter = 'contrast(16)'; k.globalCompositeOperation = 'multiply'; k.drawImage(INKB_C, 0, 0); k.restore();
  // 角落朱印
  k.fillStyle = '#C8372D'; rr(k, w - 62, h - 54, 30, 30, 4); k.fill();
  txt(k, '印', w - 47, h - 38, `900 18px ${JP}`, '#F2EBDD', 'center', 'middle');
};
