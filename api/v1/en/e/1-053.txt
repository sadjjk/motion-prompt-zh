// Thinking orb: working / 思考オーブ・作業中
// Pack: aiux (Glow & liquid) · element 05 · Category: Loading & thinking
// What happens: Around a slowly spinning dotted orb, particles with short trails orbit on three tilted paths at different speeds.
// Motion spec:
//   - Orb radius 58 px, 240 dots on a Fibonacci sphere; nearer dots are larger and brighter. It spins once per cycle, viewed 0.32 rad from above.
//   - Three orbits tilted 0.35, 1.05 and −0.6 rad, at 1.14 × the orb radius, with 8 particles each.
//   - They go round 1, 2 and 1 (in reverse) times per cycle.
//   - Each particle trails 5 fading copies. Label "Working…" at 15 px on the right.
// When to use it: Use it for general agent work or a running tool. Skip it for waits under 2 s.
// Tags: thinking-orb, working, agent, loading, ai, orbit, particles
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-053
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 05
// 05 思考球·运转（thinking-orbs state="working"）：粒子在三条倾斜轨道上绕点阵球运行
CELLS[4] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC;
  for (const p of SPHERE) dot(k, rot(p, spin, ORB.tilt), 0.45);
  const orbits = [[0.35, 0, 1], [1.05, 2.1, 2], [-0.6, 4.2, -1]];   // [倾角, 偏航, 每周期圈数]
  for (const [tilt, yaw, sp] of orbits) {
    for (let j = 0; j < 8; j++) {
      for (let tr = 5; tr >= 0; tr--) {
        const th = sp * TAU * u / CYC + j * TAU / 8 - Math.sign(sp) * tr * 0.07;
        let p = [Math.cos(th) * 1.14, 0, Math.sin(th) * 1.14];
        p = rot([p[0], p[1] * Math.cos(tilt) - p[2] * Math.sin(tilt), p[1] * Math.sin(tilt) + p[2] * Math.cos(tilt)], yaw, ORB.tilt);
        dot(k, p, tr ? 0.5 * (1 - tr / 6) : 1, tr ? 0 : 0.35);
      }
    }
  }
  orbLabel(k, 'Working…');
};
