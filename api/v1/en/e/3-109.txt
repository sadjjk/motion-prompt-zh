// Clean look / すっきり
// Category: Editing & photography / Light & color / Mood & tone recipes
// What happens: It starts with a desk scattered with small items, things stuck all over the wall and 3 patches of differently coloured light. First the 7 items on the desk disappear, leaving only the laptop, cup and plant. Then the 4 things on the wall go and the wall and floor brighten and recede. Next the coloured light patches fade and an even soft light covers everything. Finally the colours settle into one pale grey tone. The 3 kept items never change position or size. A checklist on the right turns each numbered circle blue as its ingredient arrives, and the status reads plain version → adding → finished.
// Motion spec:
//   - 5 s cycle. 0–0.6 s is the plain version; the 4 ingredients join from 0.6 s, the next every 0.7 s, each fading in over 0.5 s.
//   - The 7 items on the desk disappear and the 3 kept ones remain: laptop, cup and plant; the 4 things on the wall fade and the wall lightens to #F5F4F1.
//   - The 3 coloured light patches fade into an even soft light of white at 0.18; tone: every colour is pulled 50% towards #EEF0F2; no material detail is added.
//   - The finished look holds until 4.4 s and slides back to the plain version over 4.4–4.95 s, so 5 s and 0 s are the same frame.
// When to use it: For desk, room and product shots that should look tidy and easy to read.
// Tags: clean, minimal, declutter, even light, consistent tone, simple
// Size: 539x208 px · Cycle: 5 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-109
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 62
// 62 干净：杂乱的书桌。删去杂物只留两三样 → 背景退后（墙上东西退场、墙变浅）→ 统一柔光去掉杂乱光斑 → 统一冷暖
// 与高级感的分别：只做减法和统一，不加材质，留下的东西位置和大小不变
const CL_MESS = [[70, 108, 26, 14, '#E85D75'], [104, 112, 18, 10, '#F2C94C'], [140, 100, 10, 22, '#7FB3E0'], [276, 106, 30, 16, '#9B7BE0'], [312, 114, 16, 8, '#6FCF97'], [336, 98, 12, 24, '#F2994A'], [196, 126, 22, 5, '#56CCF2']];
const CL_WALL = [[40, 24, 40, 30, '#F2C94C'], [96, 18, 28, 40, '#E85D75'], [290, 22, 50, 26, '#7FB3E0'], [350, 30, 22, 22, '#6FCF97']];
const CL_SPOT = [[90, 60, 28, '#FFD27A'], [320, 70, 34, '#9CD3FF'], [210, 40, 22, '#FFB0C8']];   // 杂乱光斑 [x, y, r, 色]
CELLS[61] = mdCell('干净', ['删杂物只留两三样', '背景退后', '统一柔光', '统一冷暖'], (k, u, [pM, pB, pL, pT], D) => {
  const unify = a => mixc(a, '#EEF0F2', pT * 0.5);
  D.rect(0, 0, MD_W, CH, unify(mixc('#E9DCCB', '#F5F4F1', pB)));
  k.save(); k.globalAlpha = 1 - pB; CL_WALL.forEach(([x, y, w, h, c]) => D.rect(x, y, w, h, c)); k.restore();
  D.rect(30, 124, 332, 8, unify('#C9AE8A')); D.rect(0, 132, MD_W, 76, unify(mixc('#D8CBB8', '#E9E6E0', pB)));
  // 留下的三样：笔记本电脑、杯子、一盆绿植
  D.rr(168, 92, 68, 32, 3, unify('#9AA0AA')); D.rect(160, 122, 84, 3, unify('#7E848E'));
  D.rr(254, 104, 16, 20, 3, unify('#FFFFFF')); D.rect(116, 108, 14, 16, unify('#C79A6B')); D.circ(123, 100, 11, unify('#7FAF7A'));
  k.save(); k.globalAlpha = 1 - pM; CL_MESS.forEach(([x, y, w, h, c]) => D.rr(x, y, w, h, 2, c)); k.restore();
  // 杂乱光斑：几团不同颜色的硬光，统一柔光后消失，换成一层均匀的柔光
  // 用带颜色的径向渐变画：浅色墙上用 lighter 会糊成白圆，看不出是「乱光」
  k.save(); k.globalAlpha = 0.55 * (1 - pL);
  CL_SPOT.forEach(([x, y, r, c]) => { const g = k.createRadialGradient(x, y, 2, x, y, r); g.addColorStop(0, c); g.addColorStop(1, c + '00'); k.fillStyle = g; k.beginPath(); k.arc(x, y, r, 0, TAU); k.fill(); });
  k.restore();
  if (pL > 0.01) { k.save(); k.globalCompositeOperation = 'screen'; k.fillStyle = `rgba(255,255,255,${(0.18 * pL).toFixed(3)})`; k.fillRect(0, 0, MD_W, CH); k.restore(); }
});
