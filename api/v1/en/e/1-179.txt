// Flowing gradient stroke / フローグラデーションライン
// Pack: aiux (Glow & liquid) · element 17 · Category: Loading & thinking
// What happens: An infinity-shaped line sits on a dark background; blue-to-pink colour keeps flowing along it while a glowing white spark with a tail laps the line.
// Motion spec:
//   - The line is a lemniscate (infinity sign) with a 200 px half-width, drawn in 180 segments.
//   - It has 2 layers: a 12 px glow at 0.12 opacity and a 4 px core at 0.95.
//   - Hue runs between 190° and 330° with 2 colour periods along the line, and moves by one period (half the line) per 4.8 s cycle.
//   - The spark laps the line once per 4.8 s; its tail is 10% of the line length, 24 segments, 5 px wide and fading by a power of 1.5; the head is a 3.5 px white dot with an 18 px radius glow.
// When to use it: Use it as a logo decoration, a tech-style background or a waiting state that should feel continuously flowing.
// Tags: gradient, stroke, flow, comet, infinity, line, loop
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-179
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 17
// 17 渐变光流线条（改写自 KittenJam #194，Hannah，MIT）：无限符号曲线（伯努利双纽线，半宽 200px）上的多色渐变
// 沿线流动，每周期前进半条线（一个色彩周期）；一颗彗星光点沿线一周 4.8 秒，拖着 10% 线长的尾巴。
const FGS_A = 200, FGS_CX = 269.5, FGS_CY = 104, FGS_N = 180, FGS_TAIL = 0.1;
const fgsPt = s => { const t = TAU * s, d = 1 + Math.sin(t) * Math.sin(t); return [FGS_CX + FGS_A * Math.cos(t) / d, FGS_CY + FGS_A * Math.sin(t) * Math.cos(t) / d]; };
CELLS[16] = (k, w, h, u) => {
  bg(k);
  const f = u / CYC;
  const hue = s => 190 + 140 * (0.5 - 0.5 * Math.cos(TAU * (2 * s - f)));   // 色相 190°–330°，线上 2 个色彩周期
  k.lineCap = 'round';
  for (const [lw, a] of [[12, 0.12], [4, 0.95]]) {
    for (let i = 0; i < FGS_N; i++) {
      const s0 = i / FGS_N, s1 = (i + 1) / FGS_N, p0 = fgsPt(s0), p1 = fgsPt(s1);
      k.strokeStyle = hsla(hue(s0), 95, 62, a); k.lineWidth = lw;
      k.beginPath(); k.moveTo(p0[0], p0[1]); k.lineTo(p1[0], p1[1]); k.stroke();
    }
  }
  // 彗星：头部 s = u/CYC，尾巴 24 段渐隐
  for (let i = 0; i < 24; i++) {
    const s0 = f - FGS_TAIL * (1 - i / 24), s1 = f - FGS_TAIL * (1 - (i + 1) / 24), p0 = fgsPt(s0), p1 = fgsPt(s1);
    k.strokeStyle = `rgba(255,255,255,${0.9 * Math.pow((i + 1) / 24, 1.5)})`; k.lineWidth = 5;
    k.beginPath(); k.moveTo(p0[0], p0[1]); k.lineTo(p1[0], p1[1]); k.stroke();
  }
  const hp = fgsPt(f);
  const gl = k.createRadialGradient(hp[0], hp[1], 0, hp[0], hp[1], 18);
  gl.addColorStop(0, 'rgba(255,255,255,0.7)'); gl.addColorStop(1, 'rgba(255,255,255,0)');
  k.fillStyle = gl; k.fillRect(hp[0] - 18, hp[1] - 18, 36, 36);
  k.fillStyle = '#FFFFFF'; k.beginPath(); k.arc(hp[0], hp[1], 3.5, 0, TAU); k.fill();
};
