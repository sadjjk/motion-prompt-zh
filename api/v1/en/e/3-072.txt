// Window shadow pattern / 窓枠の影
// Pack: photo (Photo looks, lighting & styles) · element 36 · Category: Lighting & time of day
// What happens: Light in the shape of 3×3 panes falls slantwise on the person and the light wall, with shadow everywhere else; as the clock runs from 15:00 to 15:40 the grid slides right and leans further.
// Motion spec:
//   - Scene: wall #F3EEE6, floor #E4DCCF from y 168; a head-and-shoulders figure #5A6884 (head centre x 270, y 78, radius 29 px).
//   - Panes: 3×3 panes of 52×46 px with 9 px frame bars; the panes stay lit and everything else is multiplied with #AEA9BB as shadow (hard edges).
//   - Over one pass the grid moves from x 96 to 176 and its slant (sideways shift per 1 px of y) grows from 0.42 to 0.72.
//   - The top-right clock runs from 15:00 to 15:40.
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, ease-in-out).
// When to use it: For afternoon rooms, cafés or noir looks that should suggest passing time or stillness.
// Tags: window light pattern, window shadow, hard light, afternoon, grid shadow, lighting
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-072
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 36
// 窗格光影: 明るい壁の前の胸像。窓枠の格子の影が斜めに壁と人物へ落ち、時間とともにゆっくり横へ動いて長く伸びる。
// 光の当たる所は格子のガラスの形、それ以外は影（縁はくっきり）
CELLS[35] = cell((k, u) => {
  const a = env(u), s = u / CYC, D = pen(k);
  D.rect(0, 0, CW, CH, '#F3EEE6'); D.rect(0, 168, CW, CH - 168, '#E4DCCF'); D.rect(0, 166, CW, 3, '#D6CCBC');
  bust(k, PX, PY, PR, '#5A6884');
  if (a > 0.002) {
    // ガラス1枚の四隅を、光の傾き sh で右下へ流して壁に写す
    const tx = lerp(96, 176, s), sh = lerp(0.42, 0.72, s), PW = 52, PH = 46, G = 9;
    const mp = (x, y) => [tx + x + sh * y, y];
    k.save(); k.globalCompositeOperation = 'multiply'; k.globalAlpha = a; k.fillStyle = '#AEA9BB';
    k.beginPath(); k.rect(0, 0, CW, CH);
    for (let j = 0; j < 3; j++) for (let i = 0; i < 3; i++) {
      const x0 = i * (PW + G), y0 = 14 + j * (PH + G) * 1.08;
      const q = [mp(x0, y0), mp(x0 + PW, y0), mp(x0 + PW, y0 + PH), mp(x0, y0 + PH)];
      k.moveTo(q[0][0], q[0][1]); for (const [x, y] of q.slice(1)) k.lineTo(x, y); k.closePath();
    }
    k.fill('evenodd');
    k.restore();
  }
  hud(k, clock(lerp(900, 940, s)));
});
