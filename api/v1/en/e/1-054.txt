// Thinking orb: searching / 思考オーブ・検索中
// Pack: aiux (Glow & liquid) · element 06 · Category: Loading & thinking
// What happens: A dotted globe turns while a vertical line of light sweeps over it, brightening the dots it passes.
// Motion spec:
//   - Orb radius 58 px, 240 dots; it spins once per cycle.
//   - The scanning meridian goes round 2 times per cycle (2 × the spin), drawn 1 px wide at 0.35 opacity.
//   - Dots near the meridian brighten by up to +0.75, falling off as exp(−Δ²/0.05) with the angle Δ.
//   - Label "Searching the web…" at 15 px on the right.
// When to use it: Use it while the AI is searching the web, retrieving documents or looking something up.
// Tags: thinking-orb, searching, search, retrieval, loading, globe
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-054
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 06
// 06 思考球·搜索（state="searching"）：点阵地球自转，一条亮经线扫过，扫到的点变亮
CELLS[5] = (k, w, h, u) => {
  bg(k);
  const spin = TAU * u / CYC, sweep = 2 * TAU * u / CYC;   // 自转 1 圈、扫描 2 圈 / 周期
  for (const p of SPHERE) {
    const q = rot(p, spin, 0), lon = Math.atan2(q[0], q[2]);
    const dd = wrapPi(lon - sweep), boost = 0.75 * Math.exp(-(dd * dd) / 0.05);
    dot(k, rot(p, spin, ORB.tilt), 0.7, boost);
  }
  k.strokeStyle = 'rgba(236,236,241,0.35)'; k.lineWidth = 1; k.beginPath();
  for (let i = 0; i <= 40; i++) {
    const lat = -Math.PI / 2 + Math.PI * i / 40;
    const q = rot([Math.cos(lat) * Math.sin(sweep), Math.sin(lat), Math.cos(lat) * Math.cos(sweep)], 0, ORB.tilt);
    const X = ORB.x + q[0] * ORB.r * 1.02, Y = ORB.y - q[1] * ORB.r * 1.02;
    if (i) k.lineTo(X, Y); else k.moveTo(X, Y);
  }
  k.stroke();
  orbLabel(k, 'Searching the web…');
};
