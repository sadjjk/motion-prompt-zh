// Saturate on hover / ホバーで彩度を戻す
// Pack: lp (Landing page & web effects) · element 46 · Category: Hover & click interactions
// What happens: Of 4 muted, dim pictures, the 1st and then the 3rd come back to vivid colour and zoom slightly as the cursor rests on them, with a name tag rising from the bottom.
// Motion spec:
//   - A row of 4 geometric pictures (104×118, 16 apart) that rest muted and dim: 90% desaturated and darkened by 20%.
//   - The hovered picture regains full colour and brightness over 0.6 s (easeOut power 3) and its image scales to 1.05 inside the frame.
//   - A name tag at the bottom rises 10px and fades in; the cursor rests on pictures 1 and 3 in turn, and each returns to muted after it leaves.
// When to use it: Use it for team pages, portfolio grids or client logos where a row of images should stay calm until one is pointed at.
// Tags: saturate, grayscale, image, hover, team, zoom
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-161
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 46
// 改写自 KittenJam Motion Library #264 低饱和唤醒：一排 4 张 104×118 的几何示意图（间距 16）平时低饱和偏暗（90% 去饱和、压暗 20%）。
// 光标滑过的那张 0.6 秒 easeOut 3 次方恢复鲜艳与亮度，画面在框内放大到 1.05，底部名字标签上浮 10px 淡入；光标依次停第 1、3 张，离开后恢复。
const SAT_W = 104, SAT_H = 118, SAT_GAP = 16;
CELLS[45] = (k, w, h, u) => {
  frame(k, 'example.com/team');
  const x0 = (VW - (SAT_W * 4 + SAT_GAP * 3)) / 2, y0 = 22, x3 = x0 + 2 * (SAT_W + SAT_GAP) + 56;
  const cur = t => path([[0, 255, 160], [0.5, x0 + 56, 80], [1.7, x0 + 56, 80], [2.1, x3, 80], [3.4, x3, 80], [3.9, 255, 160], [4.8, 255, 160]], t);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    for (let i = 0; i < 4; i++) {
      const x = x0 + i * (SAT_W + SAT_GAP);
      const on = t => { const [a, b] = cur(t); return a > x && a < x + SAT_W && b > y0 && b < y0 + SAT_H; };
      const q = trans(on, u, 0.6), d = 1 - q;
      pic(k, x, y0, SAT_W, SAT_H, i, 10, 1 + 0.05 * q);
      k.save(); rr(k, x, y0, SAT_W, SAT_H, 10); k.clip();
      if (d > 0) {
        // 用灰色做 saturation 合成来去饱和（相当于 grayscale），再叠一层黑色压暗（相当于 brightness）
        k.globalCompositeOperation = 'saturation'; k.globalAlpha = 0.9 * d; k.fillStyle = '#808080'; k.fillRect(x, y0, SAT_W, SAT_H);
        k.globalCompositeOperation = 'source-over'; k.globalAlpha = 1; k.fillStyle = `rgba(0,0,0,${0.2 * d})`; k.fillRect(x, y0, SAT_W, SAT_H);
      }
      if (q > 0) {
        k.globalAlpha = q; k.fillStyle = 'rgba(255,255,255,.92)'; rr(k, x + 8, y0 + SAT_H - 30 + 10 * d, SAT_W - 16, 22, 11); k.fill();
        txt(k, ['设计', '开发', '产品', '运营'][i], x + SAT_W / 2, y0 + SAT_H - 19 + 10 * d, `700 11px ${JP}`, C.ink, 'center', 'middle');
      }
      k.restore();
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
