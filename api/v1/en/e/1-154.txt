// Rolling text button / ローリングテキストボタン
// Pack: lp (Landing page & web effects) · element 39 · Category: Hover & click interactions
// What happens: When the cursor lands, the label's characters roll up one after another and the same characters come up from below; the button flips from white to ink and the round arrow turns to point up-right.
// Motion spec:
//   - A 220×56 pill button with the label on the left and a 40px round arrow on the right; the cursor moves onto it in 0.6 s.
//   - Each character rolls up one line (26px) over 0.55 s (easeInOut power 3), staggered 0.035 s per character, while an identical character rolls in from below.
//   - At the same time the button turns from white to ink, the text from ink to white, the circle from ink to accent orange, and the arrow turns from → to ↗ (−45°).
//   - The cursor leaves at 3.4 s and everything rolls back in reverse.
// When to use it: Use it for the main buttons of portfolio or brand sites where the CTA should feel playful.
// Tags: button, rolling-text, hover, stagger, cta, invert
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-154
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 39
// 改写自 KittenJam Motion Library #102 文字翻滚按钮：220×56 胶囊按钮，左侧文字「查看作品」、右侧直径 40 的圆形箭头。
// 光标 0.6 秒移入：每个字依次延迟 0.035 秒，各在 0.55 秒内（easeInOut 3 次方）往上翻走一个行高（26px），下面同一个字补上；
// 同时按钮底色白 → 墨色、文字墨色 → 白，圆圈墨色 → 朱色，箭头从 → 转到 ↗（−45°）。3.4 秒移开后全部反向翻回。
const ROLL_W = 220, ROLL_H = 56, ROLL_TXT = '查看作品';
CELLS[38] = (k, w, h, u) => {
  frame(k, 'example.com/works');
  const bx = (VW - ROLL_W) / 2, by = (VH - ROLL_H) / 2;
  const cur = t => path([[0, 430, 150], [0.6, 250, 90], [3.4, 250, 90], [4.0, 430, 150], [4.8, 430, 150]], t);
  const on = t => { const [a, b] = cur(t); return a > bx && a < bx + ROLL_W && b > by && b < by + ROLL_H; };
  const q = trans(on, u, 0.55, eIO);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.fillStyle = mixHex('#FFFFFF', C.ink, q); rr(k, bx, by, ROLL_W, ROLL_H, ROLL_H / 2); k.fill();
    k.strokeStyle = C.ink; k.lineWidth = 1.5; k.stroke();
    const font = `800 17px ${JP}`, ty = by + ROLL_H / 2, LH = 26, tc = mixHex(C.ink, '#FFFFFF', q);
    let x = bx + 30;
    k.save(); k.beginPath(); k.rect(bx + 20, ty - LH / 2, 130, LH); k.clip();
    [...ROLL_TXT].forEach((ch, i) => {
      // 每个字的时间线整体往后错开 i×0.035 秒
      const p = trans(on, u - i * 0.035, 0.55, eIO);
      txt(k, ch, x, ty + 1 - LH * p, font, tc, 'left', 'middle');
      txt(k, ch, x, ty + 1 + LH * (1 - p), font, tc, 'left', 'middle');
      k.font = font; x += k.measureText(ch).width;
    });
    k.restore();
    const ax = bx + ROLL_W - 28;
    k.fillStyle = mixHex(C.ink, C.acc, q); k.beginPath(); k.arc(ax, ty, 20, 0, 7); k.fill();
    k.save(); k.translate(ax, ty); k.rotate(-Math.PI / 4 * q); k.strokeStyle = '#fff'; k.lineWidth = 2.2; k.lineCap = 'round'; k.lineJoin = 'round';
    k.beginPath(); k.moveTo(-7, 0); k.lineTo(7, 0); k.moveTo(2, -5); k.lineTo(7, 0); k.lineTo(2, 5); k.stroke(); k.restore();
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
