// Karaoke bouncing ball / カラオケボール
// Pack: motion2 (Motion graphics moves, part 2) · element 34 · Category: Text & shapes
// What happens: A glowing yellow ball hops left to right over seven dim characters; each one it lands on turns white, glows and pops. At the end the line dims and starts over.
// Motion spec:
//   - 7 characters (44 px) start dim; from 0.3 s the ball hops to the next character every 0.38 s (a 34 px high sine arc).
//   - A character it lands on turns from dim to white within 0.14 s, gains a yellow glow (16/8 px strokes) and pops with 1 + 0.12·e^(−9t)·cos(16t).
//   - After the last character the ball hops off to the right and fades; over 4.0–4.3 s all characters dim again.
//   - Over 4.3–4.65 s the ball fades in left of the first character, matching the frame at 0 s.
// When to use it: Use it for lyric subtitles, read-along captions synced to rhythm, or kids' reading videos.
// Tags: karaoke, lyrics, bouncing-ball, highlight, subtitle, text
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-073
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 34
// 改写自 KittenJam Motion Library #25（Hannah）· 卡拉 OK 小球：发光小球每 0.38 秒跳到下一个字（跳高 34px），落点字从暗色亮成白色并带黄色光晕和 1.12 倍弹跳；整句跳完球从右侧离场，4.0–4.3 秒字回暗，4.3 秒起球在左侧淡入复位。
CELLS[33] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '一闪一闪亮晶晶', F = `800 44px ${JP}`, CY = 124, T0 = 0.3, PER = 0.38, n = [...S].length;
  const cs = chars(k, S, F, 10), X = i => w / 2 + (i < 0 ? cs[0].x - 60 : i >= n ? cs[n - 1].x + 60 : cs[i].x);
  const YB = CY - 22 - 16;
  // 小球
  let bx, by = YB, ba = 1;
  const tEnd = T0 + n * PER;
  if (u < T0) bx = X(-1);
  else if (u < tEnd + PER) { const i = Math.floor((u - T0) / PER), fr = (u - T0) / PER - i; bx = lerp(X(i - 1), X(i), fr); by = YB - 34 * Math.sin(Math.PI * fr); if (i >= n) ba = 1 - fr; }
  else { bx = X(-1); ba = P(u, 4.3, 4.65); }
  cs.forEach((c, i) => {
    const land = T0 + (i + 1) * PER;
    const L = P(u, land - 0.04, land + 0.1) * (1 - P(u, 4.0, 4.3));
    const pop = u > land && u < 4.0 ? 1 + 0.12 * Math.exp(-9 * (u - land)) * Math.cos(16 * (u - land)) : 1;
    k.save(); k.translate(w / 2 + c.x, CY); k.scale(pop, pop);
    k.font = F; k.textAlign = 'center'; k.textBaseline = 'middle'; k.lineJoin = 'round';
    if (L > 0) [[16, 0.12], [8, 0.22]].forEach(([lw, a]) => { k.globalAlpha = a * L; k.strokeStyle = C.yel; k.lineWidth = lw; k.strokeText(c.ch, 0, 0); });
    k.globalAlpha = 1; k.fillStyle = mix('#4A5680', '#FFFFFF', L); k.fillText(c.ch, 0, 0);
    k.restore();
  });
  if (ba > 0) {
    k.save(); k.globalAlpha = ba;
    [[16, 0.12], [11, 0.22]].forEach(([r, a]) => { k.fillStyle = `rgba(255,200,61,${a})`; k.beginPath(); k.arc(bx, by, r, 0, 7); k.fill(); });
    k.fillStyle = C.yel; k.beginPath(); k.arc(bx, by, 7, 0, 7); k.fill();
    k.fillStyle = '#fff'; k.beginPath(); k.arc(bx - 2, by - 2, 2.2, 0, 7); k.fill();
    k.restore();
  }
};
