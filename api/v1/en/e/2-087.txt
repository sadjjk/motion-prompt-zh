// Text fan / 文字の扇
// Pack: motion2 (Motion graphics moves, part 2) · element 48 · Category: Text & shapes
// What happens: Seven stacked copies of 扇 swing open around a yellow pivot below into a fan, with translucent coral copies either side of the white centre one, then fold shut.
// Motion spec:
//   - 7 copies of one character pivot on a point 112 px below the character centre.
//   - Over 0.5–1.9 s they open to (i−3)×24° with a cubic ease-in-out and hold until 3.6 s.
//   - Over 3.6–4.5 s they fold back.
//   - The centre copy is opaque white; the others are coral, fainter further out (0.7 − 0.12 × distance).
// When to use it: Use it for Japanese-style or traditional logos and brand reveals.
// Tags: fan, radial, rotate, copies, japanese, logo
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-087
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 48
// 改写自 KittenJam Motion Library #24（Hannah）· 文字扇：同一个「扇」字 7 份，以字下方 112px 的点为轴，0.5–1.9 秒按 (i−3)×24° 展开成扇形，停留到 3.6 秒，3.6–4.5 秒收拢；中间一份白色不透明，两侧珊瑚色、越外越淡。
const TFAN_ORD = [0, 6, 1, 5, 2, 4, 3];   // 外侧先画，中间一份压在最上面
CELLS[47] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const PX = w / 2, PY = 196, R = 112, F = `900 80px ${JP}`;
  const o = eIO(P(u, 0.5, 1.9)) * (1 - eIO(P(u, 3.6, 4.5)));
  TFAN_ORD.forEach(i => {
    const a = (i - 3) * 24 * Math.PI / 180 * o;
    k.save(); k.translate(PX, PY); k.rotate(a);
    k.strokeStyle = 'rgba(255,241,214,.16)'; k.lineWidth = 2; k.beginPath(); k.moveTo(0, -8); k.lineTo(0, -R + 42); k.stroke();
    k.globalAlpha = i === 3 ? 1 : 0.7 - Math.abs(i - 3) * 0.12;
    txt(k, '扇', 0, -R, F, i === 3 ? '#fff' : C.coral, 'center', 'middle');
    k.restore();
  });
  k.fillStyle = C.yel; k.beginPath(); k.arc(PX, PY, 6, 0, 7); k.fill();
};
