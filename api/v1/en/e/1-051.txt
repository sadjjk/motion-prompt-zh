// Inner pulse / インナーパルス
// Pack: aiux (Glow & liquid) · element 03 · Category: Loading & thinking
// What happens: Inside a pill button, pink, orange and blue glows swell and shrink, and the outline brightens with them.
// Motion spec:
//   - Button 200 × 56 px, corner radius 28 px. The light is clipped to the button and never spills out.
//   - One breath is 2.4 s (2 per cycle), eased as 0.5 − 0.5·cos from 0 to 1 and back to 0.
//   - Three radial glows (hues 330°, 40°, 200°) grow from 46 to 80 px radius and 0.35 to 0.9 opacity, drifting ±14 px sideways.
//   - A 1.4 px rainbow outline goes from 0.3 to 0.8 opacity. Label "✦ Ask AI" at 16 px.
// When to use it: Use it to highlight an AI action button or main call to action, such as Ask AI or Generate.
// Tags: pulse, button, cta, highlight, glow, ai
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-051
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 03
// 03 内呼吸光晕（border-beam size="pulse-inner"）：彩色光在按钮内部明暗呼吸，不溢出按钮
CELLS[2] = (k, w, h, u) => {
  bg(k);
  const x = 169.5, y = 76, W = 200, H = 56, R = 28;
  const s = 0.5 - 0.5 * Math.cos(TAU * u / 2.4);          // 2.4 秒一呼一吸，周期内 2 次
  k.fillStyle = C.panel; rr(k, x, y, W, H, R); k.fill();
  k.save(); rr(k, x, y, W, H, R); k.clip();
  const drift = Math.sin(TAU * u / CYC) * 14;
  [[x + 40 + drift, 330], [x + W / 2 - drift, 40], [x + W - 40 + drift, 200]].forEach(([bx, hue]) => {
    const rad = 46 + 34 * s;
    const g = k.createRadialGradient(bx, y + H / 2, 0, bx, y + H / 2, rad);
    g.addColorStop(0, hsla(hue, 100, 62, 0.35 + 0.55 * s)); g.addColorStop(1, hsla(hue, 100, 62, 0));
    k.fillStyle = g; k.fillRect(x, y, W, H);
  });
  k.restore();
  k.strokeStyle = conic(k, x + W / 2, y + H / 2, TAU * u / CYC, 0.3 + 0.5 * s); k.lineWidth = 1.4;
  rr(k, x + 0.7, y + 0.7, W - 1.4, H - 1.4, R); k.stroke();
  txt(k, '✦  Ask AI', x + W / 2, y + H / 2 + 1, `600 16px ${FONT}`, C.text, 'center');
};
