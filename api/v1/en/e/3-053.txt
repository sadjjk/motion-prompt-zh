// Medium close-up / バストショット
// Pack: camera (Camera moves & transitions) · element 29 · Category: Shot size
// What happens: The person, chest up and slightly left of centre, says two lines: the mouth moves and a subtitle appears with each line, there is a blink in the pause between them, and a "！" pops up above the head at the end.
// Motion spec:
//   - Frame: 1.2 m to 1.9 m, bottom edge at the chest; camera 2.2 m away, the person 0.35 m left of centre leaving the right side for the subtitle.
//   - Speech in two parts, 0.4 s to 2.0 s and 2.9 s to 4.3 s; the mouth opens and closes with each, with one wide opening late in each part.
//   - A 239×26 px subtitle bar fades in and out with the speech, reading "今天的风，比昨天轻一点。"
//   - A blink in the pause at 2.55 s; a white "！" on a blue dot pops up above the head from 4.25 s to 4.5 s. The body sways slightly while speaking.
//   - Labels: top right reads "中近景 MCU"; the scale band runs from the chest to the head; background softly blurred; one pass = 4.7 s.
// When to use it: For interviews, dialogue and monologues where speaking is the point.
// Tags: medium close-up, mcu, bust shot, chest up, talking head, shot size, dialogue
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/camera.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-053
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 29
// 29 中近景：胸部以上，下沿 1.2 m 落在胸口。画的不是大动作，是「正在说话」：
// 身体轻晃、嘴跟着说话包络开合（说两段、停一拍眨一次眼），字幕跟着说话出现，最后头顶冒「！」
CELLS[28] = shot(frameCell({ lo: 1.2, hi: 1.9, name: '中近景', abbr: 'MCU', draw: (k, w, h, u) => {
  const X = -0.35;   // 人物略偏左，右侧留给字幕和气泡
  // 说话包络：两段（0.4–2.0、2.9–4.4），段间停一拍；首尾安静，正好接 RESET 的淡出
  const sp = P(u, 0.4, 0.7) * (1 - P(u, 1.7, 2.0)) + P(u, 2.9, 3.2) * (1 - P(u, 4.0, 4.3));
  // 口型：小开合为主 + 每段中后段一次大张口收尾；停顿时合拢
  const talk = sp > 0.01 ? sp * (0.30 + 0.32 * Math.max(0, wave(u, 9))) + sp * 0.36 * P(wave(u, 3, -0.9), 0, 0.9) : 0;
  const blink = clamp(1 - Math.abs(u - 2.55) / 0.09);   // 停顿中的眨眼
  const sway = wave(u, 2) * 0.014 + sp * wave(u, 1) * 0.02;   // 身体轻晃，说话时加大
  world(k, w, h, fit(1.2, 1.9, 2.2), {
    person: { x: X + sway, z: 8, fg: true }, blurBg: 1.6,   // 人物和脸在前景层，背景只轻微虚化
    extra: [[X + sway, 0, 7.99, (kk, s) => face(kk, s, { talk, blink }), true]],
    after: (kk, cc) => {
      if (sp > 0.01) {   // 字幕：跟着说话出现，显示的就是这一句的内容
        kk.save(); kk.globalAlpha = sp;
        kk.fillStyle = 'rgba(255,255,255,.92)'; rr(kk, 150, 158, 239, 26, 13); kk.fill();
        kk.fillStyle = C.ink; kk.font = `600 13px ${JP}`; kk.textAlign = 'center'; kk.textBaseline = 'middle';
        kk.fillText('今天的风，比昨天轻一点。', 270, 172);
        kk.restore();
      }
      // 最后一拍：头顶冒「！」，跟着 RESET 一起淡出
      const q = P(u, 4.25, 4.5);
      if (q > 0.01) {
        const p = proj(cc, X, 1.78, 8);
        kk.save(); kk.globalAlpha = q;
        kk.fillStyle = C.blue; kk.beginPath(); kk.arc(p.x + 30, p.y - 16, 11, 0, TAU); kk.fill();
        kk.fillStyle = '#FFFFFF'; kk.font = `700 14px ${JP}`; kk.textAlign = 'center'; kk.textBaseline = 'middle';
        kk.fillText('！', p.x + 30, p.y - 15); kk.restore();
      }
    }
  });
} }));
