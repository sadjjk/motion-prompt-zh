// Loneliness / 孤独感
// Category: Editing & photography / Light & color / Mood & tone recipes
// What happens: It starts close on a person standing near a bench on a platform. First the view pulls back and the person becomes small inside a long platform of pillars and roof. Then the surroundings fall dark and only the cone of light from 1 lamp lights him and the floor at his feet. Next the colours turn to a blue-grey night. Finally the camera keeps backing away very slowly and the figure gets smaller still. A checklist on the right turns each numbered circle blue as its ingredient arrives, and the status reads plain version → adding → finished.
// Motion spec:
//   - 5 s cycle. 0–0.6 s is the plain version; the 4 ingredients join from 0.6 s, the next every 0.7 s, each fading in over 0.5 s.
//   - The view widens from 2.6× to 1×; everything else darkens by 0.45, leaving only the cone of light from 1 lamp and the pool at his feet.
//   - A blue-grey night palette replaces the daytime colours; very slow pull-back: a further 12% shrink from 2.7 to 4.4 s.
//   - The finished look holds until 4.4 s and slides back to the plain version over 4.4–4.95 s, so 5 s and 0 s are the same frame.
// When to use it: For film-like stills, story-driven portraits and lonely moments in music videos.
// Tags: loneliness, wide shot, small figure, single light, night, pull back
// Size: 539x208 px · Cycle: 5 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-105
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 58
// 58 孤独感：夜里的空站台。拉成大远景、人变小 → 只有一盏灯照到他 → 蓝灰夜色 → 镜头极慢拉远（最后一项一直慢慢退到 4.4 秒）
const LO_P = [214, 140], LO_LAMP = [214, 46];
CELLS[57] = mdCell('孤独感', ['拉成大远景', '一盏灯只照到他', '蓝灰夜色', '镜头极慢拉远'], (k, u, [pW, pL, pN, pS], D) => {
  const c = (a, b) => mixc(a, b, pN), s = lerp(2.6, 1, pW) * (1 - 0.12 * pS * mdSlow(u, 3));
  D.rect(0, 0, MD_W, CH, c('#E3E5E8', '#202838'));
  k.save(); k.translate(196, 118); k.scale(s, s); k.translate(-LO_P[0], -LO_P[1] + 22);
  D.rect(-200, 12, 800, 14, c('#B9BDC4', '#2C3548'));   // 顶棚
  for (let x = -180; x < 600; x += 92) D.rect(x, 26, 6, 114, c('#C6C9CF', '#323B50'));   // 柱子
  D.rect(-200, 140, 800, 12, c('#CFD2D7', '#3A4459')); D.rect(-200, 140, 800, 3, c('#F2D27A', '#8A7A45'));   // 站台边 + 黄线
  D.rect(-200, 152, 800, 80, c('#9EA3AB', '#171D2A'));   // 轨道
  for (let x = -200; x < 600; x += 18) D.rect(x, 168, 10, 4, c('#8A8F98', '#1F2636'));
  D.rr(110, 126, 40, 5, 2, c('#A9ADB4', '#2B3346')); D.rect(114, 131, 3, 9, c('#A9ADB4', '#2B3346')); D.rect(143, 131, 3, 9, c('#A9ADB4', '#2B3346'));   // 长椅
  D.rect(LO_LAMP[0] - 1, 26, 2, LO_LAMP[1] - 26, c('#9FA3AA', '#3A4152')); D.rr(LO_LAMP[0] - 8, LO_LAMP[1] - 2, 16, 5, 2, c('#9FA3AA', '#3A4152'));
  figure(D, LO_P[0], LO_P[1], 30, c('#4D5A75', '#0F141F'));
  if (pL > 0.01) {   // 其余地方压暗，只有灯下一锥光
    D.rect(-400, -200, 1200, 600, `rgba(8,10,18,${(0.45 * pL).toFixed(3)})`);
    k.save(); k.globalCompositeOperation = 'lighter';
    const g = k.createLinearGradient(0, LO_LAMP[1], 0, LO_P[1] + 6);
    g.addColorStop(0, `rgba(255,214,150,${(0.5 * pL).toFixed(3)})`); g.addColorStop(1, `rgba(255,214,150,${(0.12 * pL).toFixed(3)})`);
    k.fillStyle = g; k.beginPath(); k.moveTo(LO_LAMP[0] - 6, LO_LAMP[1] + 3); k.lineTo(LO_LAMP[0] + 6, LO_LAMP[1] + 3); k.lineTo(LO_LAMP[0] + 34, LO_P[1] + 6); k.lineTo(LO_LAMP[0] - 34, LO_P[1] + 6); k.closePath(); k.fill();
    k.fillStyle = `rgba(255,230,180,${(0.9 * pL).toFixed(3)})`; k.beginPath(); k.ellipse(LO_P[0], LO_P[1] + 4, 30, 4, 0, 0, TAU); k.fill();
    k.restore();
  }
  k.restore();
});
