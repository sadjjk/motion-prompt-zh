// Category: Visual styles / Craft & patterns / Print & handmade
// What happens: A moonlit night on black. The knife scrapes out the moon's outline, horizontal lines across it that skip the craters, then waves and stars, with the blade following the line it is cutting.
// Motion spec:
//   - The card is a black coating #121212 that stays put through the loop; every scraped line is off-white #F2EEE6.
//   - 0.3–1.2 s: the outline of a moon 56 px in radius at (270, 84) is scraped; 1.2–3.6 s: 37 horizontal lines 3 px apart fill it from top to bottom, each from left to right, leaving gaps at 3 craters.
//   - 3.6–4.6 s: 5 wavy lines are scraped across the bottom of the card, each starting 0.1 s after the last; 4.6–5.0 s: 6 small cross-shaped stars appear one by one.
//   - A small triangular blade sits on the end of the line being scraped.
//   - At 5.4–6.0 s the scraped lines fade back into the black; at 0 s and 6.4 s the card is plain black, so the loop restarts cleanly.
// When to use it: Use it for night scenes, mystery, gothic moods, or any strong black-and-white print look.
// Tags: scratchboard, engraving, black, white-lines, night, process
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/drawstyles.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-083
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 12
// 刮画：全黑涂层上用刮刀一道道刮出白线，先刮月亮轮廓，再横向刮满月面（避开月坑），最后刮出海浪和星星
{
  const WHITE = '#F2EEE6', MX = 270, MY = 84, MR = 56;
  const RIM = ellPts(MX, MY, MR, MR, 31, 0.8, 72, -Math.PI / 2, 0.05);
  const CRATERS = [[248, 70, 10], [290, 96, 14], [262, 112, 7]];
  // 月面横线：37 条、间距 3 px，每条拆成避开月坑的若干段；记下每条的总长，按长度推进
  const LINES = [];
  for (let i = 0; i < 37; i++) {
    const y = MY - 54 + i * 3, half = Math.sqrt(Math.max(0, MR * MR - (y - MY) * (y - MY))) - 2;
    let parts = [[MX - half, MX + half]];
    CRATERS.forEach(([cx, cy, cr]) => {
      const dy = y - cy; if (Math.abs(dy) >= cr) return;
      const hw = Math.sqrt(cr * cr - dy * dy) + 1.5, next = [];
      parts.forEach(([a, b]) => { if (cx - hw > a) next.push([a, Math.min(b, cx - hw)]); if (cx + hw < b) next.push([Math.max(a, cx + hw), b]); });
      parts = next;
    });
    LINES.push({ y, parts, len: parts.reduce((s, [a, b]) => s + (b - a), 0) });
  }
  const WAVES = [158, 167, 176, 185, 194].map((y0, j) => { const pts = []; for (let x = 40; x <= 500; x += 4) pts.push([x, y0 + 2.5 * Math.sin(x / 14 + j)]); return pts; });
  const STARS = [];
  for (let s = 0; STARS.length < 6; s++) {
    const x = 40 + rnd(s, 41) * 460, y = 10 + rnd(s, 42) * 48;
    if (Math.hypot(x - MX, y - MY) > MR + 20 && STARS.every(([sx, sy]) => Math.hypot(sx - x, sy - y) > 50)) STARS.push([x, y]);
  }
  function knife(k, x, y) { k.beginPath(); k.moveTo(x, y); k.lineTo(x + 5, y - 11); k.lineTo(x + 10, y - 7); k.closePath(); k.fill(); }
  CELLS[11] = (k, w, h, u) => {
    // 黑色涂层就是“纸”，不随复位淡出；刮出的白线随 vis 淡去
    paper(k, '#121212', 'rgba(255,255,255,.025)', 300, 5);
    const a = vis(u);
    k.globalAlpha = a; k.strokeStyle = WHITE; k.fillStyle = WHITE; k.lineCap = 'round';
    let tip = null;
    // 0.3–1.2 秒：月亮轮廓
    const tr = seg(u, 0.3, 1.2);
    k.lineWidth = 1.4; partial(k, RIM, tr);
    if (tr > 0 && tr < 1) tip = RIM[Math.floor(tr * (RIM.length - 1))];
    // 1.2–3.6 秒：月面横线，从上到下，每条从左往右
    const p = seg(u, 1.2, 3.6) * LINES.length;
    k.lineWidth = 1.1;
    for (let i = 0; i < Math.ceil(p); i++) {
      const L = LINES[i]; let left = Math.min(1, p - i) * L.len;
      for (const [x0, x1] of L.parts) {
        if (left <= 0) break;
        const x = Math.min(x1, x0 + left); left -= x1 - x0;
        k.beginPath(); k.moveTo(x0, L.y); k.lineTo(x, L.y); k.stroke();
        if (p - i < 1) tip = [x, L.y];
      }
    }
    // 3.6–4.6 秒：海浪，从左往右，每条错开 0.1 秒
    k.lineWidth = 1.2;
    WAVES.forEach((pts, j) => {
      const t = seg(u, 3.6 + 0.1 * j, 4.2 + 0.1 * j);
      partial(k, pts, t);
      if (t > 0 && t < 1) tip = pts[Math.floor(t * (pts.length - 1))];
    });
    // 4.6–5.0 秒：星星一颗颗刮出
    k.lineWidth = 1;
    STARS.forEach(([x, y], i) => {
      const t = seg(u, 4.6 + 0.065 * i, 4.7 + 0.065 * i);
      if (t <= 0) return;
      const r = 4 * t;
      k.beginPath(); k.moveTo(x - r, y); k.lineTo(x + r, y); k.moveTo(x, y - r); k.lineTo(x, y + r); k.stroke();
    });
    // 刮刀刀尖，跟着当前刮到的位置
    if (tip) { k.globalAlpha = a * 0.7; knife(k, tip[0], tip[1]); }
    k.globalAlpha = 1;
  };
}
