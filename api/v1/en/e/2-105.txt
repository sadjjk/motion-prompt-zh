// Push transition / プッシュ
// Category: Motion graphics / Effects & transitions / Transitions
// What happens: The next scene slides in from the side and pushes the current one out with it.
// Motion spec:
//   - Sea scene → mountain scene → sea scene, 2.35 s each. The change runs at 0.45–1.6 s of each half (1.15 s), cubic ease-in-out.
//   - The two scenes stay joined and move together by the full frame width (539 px); no overlap and no fade.
//   - Direction: sea→mountain moves left (the next scene enters from the right); mountain→sea moves right.
//   - A navy line (width 6) marks the seam between the two scenes.
// When to use it: Use it to step through slides, photos or steps, when the change should feel continuous and moving forward.
// Tags: push, push-transition, slide, transition, scene-change, carousel
// Size: 539x208 px · Cycle: 4.7 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-105
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 17
// 17 プッシュ（次の画面が横から入ってきて、前の画面を一緒に押し出す）
const SC17 = [scSea, scMtn];
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1)), dir = a === 0 ? -1 : 1;
  if (p <= 0) return SC17[a](k, w, h);
  if (p >= 1) return SC17[b](k, w, h);
  k.save(); k.translate(dir * p * w, 0); SC17[a](k, w, h); k.restore();
  k.save(); k.translate(dir * (p - 1) * w, 0); SC17[b](k, w, h); k.restore();
  const ex = dir < 0 ? w * (1 - p) : w * p;
  k.fillStyle = C.navy; k.fillRect(ex - 3, 0, 6, h);
});
