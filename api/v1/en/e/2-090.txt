// What happens: 立体 with a white face and darker coral layers stacked behind for depth turns slowly once around the vertical axis, showing its thickness side-on.
// Motion spec:
//   - A white face with 14 coral layers behind it (2.6 px deep each, darker further back).
//   - It turns once around the vertical axis every 4.7 s at a constant rate (scaleX = cos).
//   - A 12° tilt shifts the layers slightly upward.
//   - When it faces away the draw order flips so the deepest layer is in front.
// When to use it: Use it for logos, titles or game-style headings.
// Tags: extrude, 3d, depth, rotate, title, logo
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-090
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 51
// 改写自 KittenJam Motion Library #61（Hannah）· 立体字：「立体」由 14 层珊瑚色阶（每层深 2.6px，越深越暗）叠出厚度，正面白色；整组绕竖轴每 4.7 秒匀速转一圈（scaleX = cos），带 12° 俯角的上下错位，转到侧面能看到厚度。
CELLS[50] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '立体', F = `900 96px ${JP}`, th = Math.PI * 2 * u / CYC, c = Math.cos(th), s = Math.sin(th), TX = Math.tan(12 * Math.PI / 180);
  const sx = Math.abs(c) < 0.02 ? (c < 0 ? -0.02 : 0.02) : c;
  k.fillStyle = 'rgba(0,0,0,.3)'; k.beginPath(); k.ellipse(w / 2, 172, 130 * Math.max(0.3, Math.abs(c)), 7, 0, 0, 7); k.fill();
  const ids = []; for (let i = 14; i >= 0; i--) ids.push(i);   // 0 = 正面
  if (c < 0) ids.reverse();                                     // 转到背面时近处是最深一层
  ids.forEach(i => {
    const d = i * 2.6;
    k.save(); k.translate(w / 2 - d * s, h / 2 - 4 - d * TX); k.scale(sx, 1);
    txt(k, S, 0, 0, F, i ? `hsl(6,72%,${54 - i * 2}%)` : '#fff', 'center', 'middle');
    k.restore();
  });
};
