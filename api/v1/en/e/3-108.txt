// Dreamy look / ドリーミー
// Category: Editing & photography / Light & color / Mood & tone recipes
// What happens: It starts from a plain daytime picture of a person, head and shoulders, among flowers. First the sun appears behind them and a bright line traces the edge of the head and shoulder. Then the brightness around the sun blooms outward and small points of light on the flowers twinkle. Next the 4 near flowers grow and blur into soft blobs of colour. Finally the whole palette moves to pink, lavender and cream. A checklist on the right turns each numbered circle blue as its ingredient arrives, and the status reads plain version → adding → finished.
// Motion spec:
//   - 5 s cycle. 0–0.6 s is the plain version; the 4 ingredients join from 0.6 s, the next every 0.7 s, each fading in over 0.5 s.
//   - The sun sits behind the person at (236, 60), with a 2.4 px rim light on the head and shoulder; the bloom reaches a 170 px radius and 9 points of light twinkle 2 times per cycle.
//   - The 4 near flowers grow from 0.5× to 1× and blur by 7 px; the 46 flowers, sky and ground shift to pink, lavender and cream.
//   - The finished look holds until 4.4 s and slides back to the plain version over 4.4–4.95 s, so 5 s and 0 s are the same frame.
// When to use it: For flower-field and wedding portraits, fantasy-style photos and soft product shots.
// Tags: dreamy, backlight, rim light, bloom, foreground blur, pastel
// Size: 539x208 px · Cycle: 5 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-108
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 61
// 61 梦幻感：花丛中的人。逆光轮廓（人背后的太阳，边缘一圈亮）→ 高光晕开 → 前景花虚化 → 粉紫奶白配色
const DR_P = { x: 196, y: 92, R: 24 }, DR_SUN = [236, 60];
const DR_FLOW = () => once('drFlow', () => { const r = rng(61), o = []; for (let i = 0; i < 46; i++) o.push([r() * MD_W, 128 + r() * 70, 4 + r() * 5, i % 3]); return o; });
const DR_FG = [[34, 186, 30], [96, 200, 22], [318, 190, 28], [372, 176, 20]];   // 前景大花 [x, y, r]
const DR_COL = [['#E85D75', '#F6B6D0'], ['#F2C94C', '#FFF1D6'], ['#7FB3E0', '#C9B6F0']];   // [原色, 粉紫奶白版]
CELLS[60] = mdCell('梦幻感', ['逆光轮廓', '高光晕开', '前景花虚化', '粉紫奶白配色'], (k, u, [pR, pG, pF, pC], D) => {
  const T = u / CYC * TAU, c = (a, b) => mixc(a, b, pC);
  D.rect(0, 0, MD_W, CH, c(mixc('#BFD9EE', '#F4D9B0', pR), mixc('#F3DDEB', '#FFF4F0', pR)));
  D.rect(0, 120, MD_W, 88, c('#8DBF6E', '#D9C8E4'));
  if (pR > 0.01) sun(D, DR_SUN[0], DR_SUN[1], 14, { sun: c('#FFE7A8', '#FFF6EE'), glow: c('#FFE7A8', '#FFE4F0') }, pR);
  DR_FLOW().forEach(([x, y, r, j]) => D.circ(x, y, r, c(DR_COL[j][0], DR_COL[j][1])));
  bust(k, DR_P.x, DR_P.y, DR_P.R, c(mixc('#4D5A75', '#3A3550', pR), '#8C7A9E'));
  if (pR > 0.01) {   // 轮廓光：沿头和肩外缘一圈亮边，偏向太阳一侧
    k.save(); k.globalAlpha = pR; k.strokeStyle = c('#FFE9B0', '#FFF6F8'); k.lineWidth = 2.4;
    k.beginPath(); k.arc(DR_P.x, DR_P.y, DR_P.R, -2.4, 0.6); k.stroke();
    k.beginPath(); k.moveTo(DR_P.x + 1.44 * DR_P.R, DR_P.y + 1.4 * DR_P.R); k.lineTo(DR_P.x + 1.94 * DR_P.R, DR_P.y + 1.8 * DR_P.R); k.stroke(); k.restore();
  }
  if (pG > 0.01) {   // 高光晕开：亮部往外扩一层柔光，花上的小光点轻轻闪（一轮 2 次）
    k.save(); k.globalCompositeOperation = 'screen';
    const g = k.createRadialGradient(DR_SUN[0], DR_SUN[1], 6, DR_SUN[0], DR_SUN[1], 170);
    g.addColorStop(0, `rgba(255,248,236,${(0.75 * pG).toFixed(3)})`); g.addColorStop(1, 'rgba(255,248,236,0)'); k.fillStyle = g; k.fillRect(0, 0, MD_W, CH);
    for (let i = 0; i < 9; i++) { const a = 0.5 + 0.5 * Math.sin(T * 2 + i * 1.7); k.fillStyle = `rgba(255,255,255,${(0.6 * a * pG).toFixed(3)})`; k.beginPath(); k.arc(40 + i * 40, 70 + (i % 3) * 22, 3 + (i % 2) * 2, 0, TAU); k.fill(); }
    k.restore();
  }
  // 前景大花：一开始是清楚的小花，虚化后变成大而软的色块
  k.save(); mdBlur(k, 7 * pF);
  DR_FG.forEach(([x, y, r], i) => { const j = i % 3; D.circ(x, y, r * lerp(0.5, 1, pF), c(DR_COL[j][0], DR_COL[j][1])); });
  k.restore();
});
