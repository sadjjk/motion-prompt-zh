// Frame within a frame / フレームインフレーム
// Pack: photo (Photo looks, lighting & styles) · element 30 · Category: Composition
// What happens: A dark doorway fades in close to the camera and frames the distant person a second time; a light strip along its inner edge shows that it has thickness.
// Motion spec:
//   - The person stands at the centre, x 270, 56 px tall; a tree on the left and a building on the right behind.
//   - Doorway: 168 px wide from y 30 to the bottom edge, centred; everything outside it is filled with dark #33405A.
//   - A 6 px light edge #CFCECB runs along the inside of the frame to show its thickness; a dashed line traces the opening and "框中框" is labelled at the top.
//   - Guides: amber #F5A524 dashed lines, 1.6 px wide, fading in with the effect amount.
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, ease-in-out).
// When to use it: To pull the eye to the subject, add layers, or suggest peeking or separation.
// Tags: frame within a frame, framing, doorway, window, composition, depth
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-062
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 30
// 框中框: 近处的门框淡入，把远处的主体再框一次；门框内沿留一条浅色边，表示它有厚度（不是后期加的装饰边框）
CELLS[29] = cell((k, u) => {
  const a = env(u), D = field(k, BASE, 150);
  tree(D, 120, 150, 14, BASE); building(D, 400, 150, 44, 50, BASE, 4);
  figure(D, CW / 2, 162, 56, BASE.person);
  const hx = CW / 2 - 84, hy = 30, hw = 168, hh = CH - 30;
  k.save(); k.globalAlpha = a;
  k.fillStyle = BASE.shade; k.beginPath(); k.rect(0, 0, CW, CH); k.rect(hx, hy, hw, hh); k.fill('evenodd');
  k.fillStyle = BASE.edge; k.fillRect(hx - 6, hy - 6, hw + 12, 6); k.fillRect(hx - 6, hy, 6, hh); k.fillRect(hx + hw, hy, 6, hh);
  k.restore();
  guide(k, [[hx, hy], [hx + hw, hy], [hx + hw, CH], [hx, CH], [hx, hy]], a);
  tag(k, '框中框', CW / 2, 16, a);
});
