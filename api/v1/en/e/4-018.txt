// Spirograph / スピログラフ
// Pack: scenes (Styled scenes) · element 18 · Category: Patterns & graphics
// What happens: Translucent gears roll around 3 rings on cream paper in turn, their pen tips tracing red, blue and green rosettes that close and hold.
// Motion spec:
//   - Cream background #F2EDE3. 3 fixed rings of radius 84 px; the gears roll inside them.
//   - Gear radii 32 / 18 / 13 px, pen offsets 25 / 14 / 8 px, inked red #D2452F, blue #1F3B8A and green #2E7D63.
//   - The 3 curves start 0.4 s apart and each takes 3.6 s to draw, closing after 8 / 3 / 13 laps.
//   - A translucent gear and pen dot show while drawing; the closed curves hold, then fade out over 5.4–6.0 s (6.4 s cycle).
// When to use it: Openers for handmade, stationery or education brands, poster backgrounds, or section breaks in a presentation.
// Tags: spirograph, hypotrochoid, gears, line-art, handmade, drawing, pattern
// Size: 539x208 px · Cycle: 6.4 s
// Needs: kit scenes -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/scenes.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-018
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 18
// 18 万花尺：三组齿轮各在半径 84 的圆环内滚动，笔尖错峰 0.4 秒画出红、蓝、绿三朵繁花曲线（各 3.6 秒），闭合后停留，5.4 秒淡出
const SPIRO_R = 84;
const SPIRO_SETS = [[100, 32, 25, '#D2452F', 8], [269.5, 18, 14, '#1F3B8A', 3], [439, 13, 8, '#2E7D63', 13]]; // [cx, r, d, 墨色, 闭合圈数 r/gcd(R,r)]
CELLS[17] = (k, w, h, u) => {
  k.fillStyle = '#F2EDE3'; k.fillRect(0, 0, CW, CH);
  const cy = 104, R = SPIRO_R;
  k.strokeStyle = 'rgba(40,40,40,.25)'; k.lineWidth = 4;
  for (const s of SPIRO_SETS) { k.beginPath(); k.arc(s[0], cy, R + 3, 0, TAU); k.stroke(); }
  const fade = fadeIn(u);
  SPIRO_SETS.forEach(([cx, r, d, col, turns], si) => {
    const tm = seg(u, 0.2 + si * 0.4, 3.8 + si * 0.4); if (tm <= 0) return;
    const T = turns * TAU * tm; let px = cx + R - r + d, py = cy;
    k.save(); k.globalAlpha = 0.85 * fade; k.strokeStyle = col; k.lineWidth = 0.9; k.beginPath(); k.moveTo(px, py);
    for (let t = 0.03; ; t += 0.03) {                       // 步长 0.03 rad，最后一点落在 T
      const tt = Math.min(t, T);
      px = cx + (R - r) * Math.cos(tt) + d * Math.cos((R - r) / r * tt); py = cy + (R - r) * Math.sin(tt) - d * Math.sin((R - r) / r * tt); k.lineTo(px, py);
      if (tt >= T) break;
    }
    k.stroke(); k.restore();
    if (tm < 1) {                                            // 画线中显示半透明齿轮与笔尖
      const gx = cx + (R - r) * Math.cos(T), gy = cy + (R - r) * Math.sin(T);
      k.strokeStyle = 'rgba(40,40,40,.35)'; k.fillStyle = 'rgba(255,255,255,.35)'; k.lineWidth = 0.8;
      k.beginPath(); k.arc(gx, gy, r, 0, TAU); k.fill(); k.stroke();
      k.fillStyle = col; k.beginPath(); k.arc(px, py, 2, 0, TAU); k.fill();
    }
  });
  grain(k, 0.07, 'multiply');
};
