// Rotating gradient border / 回転グラデーションボーダー
// Pack: lp (Landing page & web effects) · element 30 · Category: Hover & click interactions
// What happens: A multicolour gradient circles the border of a button on a dark page; when the cursor hovers, it spins faster and a glow appears around it.
// Motion spec:
//   - A 200×52 pill button on a dark page with a 3px conic-gradient border (vermilion → blue → purple → vermilion).
//   - The gradient angle turns 1 full turn per cycle at base speed; while the cursor hovers (0.7–3.8 s, easeInOut power 2) it adds 2 more turns, a whole 3 turns per 4.8 s cycle, so the loop is seamless.
//   - The cursor reaches the button at 0.6 s; an outer glow (shadowBlur 26) comes on over 0.25 s with easeOut and goes off over 0.25 s after the cursor leaves at 3.8 s.
// When to use it: Use it to make the main call to action stand out on dark heroes or pricing pages.
// Tags: gradient-border, conic-gradient, glow, button, hover, animated-border
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-122
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 30
// 改写自 KittenJam Motion Library #110 旋转光边按钮：深色页面中央 200×52 胶囊按钮，外框是一圈 3px 锥形渐变（朱 → 蓝 → 紫 → 朱）。
// 渐变角度 = 每周期 1 圈的基础转速 + 光标悬停期间（0.7–3.8 秒，easeInOut 2 次方）额外 2 圈，合计每周期整数 3 圈，首尾无缝。
// 光标 0.6 秒移到按钮上，外光晕（shadowBlur 26）0.25 秒 easeOut 亮起；3.8 秒移开后 0.25 秒熄灭。
const RGB_BW = 200, RGB_BH = 52;
CELLS[29] = (k, w, h, u) => {
  frame(k);
  const [cx, cy] = path([[0, 430, 150], [0.6, 300, 94], [3.8, 300, 94], [4.4, 430, 150], [4.8, 430, 150]], u);
  const glow = eOut(P(u, 0.55, 0.8)) * (1 - eOut(P(u, 3.8, 4.05)));
  const ang = Math.PI * 2 * (u / CYC + 2 * eIOs(P(u, 0.7, 3.8))) - Math.PI / 2;
  view(k, () => {
    k.fillStyle = '#151A26'; k.fillRect(0, 0, VW, VH);
    const bx = (VW - RGB_BW) / 2, by = (VH - RGB_BH) / 2;
    const g = k.createConicGradient(ang, VW / 2, VH / 2);
    [[0, C.acc], [0.33, C.blue], [0.66, '#B57CFF'], [1, C.acc]].forEach(([o, c]) => g.addColorStop(o, c));
    // 外光晕：同一圈渐变带阴影，悬停时出现
    if (glow > 0) { k.save(); k.globalAlpha = glow; k.shadowColor = 'rgba(181,124,255,.9)'; k.shadowBlur = 26; k.fillStyle = g; rr(k, bx - 3, by - 3, RGB_BW + 6, RGB_BH + 6, (RGB_BH + 6) / 2); k.fill(); k.restore(); }
    k.fillStyle = g; rr(k, bx - 3, by - 3, RGB_BW + 6, RGB_BH + 6, (RGB_BH + 6) / 2); k.fill();
    k.fillStyle = '#1C2333'; rr(k, bx, by, RGB_BW, RGB_BH, RGB_BH / 2); k.fill();
    txt(k, '开始使用', VW / 2, VH / 2 + 1, `700 16px ${JP}`, '#fff', 'center', 'middle', '1px');
    cursor(k, cx, cy, 0);
  });
};
