// Compound growth curve / 複利カーブ
// Category: Motion graphics / Shapes & objects / Information & tech
// What happens: On left and bottom axes, the growth curve for 10% per period over 40 periods is drawn from the origin. A white dot rides the tip to the right, the curve shoots up near the end, and 「×45」 appears at the top right.
// Motion spec:
//   - Grey axes frame a plot area 460 × 150 px; the curve is y = (1.1^(40x) − 1) / (1.1^40 − 1), the growth of 10% per period over 40 periods scaled to the plot.
//   - At 0.3–3.6 s the gold line, 3 px wide with a soft glow, is drawn from left to right with ease-in-out; a white dot of radius 5 px sits at its tip.
//   - At 3.6–4.0 s the label 「×45」 fades in at the top right in 30 px bold gold, since 1.1^40 ≈ 45.
//   - At 5.4–6.0 s the curve and label fade out; at 0 s only the axes are drawn, so the loop restarts cleanly.
// When to use it: Use it to explain compound interest, habits or learning, where small steps add up to a big difference later.
// Tags: compound-interest, exponential, growth-curve, line-chart, finance, data-visualization, short-video
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/vibefx.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-127
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 04
// 复利曲线：每期 +10%、共 40 期的增长曲线从坐标原点画出，先平后陡，亮点沿线走，最后标出「×45」
CELLS[3] = (k, w, h, u) => {
  bg(k);
  const F = fadeIn(u), OX = 40, OY = 184, W = 460, H = 150, G = Math.pow(1.1, 40);
  const f = x => (Math.pow(1.1, 40 * x) - 1) / (G - 1);
  k.strokeStyle = '#3A3D48'; k.lineWidth = 2;
  k.beginPath(); k.moveTo(OX, OY - H - 10); k.lineTo(OX, OY); k.lineTo(OX + W + 10, OY); k.stroke();
  k.globalAlpha = F;
  const p = inOut3(seg(u, 0.3, 3.6)), n = 80;
  if (p > 0) {
    k.save(); k.strokeStyle = GOLD; k.lineWidth = 3; k.lineJoin = 'round'; k.shadowColor = GOLD; k.shadowBlur = 8;
    k.beginPath();
    for (let i = 0; i <= n; i++) { const x = p * i / n; const X = OX + W * x, Y = OY - H * f(x); if (i) k.lineTo(X, Y); else k.moveTo(X, Y); }
    k.stroke(); k.restore();
    const X = OX + W * p, Y = OY - H * f(p);
    k.fillStyle = '#FFFFFF'; k.beginPath(); k.arc(X, Y, 5, 0, TAU); k.fill();
  }
  const L = seg(u, 3.6, 4.0);
  if (L > 0) { k.globalAlpha = F * L; txt(k, '×45', OX + W - 8, OY - H + 6, 30, GOLD, 'right', 800); }
  k.globalAlpha = 1;
};
