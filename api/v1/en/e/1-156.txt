// Border draw button / ボーダードローボタン
// Pack: lp (Landing page & web effects) · element 41 · Category: Hover & click interactions
// What happens: When the cursor lands, an orange line runs clockwise from the top-left corner around the thinly outlined button and the label turns orange; when it leaves, the line retracts in reverse.
// Motion spec:
//   - A 180×52 button with only a faint 1px outline; the cursor moves onto it in 0.6 s.
//   - A 2px accent border draws from the top-left corner: top edge rightward → right edge down → bottom edge leftward → left edge up, 0.18 s per edge (0.72 s total, linear).
//   - The label turns accent orange while the last edge draws.
//   - The cursor leaves at 3.2 s and the border retracts in reverse order.
// When to use it: Use it on minimal sites where a quiet outline button needs a clear hover cue.
// Tags: button, border, outline, draw, hover, line
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-156
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 41
// 改写自 KittenJam Motion Library #171 边框描绘按钮：180×52 按钮平时只有 1px 淡色细框。光标 0.6 秒移入后，2px 朱色边框从左上角依次画出：
// 上边向右 → 右边向下 → 下边向左 → 左边向上，每段 0.18 秒（共 0.72 秒、匀速），最后一段画的同时文字变成朱色；3.2 秒移出后按相反顺序收回。
const BDRAW_W = 180, BDRAW_H = 52;
CELLS[40] = (k, w, h, u) => {
  frame(k, 'example.com/contact');
  const bx = (VW - BDRAW_W) / 2, by = (VH - BDRAW_H) / 2;
  const cur = t => path([[0, 430, 150], [0.6, 262, 88], [3.2, 262, 88], [3.8, 430, 150], [4.8, 430, 150]], t);
  const on = t => { const [a, b] = cur(t); return a > bx && a < bx + BDRAW_W && b > by && b < by + BDRAW_H; };
  // 总进度匀速 0→1（离开时 1→0），四段各占四分之一，所以收回时顺序自然相反
  const g = trans(on, u, 0.72, x => x), sg = [0, 1, 2, 3].map(j => clamp(g * 4 - j));
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.fillStyle = '#fff'; k.fillRect(bx, by, BDRAW_W, BDRAW_H);
    k.strokeStyle = C.faint; k.lineWidth = 1; k.strokeRect(bx + 0.5, by + 0.5, BDRAW_W - 1, BDRAW_H - 1);
    const L = bx + 1, R = bx + BDRAW_W - 1, T = by + 1, B = by + BDRAW_H - 1;
    const seg = (x1, y1, x2, y2, f) => { if (f <= 0) return; k.beginPath(); k.moveTo(x1, y1); k.lineTo(lerp(x1, x2, f), lerp(y1, y2, f)); k.stroke(); };
    k.strokeStyle = C.acc; k.lineWidth = 2; k.lineCap = 'square';
    seg(L, T, R, T, sg[0]); seg(R, T, R, B, sg[1]); seg(R, B, L, B, sg[2]); seg(L, B, L, T, sg[3]);
    k.lineCap = 'butt';
    txt(k, '联系我们', VW / 2, VH / 2 + 1, `700 15px ${JP}`, mixHex(C.ink, C.acc, sg[3]), 'center', 'middle', '2px');
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
