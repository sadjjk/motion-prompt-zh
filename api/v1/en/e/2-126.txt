// Growing coin stacks / 積み上がるコイン
// Category: Motion graphics / Shapes & objects / Objects & machines
// What happens: 5 stacks of gold coins on a ground line. The left stack starts first and each new coin drops a little onto the top; by 3.6 s the stacks hold 4, 7, 10, 13 and 16 coins.
// Motion spec:
//   - 5 stacks stand 80 px apart on a ground line; each coin is an ellipse 60 × 16 px with a gold top and a darker gold edge, and coins sit 9 px apart.
//   - Stack c holds up to 4 + 3·c coins (4, 7, 10, 13, 16); it starts at 0.3 + 0.3·c s and reaches its full height at 3.6 s, adding coins at an even rate.
//   - Each new coin drops 20 px onto the stack with an ease-out; the top coin of each stack has a short white highlight.
//   - At 5.4–6.0 s the coins fade out; at 0 s only the ground line is drawn, so the loop restarts cleanly.
// When to use it: Use it to show savings, income or user numbers rising step by step in a friendlier way than a chart.
// Tags: coin-stack, coins, growth, money, bar-chart, savings, short-video
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/vibefx.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-126
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 03
// 硬币堆高：5 列金币从左到右依次加高（4、7、10、13、16 枚），像柱状图一样长起来
CELLS[2] = (k, w, h, u) => {
  bg(k);
  const F = fadeIn(u), BASE = 184, STEP = 9;
  k.strokeStyle = '#3A3D48'; k.lineWidth = 2;
  k.beginPath(); k.moveTo(50, BASE + 10); k.lineTo(490, BASE + 10); k.stroke();
  k.globalAlpha = F;
  for (let c = 0; c < 5; c++) {
    const x = 110 + 80 * c, max = 4 + 3 * c, p = seg(u, 0.3 + 0.3 * c, 3.6) * max;
    const n = Math.floor(p + 1e-9), frac = p - n;
    const total = Math.min(max, n + (frac > 0 ? 1 : 0));
    for (let i = 0; i < total; i++) {
      let y = BASE - STEP * i;
      if (i === n) y -= 20 * (1 - out3(clamp(frac / 0.4)));
      k.fillStyle = GOLD_D; k.beginPath(); k.ellipse(x, y + 4, 30, 8, 0, 0, TAU); k.fill();
      k.fillStyle = GOLD; k.beginPath(); k.ellipse(x, y, 30, 8, 0, 0, TAU); k.fill();
      if (i === total - 1) {
        k.strokeStyle = 'rgba(255,255,255,0.6)'; k.lineWidth = 2;
        k.beginPath(); k.ellipse(x, y, 20, 4, 0, Math.PI * 1.1, Math.PI * 1.6); k.stroke();
      }
    }
  }
  k.globalAlpha = 1;
};
