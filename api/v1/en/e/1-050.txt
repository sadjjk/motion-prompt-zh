// Line beam / ラインビーム
// Pack: aiux (AI interface motion) · element 02 · Category: Loading & thinking
// What happens: Only the bottom edge of a rounded search field lights up, and the light glides from left to right and back.
// Motion spec:
//   - Search field 380 × 48 px, corner radius 24 px. Only the lower half of the border lights up.
//   - The light centre travels between 60 px from each end: right at 0.1–2.3 s, back left at 2.5–4.7 s (cubic ease-in-out).
//   - The lit band is 180 px wide, brightest at the centre: a 6 px layer with 14 px blur plus a 1.6 px core.
//   - A soft flattened glow sits under the edge (radius 70 px, squashed to 0.35 height, opacity 0.28).
// When to use it: Use it on a search field, command bar or ask box between submitting and getting results.
// Tags: line-beam, search, loading, input, glow, sweep
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit aiux -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/aiux.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-050
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 02
// 02 底边光束（border-beam size="line"）：光沿搜索框底边左右扫动，表示提交后正在加载
CELLS[1] = (k, w, h, u) => {
  bg(k);
  const x = 79.5, y = 80, W = 380, H = 48, R = 24;
  field(k, x, y, W, H, R, 'Search docs…', false);
  const s = u < 2.4 ? inOut3(seg(u, 0.1, 2.3)) : 1 - inOut3(seg(u, 2.5, 4.7));
  const cx = x + 60 + (W - 120) * s, hue = 200 + 360 * u / CYC;
  // 底边描边：水平渐变只在光束窗口内可见，并裁到输入框下半部分
  k.save(); k.beginPath(); k.rect(0, y + H / 2, CW, CH); k.clip();
  const g = k.createLinearGradient(cx - 90, 0, cx + 90, 0);
  g.addColorStop(0, hsla(hue, 100, 62, 0)); g.addColorStop(0.3, hsla(hue, 100, 62, 0.7));
  g.addColorStop(0.5, hsla(hue + 60, 100, 72, 1)); g.addColorStop(0.7, hsla(hue + 120, 100, 62, 0.7)); g.addColorStop(1, hsla(hue + 120, 100, 62, 0));
  for (const [lw, blur] of [[6, 14], [1.6, 0]]) {
    k.strokeStyle = g; k.lineWidth = lw; k.shadowColor = hsla(hue + 60, 100, 60, 0.9); k.shadowBlur = blur;
    rr(k, x + 0.5, y + 0.5, W - 1, H - 1, R); k.stroke();
  }
  k.restore(); k.shadowBlur = 0; k.shadowColor = 'transparent';
  // 边框下方的柔光
  const rg = k.createRadialGradient(cx, y + H, 0, cx, y + H, 70);
  rg.addColorStop(0, hsla(hue + 60, 100, 65, 0.28)); rg.addColorStop(1, hsla(hue + 60, 100, 65, 0));
  k.fillStyle = rg; k.save(); k.translate(cx, y + H); k.scale(1, 0.35); k.translate(-cx, -(y + H)); k.fillRect(cx - 70, y + H - 70, 140, 140); k.restore();
  // 放大镜图标
  k.strokeStyle = C.muted; k.lineWidth = 1.6; k.beginPath(); k.arc(x + W - 30, y + H / 2 - 2, 6, 0, TAU);
  k.moveTo(x + W - 25.5, y + H / 2 + 2.5); k.lineTo(x + W - 21, y + H / 2 + 7); k.stroke();
};
