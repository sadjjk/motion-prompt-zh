// Marching ants selection / マーチングアンツ
// Pack: lp (Landing page & web effects) · element 48 · Category: Loading, navigation & ambient loops
// What happens: In a design-tool canvas, the blue dashed box around a layer keeps crawling in one direction, and the size label below changes as the box grows and shrinks.
// Motion spec:
//   - A blue 1.5px dashed selection box (dash 6, gap 5) whose dash offset moves −11 every 0.48 s (10 rounds per cycle), so the line keeps flowing one way.
//   - Each corner has an 8×8 handle with a white fill and a blue outline.
//   - The box resizes between 200×90 and 280×120: growing over 0.4–2.0 s and shrinking over 2.8–4.4 s (easeInOut power 3).
//   - A label below shows the rounded width × height live.
// When to use it: Use it when introducing design tools, image editors or no-code products to show selecting and editing.
// Tags: marching-ants, selection, dashed, handles, design-tool, loop
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-163
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 48
// 改写自 KittenJam Motion Library #325 选取框蚂蚁线：蓝色 1.5px 虚线选取框（线段 6、间隔 5），虚线偏移每 0.48 秒走 −11（一周期 10 轮），持续同向流动；
// 四角各一个 8×8 白底蓝框控制点。框尺寸在 200×90 与 280×120 之间变化（0.4–2.0 秒放大、2.8–4.4 秒缩小，easeInOut 3 次方），下方标签实时显示取整后的宽 × 高。
const ANTS_DASH = [6, 5], ANTS_STEP = 0.48;
CELLS[47] = (k, w, h, u) => {
  frame(k, 'example.com/editor');
  const s = steps([[0.4, 0], [2.0, 1], [2.8, 1], [4.4, 0]], u);
  const bw = lerp(200, 280, s), bh = lerp(90, 120, s), x = VW / 2 - bw / 2, y = 72 - bh / 2;
  view(k, () => {
    k.fillStyle = '#F7F8FA'; k.fillRect(0, 0, VW, VH);
    k.fillStyle = C.faint; for (let gx = 12; gx < VW; gx += 20) for (let gy = 12; gy < VH; gy += 20) k.fillRect(gx, gy, 2, 2);
    // 被选中的图层（随选取框一起改变尺寸）
    k.fillStyle = C.blueL; k.fillRect(x, y, bw, bh);
    k.fillStyle = C.acc; k.beginPath(); k.arc(x + 30, y + bh / 2, 14, 0, 7); k.fill();
    txt(k, '设计工具', x + 54, y + bh / 2 + 1, `800 16px ${JP}`, C.ink, 'left', 'middle');
    // 先描一圈白线垫底，虚线的间隙才看得清
    k.lineWidth = 1.5; k.strokeStyle = '#fff'; k.strokeRect(x, y, bw, bh);
    k.strokeStyle = C.blue; k.setLineDash(ANTS_DASH); k.lineDashOffset = -11 * (mod(u, ANTS_STEP) / ANTS_STEP); k.strokeRect(x, y, bw, bh);
    k.setLineDash([]); k.lineDashOffset = 0;
    [[x, y], [x + bw, y], [x, y + bh], [x + bw, y + bh]].forEach(([a, b]) => { k.fillStyle = '#fff'; k.fillRect(a - 4, b - 4, 8, 8); k.strokeStyle = C.blue; k.strokeRect(a - 4, b - 4, 8, 8); });
    const lab = `${Math.round(bw)} × ${Math.round(bh)}`;
    k.font = `700 11px ${EN}`; const lw = k.measureText(lab).width + 16;
    k.fillStyle = C.blue; rr(k, VW / 2 - lw / 2, y + bh + 10, lw, 18, 4); k.fill();
    txt(k, lab, VW / 2, y + bh + 19.5, `700 11px ${EN}`, '#fff', 'center', 'middle');
  });
};
