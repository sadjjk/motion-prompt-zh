// Ink line and wash / ペン画と淡彩
// Category: Visual styles / Craft & patterns / Drawing & painting media
// What happens: A small bird on a branch. Brown, cream and branch-coloured washes bloom one at a time with backruns at their edges; then ink lines draw the body, head, tail, beak, branch, legs and wing, and the eye is dotted in last.
// Motion spec:
//   - Warm off-white paper. A small bird about 120 px from tail to beak on a slanting branch, with its head at (292, 78).
//   - 0.3–1.8 s: 4 washes fade in 0.3 s apart, each over 0.6 s at 35–40% opacity; the 3 on the bird are 6–8 px larger than its outline so the colour spills past it, and the branch wash is a slanting band 12–18 px wide along the branch; 1.2–1.8 s: 4 darker rings appear inside them like backruns.
//   - 1.8–4.4 s: 8 ink lines in #1B1B1B are drawn one after another; along each line the width is 0.6 + 2.2·sin(πs) px, with s going from 0 to 1, so every stroke tapers at both ends.
//   - 4.4–4.8 s: the eye grows into an ink dot 2.6 px in radius.
//   - At 5.4–6.0 s the drawing fades back to blank paper; at 0 s and 6.4 s the card is blank paper, so the loop restarts cleanly.
// When to use it: Use it for nature, travel or urban sketching, or anything that should feel like a magazine illustration.
// Tags: drawing, ink, wash, watercolour, line-weight, process
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/drawstyles.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-077
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 06
// 墨线淡彩：顺序与铅笔水彩相反，先铺湿润的淡彩，颜色漫出轮廓、边缘起回流花边，再用软笔尖勾出两头细中间粗的墨线
{
  const INK = '#1B1B1B';
  // 小鸟：身体、头、尾巴、喙；站在一根斜枝上
  const BODY = blob(262, 104, 34, 11, 0.03, 0.78);
  const HEAD = blob(292, 78, 17, 12, 0.03);
  // 枝条的淡彩：沿墨线 (150,178)→(392,138) 的一条斜带，半宽 6–9 px，比墨线略长，跟枝条同向
  // （原先用横向椭圆，看起来像鸟脚下的影子，和斜枝对不上）
  const BRANCH_WASH = [];
  {
    const top = [], bot = [];
    for (let i = 0; i <= 24; i++) {
      const s = i / 24, x = 146 + 250 * s, y = 178.7 - 41.4 * s, hw = 6 + 3 * Math.sin(Math.PI * s);
      top.push([x, y - hw + (rnd(i, 17) - 0.5) * 2]); bot.push([x, y + hw + (rnd(i, 18) - 0.5) * 2]);
    }
    BRANCH_WASH.push(...top, ...bot.reverse());
  }
  const WASH = [
    [blob(260, 106, 42, 13, 0.07, 0.8), 'rgba(196,140,90,.40)'],
    [blob(292, 78, 23, 14, 0.07), 'rgba(196,140,90,.40)'],
    [blob(262, 118, 26, 15, 0.08, 0.6), 'rgba(230,200,150,.40)'],
    [BRANCH_WASH, 'rgba(120,100,80,.35)'],
  ];
  const BLOOMS = [[240, 96, 9], [280, 116, 7], [296, 72, 6], [222, 162, 8]];
  const LINES = [
    [...BODY.slice(14), ...BODY.slice(0, 8)],
    [...HEAD.slice(10), ...HEAD.slice(0, 26)],
    [[230, 96], [206, 88], [198, 96], [230, 108]],
    [[307, 76], [322, 80], [306, 84]],
    [[150, 178], [210, 166], [270, 158], [340, 146], [392, 138]],
    [[256, 128], [254, 146], [258, 158]], [[270, 128], [272, 146], [276, 156]],
    [[240, 100], [258, 92], [276, 104]],
  ];
  CELLS[5] = (k, w, h, u) => {
    paper(k, '#F8F4EA', 'rgba(0,0,0,.04)', 500, 13);
    const a = vis(u);
    // 0.3–1.8 秒：淡彩一块块铺开，透明度随进度上升
    WASH.forEach(([pts, col], i) => {
      const t = seg(u, 0.3 + 0.3 * i, 0.9 + 0.3 * i);
      if (t <= 0) return;
      k.globalAlpha = a * t; k.fillStyle = col; fillPts(k, pts);
    });
    // 回流花边：边缘更深的小色块
    BLOOMS.forEach(([x, y, r], i) => {
      const t = seg(u, 1.2 + 0.12 * i, 1.8);
      if (t <= 0) return;
      k.globalAlpha = a * t; k.strokeStyle = 'rgba(150,100,60,.35)'; k.lineWidth = 1.4;
      strokePts(k, [...blob(x, y, r, 80 + i, 0.15), blob(x, y, r, 80 + i, 0.15)[0]]);
    });
    // 1.8–4.4 秒：墨线，每条两头细中间粗
    k.globalAlpha = a; k.strokeStyle = INK; k.lineCap = 'round';
    const tl = seg(u, 1.8, 4.4) * LINES.length;
    for (let j = 0; j < Math.ceil(tl); j++) {
      const pts = LINES[j], f = Math.min(1, tl - j), m = (pts.length - 1) * f;
      for (let i = 0; i < Math.ceil(m); i++) {
        const s = (i + 0.5) / (pts.length - 1), g = Math.min(1, m - i);
        const [x0, y0] = pts[i], [x1, y1] = pts[i + 1];
        k.lineWidth = 0.6 + 2.2 * Math.sin(Math.PI * s);
        k.beginPath(); k.moveTo(x0, y0); k.lineTo(mix(x0, x1, g), mix(y0, y1, g)); k.stroke();
      }
    }
    const e = seg(u, 4.4, 4.8);
    if (e > 0) { k.fillStyle = INK; k.beginPath(); k.arc(298, 74, 2.6 * e, 0, TAU); k.fill(); }
    k.globalAlpha = 1;
  };
}
