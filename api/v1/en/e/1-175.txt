// Scroll zoom to full / スクロールで全画面ズーム
// Pack: uxmotion (Interaction & data motion) · element 62 · Category: Scroll interactions
// What happens: A small rounded image sits in the middle of a dark screen. As scrolling progresses it grows to fill the frame, its corners turn square and the scene inside eases back; late in the scroll a title rises in from below, and scrolling back reverses it all.
// Motion spec:
//   - Scroll progress goes 0 → 1 from 0.4–4.6 s, with the scrollbar on the right in sync.
//   - The image grows linearly from 38% × 46% of the frame to 100%, with corners going 16 → 0 px.
//   - Inside, the scene scales from 1.3× down to 1× at the same time.
//   - From 70%–100% progress the title rises 20 px and fades in.
//   - From 5.4–6.3 s the scroll returns and everything plays in reverse.
// When to use it: Use it in travel stories, product features and brand stories to give one image a dramatic reveal.
// Tags: scroll, zoom, full-bleed, image, scrollytelling, hero
// Size: 539x208 px · Cycle: 6.4 s
// Needs: kit uxmotion -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/uxmotion.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-175
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 62
// 62 滚动放大全幅图片（KittenJam #153）：滚动进度 0.4–4.6 秒由 0 到 1（右侧滚动条同步）；中央小图宽高从画面的 38%×46% 线性放大到 100%，
// 圆角 16→0px，图内画面同时从 1.3 倍缩回 1 倍；进度 70%–100% 时标题从下 20px 淡入；5.4–6.3 秒滚回，全部倒退。
CELLS[61] = (k, w, h, u) => {
  k.fillStyle = UXD.page; k.fillRect(0, 0, CW, CH);
  const p = UXA_scroll(u, 0.4, 4.6), fw = CW * mix(0.38, 1, p), fh = CH * mix(0.46, 1, p), r = mix(16, 0, p), is = mix(1.3, 1, p);
  const hint = 1 - seg(p, 0, 0.25);
  if (hint > 0) {
    k.globalAlpha = hint; txt(k, '第 03 篇 · 旅途', CW / 2, 40, UXB_f(700, 10.5), UXD.ink3, 'center');
    txt(k, '向下滚动 ↓', CW / 2, 182, UXB_f(700, 10.5), UXD.ink3, 'center'); k.globalAlpha = 1;
  }
  const x = (CW - fw) / 2, y = (CH - fh) / 2;
  k.save(); rr(k, x, y, fw, fh, r); k.clip();
  const iw = CW * is, ih = CH * is;
  UXB_pic(k, (CW - iw) / 2, (CH - ih) / 2, iw, ih, ['sea', '#F2A07B', '#24406B', '#FFE0A8']);
  const ca = seg(p, 0.7, 1);
  if (ca > 0) {
    const g = k.createLinearGradient(0, CH * 0.45, 0, CH); g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, `rgba(0,0,0,${0.5 * ca})`);
    k.fillStyle = g; k.fillRect(0, 0, CW, CH);
    k.globalAlpha = ca; txt(k, '海边的第七天', 36, CH - 34 + 20 * (1 - ca), UXB_f(900, 24), '#FFFFFF');
    txt(k, '把城市留在身后', 36, CH - 14 + 20 * (1 - ca), UXB_f(700, 11), 'rgba(255,255,255,.8)'); k.globalAlpha = 1;
  }
  k.restore();
  scrollbar(k, p, 0.3, CW - 8, 10, CH - 10, true);
};
