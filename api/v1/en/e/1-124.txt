// Card deal / カードディール
// Pack: lp (Landing page & web effects) · element 32 · Category: Hover & click interactions
// What happens: Pressing Deal sends 5 cards up from the bottom in turn into a fan; later, Collect sends them back down starting with the last card.
// Motion spec:
//   - The cursor clicks Deal at 0.5 s; 5 cards (60×84) leave the bottom one by one, 0.12 s apart, 0.55 s each.
//   - They fan out to −24°…+24° (12° apart) around a shared pivot 132 below the card bottoms; the angle uses back-out with overshoot 1.7 while the cards rise 150px with easeOut power 3.
//   - At 3.1 s Collect is clicked; from 3.2 s, starting with the last card, each card folds back and drops 0.1 s apart over 0.35 s (easeInOut power 3).
// When to use it: Use it to reveal a set of card-style products, plans or perks, or for game-like moments.
// Tags: card-deal, fan, cards, stagger, overshoot, click
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-124
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 32
// 改写自 KittenJam Motion Library #98 发牌展开：光标 0.5 秒点「发牌」，5 张 60×84 卡片从视口下方依次发出（间隔 0.12 秒、每张 0.55 秒），
// 绕下方同一枢轴展开成 −24°…+24°（每张差 12°）的扇形，角度用 back-out（过冲 1.7）回弹；停留到 3.1 秒点「收牌」，
// 从最后一张开始每张间隔 0.1 秒、0.35 秒 easeInOut 3 次方收回下方。
const DEAL_N = 5, DEAL_W = 60, DEAL_H = 84, DEAL_R = 132;
const DEAL_BACK = x => { const s = 1.7; return 1 + (s + 1) * Math.pow(x - 1, 3) + s * Math.pow(x - 1, 2); };
CELLS[31] = (k, w, h, u) => {
  frame(k, 'example.com/cards');
  const [cx, cy] = path([[0, 430, 150], [0.4, 62, 136], [3.4, 62, 136], [4.2, 430, 150], [4.8, 430, 150]], u);
  const out = u >= 0.5 && u < 3.1;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    btn(k, 24, 122, 76, 28, out ? '收牌' : '发牌', out ? C.ink : C.acc, '#fff', 12);
    const px = VW / 2 + 30, pyBase = 64 + DEAL_R;
    for (let i = 0; i < DEAL_N; i++) {
      const pin = P(u, 0.6 + i * 0.12, 1.15 + i * 0.12);
      const pout = eIO(P(u, 3.2 + (DEAL_N - 1 - i) * 0.1, 3.55 + (DEAL_N - 1 - i) * 0.1));
      const q = pin * (1 - pout);
      if (q <= 0) continue;
      const a = (-24 + 12 * i) * Math.PI / 180 * (u < 3.2 ? DEAL_BACK(pin) : 1 - pout);
      const py = pyBase + 150 * (1 - (u < 3.2 ? eOut(pin) : 1 - pout));
      k.save(); k.translate(px, py); k.rotate(a); k.translate(-DEAL_W / 2, -DEAL_R - DEAL_H / 2);
      k.shadowColor = 'rgba(28,35,51,.18)'; k.shadowBlur = 8; k.shadowOffsetY = 3;
      k.fillStyle = '#fff'; rr(k, 0, 0, DEAL_W, DEAL_H, 7); k.fill(); k.shadowColor = 'transparent';
      pic(k, 5, 5, DEAL_W - 10, 46, i, 5);
      bars(k, 7, 58, 40, 2, C.faint, 5, 9);
      k.restore();
    }
    click(k, cx, cy, u, [0.5, 3.1]);
  });
};
