// Liquid fill / リキッドフィル
// Pack: motion2 (Motion graphics moves, part 2) · element 40 · Category: Text & shapes
// What happens: Mint-outlined hollow text fills from the bottom with a wavy surface and rising bubbles, holds full, then drains quickly; a percentage shows below.
// Motion spec:
//   - 132 px text outlined in mint at 3 px; the liquid is clipped to the letter shapes.
//   - The level rises over 0.3–3.3 s (quadratic ease-in-out) and drops back over 4.2–4.6 s.
//   - The surface is two sine waves with height 6 px·sin(π·level) (flat when empty or full), flowing on a 1.57 s period.
//   - 10 bubbles rise through the liquid; a 0–100% readout sits below.
// When to use it: Use it for progress, charging, hitting a goal, or a drinks brand headline.
// Tags: liquid, fill, wave, progress, water, text
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-079
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 40
// 改写自 KittenJam Motion Library #73（Hannah）· 液体填满：薄荷色描边的空心字，0.3–3.3 秒正弦液面由下往上填满（波高 6px·sin(π·液位)，前后两层波错相），液中气泡上浮，停留到 4.2 秒后 0.4 秒清空；下方显示百分比。
const LIQF_C = mkCanvas(CW, CH), LIQF_G = LIQF_C.getContext('2d');
CELLS[39] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '满格', F = `900 132px ${JP}`, TOP = h / 2 - 66, BOT = h / 2 + 52, CXT = w / 2, CYT = h / 2 - 6;
  const lv = u < 0.3 ? 0 : u < 3.3 ? eIOs(P(u, 0.3, 3.3)) : u < 4.2 ? 1 : 1 - eIOs(P(u, 4.2, 4.6));
  const g = LIQF_G; g.save(); g.clearRect(0, 0, CW, CH);
  const sy = lerp(BOT + 4, TOP - 6, lv), A = 6 * Math.sin(Math.PI * lv), ph = Math.PI * 2 * u / (CYC / 3);
  const wave = (col, off, dy) => { g.fillStyle = col; g.beginPath(); g.moveTo(0, CH); for (let x = 0; x <= CW; x += 6) g.lineTo(x, sy + dy + A * Math.sin(x * 0.035 - ph + off)); g.lineTo(CW, CH); g.closePath(); g.fill(); };
  wave('#1F9E7A', 1.3, -3); wave(C.mint, 0, 0);
  for (let b = 0; b < 10; b++) {
    const bx = CXT + (H01(b) - 0.5) * 240, by = BOT - mod(u / CYC * 2 + H01(b + 20), 1) * (BOT - TOP);
    if (by > sy + 4) { g.fillStyle = 'rgba(255,255,255,.45)'; g.beginPath(); g.arc(bx + 3 * Math.sin(by * 0.1), by, 2 + 2 * H01(b + 40), 0, 7); g.fill(); }
  }
  g.globalCompositeOperation = 'destination-in'; g.font = F; g.textAlign = 'center'; g.textBaseline = 'middle'; g.fillStyle = '#000'; g.fillText(S, CXT, CYT);
  g.restore();
  k.drawImage(LIQF_C, 0, 0);
  k.font = F; k.textAlign = 'center'; k.textBaseline = 'middle'; k.lineJoin = 'round'; k.strokeStyle = C.mint; k.lineWidth = 3; k.strokeText(S, CXT, CYT);
  txt(k, `${Math.round(lv * 100)}%`, w / 2, h - 14, `700 13px ui-monospace, Menlo, monospace`, C.dim, 'center', 'alphabetic');
};
