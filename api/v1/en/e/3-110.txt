// Healing vibe / 癒やし
// Category: Editing & photography / Light & color / Mood & tone recipes
// What happens: It starts from a plain, cool-toned picture of a steaming cup of tea on a windowsill. First the window turns warm, low sunlight slants across the sill and the cup throws a long shadow. Then a striped blanket slides in from the bottom left. Next a warm white wash lowers the contrast and fine grain flickers over it. Finally the camera pushes in very slowly towards the cup. The steam sways the whole time. A checklist on the right turns each numbered circle blue as its ingredient arrives, and the status reads plain version → adding → finished.
// Motion spec:
//   - 5 s cycle. 0–0.6 s is the plain version; the 4 ingredients join from 0.6 s, the next every 0.7 s, each fading in over 0.5 s.
//   - Low warm sunlight falls from the window across the sill as a slanted band, with a long cup shadow; the blanket slides in 60 px from the left with 2 stripes.
//   - A warm white wash at 0.28 lowers the contrast and 520 grain dots jump 12 times per second; very slow push-in from 1× to 1.1× over 2.7–4.4 s; the steam sways 1 time per cycle.
//   - The finished look holds until 4.4 s and slides back to the plain version over 4.4–4.95 s, so 5 s and 0 s are the same frame.
// When to use it: For lifestyle, café, reading and weekend vlog shots that show an unhurried everyday moment.
// Tags: healing, cozy, warm light, film grain, slow push-in, everyday life
// Size: 539x208 px · Cycle: 5 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-110
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 63
// 63 治愈感：窗边一杯茶。低角度暖阳斜照进来 → 加上随手搭的毯子（生活痕迹）→ 降低对比加颗粒 → 镜头极慢推进
const HL_CUP = { x: 214, y: 140 }, HL_WIN = { x: 40, y: 18, w: 128, h: 104 };
const HL_GRAIN = () => once('hlGrain', () => { const r = rng(63), o = []; for (let i = 0; i < 520; i++) o.push([r() * MD_W, r() * CH, r() < 0.5]); return o; });
CELLS[62] = mdCell('治愈感', ['低角度暖阳', '随手搭的毯子', '降对比加颗粒', '镜头极慢推进'], (k, u, [pS, pB, pG, pZ], D) => {
  const z = 1 + 0.1 * pZ * mdSlow(u, 3), T = u / CYC * TAU, W = HL_WIN;
  k.save(); k.translate(HL_CUP.x, HL_CUP.y - 30); k.scale(z, z); k.translate(-HL_CUP.x, -(HL_CUP.y - 30));
  D.rect(0, 0, MD_W, CH, '#DCDDE0');
  D.rect(W.x, W.y, W.w, W.h, mixc('#C9D3DE', '#FFE2B0', pS)); D.frame(W.x, W.y, W.w, W.h, 5, '#F4F2EE'); D.rect(W.x + W.w / 2 - 2, W.y, 4, W.h, '#F4F2EE');
  D.rect(0, 140, MD_W, 68, '#C8BBA8');   // 窗台
  if (pS > 0.01) {   // 低角度暖阳：从窗斜着落到窗台一长条，茶杯拉出长影
    k.save(); k.globalCompositeOperation = 'screen'; k.fillStyle = `rgba(255,190,110,${(0.5 * pS).toFixed(3)})`;
    k.beginPath(); k.moveTo(W.x + 4, W.y + 6); k.lineTo(W.x + W.w - 4, W.y + 6); k.lineTo(330, 200); k.lineTo(150, 200); k.closePath(); k.fill(); k.restore();
    k.save(); k.globalAlpha = 0.3 * pS; D.poly([[HL_CUP.x - 10, HL_CUP.y], [HL_CUP.x + 10, HL_CUP.y], [HL_CUP.x + 74, HL_CUP.y + 30], [HL_CUP.x + 56, HL_CUP.y + 30]], '#6E5A44'); k.restore();
  }
  if (pB > 0.01) {   // 毯子：从左下角滑进来，带两条格纹
    k.save(); k.globalAlpha = pB; k.translate(-60 * (1 - pB), 0);
    D.poly([[0, 150], [120, 140], [170, 168], [150, 208], [0, 208]], '#E6B8A2'); D.line([[10, 160], [130, 152], [160, 176]], 3, '#F3D6C6'); D.line([[0, 186], [150, 182]], 3, '#F3D6C6');
    k.restore();
  }
  D.rr(HL_CUP.x - 13, HL_CUP.y - 26, 26, 26, 4, '#F6F1E8'); D.rect(HL_CUP.x - 13, HL_CUP.y - 26, 26, 4, '#C98F4E');   // 茶杯 + 茶面
  k.strokeStyle = '#F6F1E8'; k.lineWidth = 3; k.beginPath(); k.arc(HL_CUP.x + 15, HL_CUP.y - 14, 6, -1.2, 1.2); k.stroke();
  k.strokeStyle = 'rgba(255,255,255,.6)'; k.lineWidth = 1.6; k.lineCap = 'round';   // 热气：一轮摆 1 次
  for (let j = 0; j < 2; j++) { k.beginPath(); for (let s = 0; s <= 10; s++) { const y = HL_CUP.y - 30 - s * 2.4; k[s ? 'lineTo' : 'moveTo'](HL_CUP.x - 4 + j * 8 + 2.5 * Math.sin(s * 0.7 + T + j * 2), y); } k.stroke(); }
  k.restore();
  if (pG > 0.01) {   // 降对比：盖一层暖白；颗粒：固定点阵按 u 跳着平移（12 帧/秒）
    k.save(); k.fillStyle = `rgba(250,238,222,${(0.28 * pG).toFixed(3)})`; k.fillRect(0, 0, MD_W, CH);
    const off = Math.floor(u * 12) * 7;
    HL_GRAIN().forEach(([x, y, w]) => { k.fillStyle = w ? `rgba(255,255,255,${(0.35 * pG).toFixed(3)})` : `rgba(60,40,20,${(0.22 * pG).toFixed(3)})`; k.fillRect((x + off) % MD_W, y, 1.2, 1.2); });
    k.restore();
  }
});
