// Cut-paper title sequence / 切り絵のタイトルバック
// Category: Visual styles / Art movements / Modern & postmodern
// What happens: On vermilion, black building shapes slice in from the left, right and below one after another to form a city skyline, with window slits cut into them. A black vertical line splits the frame top to bottom, the characters 夜 and 行 cut in on either side, then small credits for director, cast and English title follow. Finally the type exits left and the shapes slice back out the way they came, leaving plain vermilion.
// Motion spec:
//   - Ground #E8562A, shapes #14110F, type #F6EFE3; 10 building shapes.
//   - 0.3–1.0 s the 10 shapes slice in from the left, right or below, 0.18 s each.
//   - 1.3–1.6 s a 6 px vertical line at x 270 grows from top to bottom.
//   - 1.9 s the 52 px serif characters 夜 and 行 cut in either side of the line; 2.5 / 2.8 / 3.1 s three lines of credits.
//   - 5.4–5.7 s the type exits left; from 5.4 s the shapes slice back out in reverse, the last in going first, 0.03 s apart.
// When to use it: Use it for suspense, film history and big-event openers that need a bold title card.
// Tags: title-sequence, cut-paper, silhouette, hard-cut, film-title, graphic, explainer
// Size: 539x208 px · Cycle: 6.4 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/explainer.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-055
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 16
// 16 剪影片头：橘红底，黑色硬边色块从画外切进来拼成城市天际线，一道竖线从上到下劈开画面，
// 片名「夜行」和演职员小字依次硬切入；复位时色块按反方向切出，回到纯橘红
const TT_BG = '#E8562A', TT_INK = '#14110F', TT_CREAM = '#F6EFE3';
// 城市剪影色块：[x, 顶边 y, 宽, 切入方向（-1 从左、1 从右、0 从下）, 切入时刻]
const TT_BLDG = [
  [0, 120, 52, -1, 0.3], [48, 92, 40, 0, 0.42], [86, 132, 60, -1, 0.5], [142, 70, 30, 0, 0.6], [170, 112, 48, 1, 0.66],
  [324, 104, 44, -1, 0.72], [364, 60, 34, 0, 0.8], [396, 124, 56, 1, 0.86], [450, 84, 38, 0, 0.94], [486, 116, 53, 1, 1.0],
];
const TT_CUT = 0.18;                                             // 每块切入耗时（硬切感：很短）
const TT_LINE_X = 270, TT_LINE = [1.3, 1.6];                     // 竖线 x、劈开时段
const TT_TITLE = 1.9, TT_CREDITS = [2.5, 2.8, 3.1];             // 片名、三行小字的切入时刻
const TT_OUT = [SHOW, SHOW + 0.6];
CELLS[15] = (k, w, h, u) => {
  k.fillStyle = TT_BG; k.fillRect(0, 0, CW, CH);
  // 复位：后进来的先出去，方向和切入相反
  const N = TT_BLDG.length;
  k.fillStyle = TT_INK;
  TT_BLDG.forEach(([x, y, wd, dir, t0], i) => {
    const inn = seg(u, t0, t0 + TT_CUT), os = TT_OUT[0] + (N - 1 - i) * 0.03, o = seg(u, os, os + TT_CUT);
    if (inn <= 0 || o >= 1) return;
    const e = inOut3(inn) * (1 - inOut3(o));
    const dx = dir ? dir * (CW * 0.6) * (1 - e) : 0, dy = dir ? 0 : (CH - y) * (1 - e);
    k.fillRect(x + dx, y + dy, wd, CH - y);
    // 窗格：在色块上挖出几道橘红缝，剪纸感
    if (wd > 40 && e > 0.99) { k.fillStyle = TT_BG; for (let r = y + 14; r < CH - 20; r += 16) k.fillRect(x + dx + 8, r + dy, wd - 16, 3); k.fillStyle = TT_INK; }
  });
  // 竖线：从上往下劈开
  const lp = inOut3(seg(u, TT_LINE[0], TT_LINE[1])) * (1 - seg(u, TT_OUT[0] + 0.25, TT_OUT[0] + 0.25 + TT_CUT));
  if (lp > 0) { k.fillStyle = TT_INK; k.fillRect(TT_LINE_X - 3, 0, 6, CH * lp); }
  // 片名与小字：硬切出现（无渐变），切出时整体向左平移出画
  const shift = CW * inOut3(seg(u, TT_OUT[0], TT_OUT[0] + 0.3));
  k.textBaseline = 'middle';
  if (u >= TT_TITLE) {
    // 片名两字分立在竖线两侧，奶白色压在剪影之上
    k.fillStyle = TT_CREAM; k.font = `900 52px ${SERIF}`;
    k.textAlign = 'right'; k.fillText('夜', TT_LINE_X - 14 - shift, 46);
    k.textAlign = 'left'; k.fillText('行', TT_LINE_X + 14 - shift, 46);
  }
  const CR = [['导演  林一川', TT_LINE_X - 14, 'right', 84], ['主演  周白  沈青', TT_LINE_X + 14, 'left', 84], ['NIGHT WALK · 2026', TT_LINE_X + 14, 'left', 100]];
  CR.forEach(([s, x, al, y], i) => {
    if (u < TT_CREDITS[i]) return;
    // 小字跨在橘红底和黑色楼顶上，统一用奶白，两种底上都看得清
    k.fillStyle = TT_CREAM; k.font = i === 2 ? `700 9px ${DISP}` : `600 10px ${SANS}`; k.textAlign = al;
    k.fillText(s, x - shift, y);
  });
};
