// Floating label / フローティングラベル
// Pack: ui (UI component names) · element 38 · Category: Choosing & switching
// What happens: The cursor clicks the field: the label shrinks, moves up and turns blue, the underline spreads from the centre and an email address is typed letter by letter; it is then cleared and blurred.
// Motion spec:
//   - A 350×52 input; the label "Email address" starts inside the field in grey at 15px.
//   - The cursor clicks in at 0.9 s: over 0.25 s (easeOut power 3) the label shrinks to 11px, rises 15px and turns blue, and the blue underline grows from the centre to 334px over 0.3 s.
//   - From 1.4 s "hello@kitten.cn" is typed at 0.08 s per character, with the caret blinking every 0.5 s.
//   - From 3.3 s it is deleted at 0.025 s per character; at 3.85 s a click outside blurs the field and the label and underline return over 0.25 s.
// When to use it: Use it in sign-in, sign-up or contact forms where the field name should stay visible while typing.
// Tags: floating-label, input, text-field, form, focus, underline
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/ui.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-116
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 38
// 改写自 KittenJam Motion Library #148 输入框浮动标签：350×52 输入框，标签「邮箱地址」初始在框内居中 15px 灰字。
// 光标 0.9 秒点入：标签 0.25 秒 easeOut 缩到 11px、上浮 15px 变蓝；底部蓝线 0.3 秒从中间向两侧展开到 350px。
// 1.4 秒起逐字输入「hello@kitten.cn」（每字 0.08 秒），光标竖线 0.5 秒闪烁；3.3 秒起每字 0.025 秒删空，3.85 秒点框外失焦，标签与底线 0.25 秒还原。
const FLOAT_S = 'hello@kitten.cn';
CELLS[37] = ui((w, h, t) => {
  pageLines(96, 14, 350);
  const bx = 96, by = 74, bw = 350, bh = 52;
  const focus = eo(seg(t, .9, 1.15)) * (1 - eo(seg(t, 3.85, 4.1)));
  const typed = Math.max(0, Math.min(FLOAT_S.length, Math.floor((t - 1.4) / .08) + 1));
  const del = t >= 3.3 ? Math.min(FLOAT_S.length, Math.floor((t - 3.3) / .025) + 1) : 0;
  const n = t < 1.4 ? 0 : Math.max(0, typed - del);
  fillRR(bx, by, bw, bh, 8, focus > 0 ? '#fff' : C.block); strokeRR(bx, by, bw, bh, 8, C.line2, 1);
  // 底线：灰色常驻，蓝线从中点展开
  X.fillStyle = C.line2; X.fillRect(bx + 8, by + bh - 2, bw - 16, 2);
  const lw = (bw - 16) * eo(seg(t, .9, 1.2)) * (1 - eo(seg(t, 3.85, 4.1)));
  X.fillStyle = C.blue; X.fillRect(bx + bw / 2 - lw / 2, by + bh - 2, lw, 2);
  const fl = n > 0 ? 1 : focus;
  const ls = lerp(15, 11, fl), ly = by + bh / 2 - 15 * fl;
  text('邮箱地址', bx + 16, ly, ls, fl > .5 ? C.blue : C.hint, fl > .5 ? 600 : 500);
  const s = FLOAT_S.slice(0, n);
  text(s, bx + 16, by + 33, 15, C.ink, 500, 'left', NUMF);
  if (focus > .5 && Math.floor((t - .9) / .5) % 2 === 0) { const cxp = bx + 17 + tw(s, 15, 500, NUMF); X.fillStyle = C.blue; X.fillRect(cxp, by + 24, 1.5, 18); }
  pointer([[0, w + 30, h + 30], [.8, 300, 104], [1.3, 300, 104], [1.9, 470, 160], [3.7, 470, 160], [4.4, w + 30, h + 30]], [.9, 3.85], t);
});
