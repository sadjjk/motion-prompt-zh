// Coin flip / コインフリップ
// Pack: motion2 (Motion graphics moves, part 2) · element 30 · Category: Text & shapes
// What happens: Four gold coins carrying characters hop up in turn, spin twice, land and bounce once; the back shows a star.
// Motion spec:
//   - 4 coins of 36 px radius sit 96 px apart; they start at 0.35 s, 0.2 s apart, and each move lasts 1.7 s.
//   - In the first 28% a coin rises 96 px (cubic ease-out) through one turn, and by 42% it falls through a second turn (the turn is drawn as scaleX = cos).
//   - A 17 px hop follows until 48%, landing at 54%; the shadow shrinks with height.
//   - The front shows the character, the back a star; the rim is a darker disc offset 4 px in the turn direction.
// When to use it: Use it for playful captions about luck, prizes, rewards or games.
// Tags: coin, flip, spin, hop, playful, text
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-069
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 30
// 改写自 KittenJam Motion Library #3（Hannah）· 硬币翻转：4 枚字币依次（间隔 0.2 秒）绕 Y 轴翻两圈（scaleX = cos）并上跳 96px，落地后再小跳 17px；正面是字，背面是星。
const COIN_S = '好运连连', COIN_D = 0.2, COIN_T0 = 0.35, COIN_L = 1.7;
CELLS[29] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const R = 36, GAP = 96, FLOOR = 170;
  k.fillStyle = C.navy2; rr(k, 40, FLOOR, w - 80, 4, 2); k.fill();
  [...COIN_S].forEach((ch, i) => {
    const x = w / 2 + (i - 1.5) * GAP, t = (u - COIN_T0 - i * COIN_D) / COIN_L;
    let y = 0, rot = 0;
    if (t > 0 && t < 1) {
      if (t < 0.28) { const q = t / 0.28; y = -96 * eOut(q); rot = Math.PI * 2 * q; }
      else if (t < 0.42) { const q = (t - 0.28) / 0.14; y = -96 * (1 - eIn(q)); rot = Math.PI * 2 * (1 + q); }
      else if (t < 0.48) { const q = (t - 0.42) / 0.06; y = -17 * eOut(q); }
      else if (t < 0.54) { const q = (t - 0.48) / 0.06; y = -17 * (1 - eIn(q)); }
    }
    // 影子随高度变小
    const hk = 1 + y / 160;
    k.fillStyle = 'rgba(0,0,0,.35)'; k.beginPath(); k.ellipse(x, FLOOR - 1, R * hk, 5 * hk, 0, 0, 7); k.fill();
    const c = Math.cos(rot), sx = Math.max(0.04, Math.abs(c));
    k.save(); k.translate(x, FLOOR - R - 6 + y);
    // 硬币厚度：深色边先画，随翻转方向左右偏移
    k.save(); k.translate(4 * Math.sin(rot), 0); k.scale(sx, 1); k.fillStyle = '#B87A0C'; k.beginPath(); k.arc(0, 0, R, 0, 7); k.fill(); k.restore();
    k.scale(sx, 1);
    k.fillStyle = c >= 0 ? C.yel : '#F0B429'; k.beginPath(); k.arc(0, 0, R, 0, 7); k.fill();
    k.strokeStyle = 'rgba(15,26,61,.35)'; k.lineWidth = 3; k.beginPath(); k.arc(0, 0, R - 6, 0, 7); k.stroke();
    if (c >= 0) txt(k, ch, 0, 2, `900 36px ${JP}`, C.navy, 'center', 'middle');
    else { k.fillStyle = C.navy; star(k, 0, 1, 15, 0, 0.45); k.fill(); }
    k.restore();
  });
};
