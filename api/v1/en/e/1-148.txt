// Speed dial / スピードダイヤル
// Pack: ui (UI component names) · element 50 · Category: Opening & overlays
// What happens: The cursor clicks the blue + at lower right; it turns 135° into × as the page dims, 3 labelled mini buttons pop out above it from the bottom up, and a second click tucks them away in reverse.
// Motion spec:
//   - A blue + button of radius 22 at lower right; the cursor clicks it at 0.9 s and it turns 135° into × in 0.3 s (easeOut power 3) while a 0.35 dim layer covers the page.
//   - 3 mini buttons of radius 16, 44 apart, pop out above it from the bottom up, 0.06 s apart, each in 0.3 s with back.out(1.4) from 0.3 scale and 12px lower, with text labels on the left.
//   - Over 1.9–2.6 s the cursor rests on "Photo" and its fill turns light blue.
//   - A second click at 3.2 s tucks them away from the top down, 0.05 s apart, 0.2 s each, and the button turns back into +.
// When to use it: Use it in mobile apps that group frequent create actions such as notes, photos or sharing.
// Tags: speed-dial, fab, floating-action-button, menu, expand, click
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/ui.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-148
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 50
// 改写自 KittenJam Motion Library #318 浮动按钮展开菜单：右下半径 22 的蓝色「+」按钮。光标 0.9 秒点击：按钮 0.3 秒 easeOut 3 次方旋转 135° 成「×」，
// 页面盖上 0.35 的暗层；上方三个半径 16 的小按钮（间距 44）自下而上每隔 0.06 秒按 back.out(1.4) 0.3 秒从 0.3 倍、低 12px 处弹出，左侧带文字标签。
// 1.9–2.6 秒光标停在「拍照」上（底色变浅蓝）；3.2 秒再点按钮，小按钮自上而下每隔 0.05 秒 0.2 秒收回，按钮转回「+」。
const FAB_ITEMS = ['新建笔记', '拍照', '分享链接'];
const FAB_back = x => 1 + 2.4 * Math.pow(x - 1, 3) + 1.4 * Math.pow(x - 1, 2);
CELLS[49] = ui((w, h, t) => {
  pageLines(30, 20, 400);
  for (let i = 0; i < 3; i++) fillRR(30, 82 + i * 34, 400, 26, 6, C.block);
  const op = eo(seg(t, .9, 1.2)) * (1 - eio(seg(t, 3.2, 3.45)));
  if (op > 0) { X.fillStyle = `rgba(18,20,28,${.35 * op})`; X.fillRect(0, 0, w, h); }
  const fx = 490, fy = 158;
  const hv = seg(t, 1.9, 2.05) * (1 - seg(t, 2.6, 2.75));
  FAB_ITEMS.forEach((s, i) => {
    const k = FAB_back(seg(t, 1.0 + i * .06, 1.3 + i * .06)) * (1 - eio(seg(t, 3.2 + (2 - i) * .05, 3.4 + (2 - i) * .05)));
    if (k <= .01) return;
    const y = fy - 44 * (i + 1) + 12 * (1 - k), sc = .3 + .7 * k;
    X.save(); X.globalAlpha = clamp(k);
    const lw = tw(s, 12, 600) + 20;
    fillRR(fx - 28 - lw, y - 12, lw, 24, 12, '#fff'); text(s, fx - 28 - lw / 2, y, 12, C.ink, 600, 'center');
    X.translate(fx, y); X.scale(sc, sc);
    shadowed(() => circle(0, 0, 16, i === 1 && hv > 0 ? C.lb : '#fff'), 10, 3, .2);
    if (i === 0) fillRR(-5, -6, 10, 12, 2, C.blue);
    else if (i === 1) { fillRR(-7, -4, 14, 10, 2, C.blue); circle(0, 1, 3, '#fff'); }
    else { circle(-4, 0, 2.5, C.blue); circle(4, -4, 2.5, C.blue); circle(4, 4, 2.5, C.blue); }
    X.restore();
  });
  shadowed(() => circle(fx, fy, 22, C.blue), 14, 5, .25);
  X.save(); X.translate(fx, fy); X.rotate(Math.PI * .75 * op);
  X.strokeStyle = '#fff'; X.lineWidth = 2.6; X.lineCap = 'round';
  X.beginPath(); X.moveTo(-9, 0); X.lineTo(9, 0); X.moveTo(0, -9); X.lineTo(0, 9); X.stroke();
  X.restore();
  pointer([[0, w + 30, h + 30], [.8, fx, fy + 2], [1.0, fx, fy + 2], [1.9, fx, 72], [2.6, fx, 72], [3.1, fx, fy + 2], [3.3, fx, fy + 2], [4.4, w + 30, h + 30]], [.9, 3.2], t);
});
