// Hide-on-scroll header / スクロールで隠れるヘッダー
// Category: UI & Web / Pages / Scrolling
// What happens: Scrolling down slides the header up out of view; scrolling back up a little slides it back in.
// Motion spec:
//   - Scroll: 0→200 px at 0.4–1.5 s, 200→130 px at 2.1–2.6 s, 130→0 at 3.3–4.3 s (cubic ease-in-out).
//   - A 34 px header (logo, three menu items, button).
//   - Once a downward scroll passes the 34 px header height, it slides 34 px up out of view over 0.3 s (cubic ease-out).
//   - When an upward scroll starts it slides back over the same 0.3 s and stays while scrolling up to the top.
//   - While shown, a 12 px shadow sits under it (up to 0.16 opacity, fading in over the first 24 px of scroll).
// When to use it: Use it on small screens to give the content more room while reading and bring the menu back only when the reader scrolls up.
// Tags: hide-on-scroll, headroom, auto-hide-header, navbar, header, scroll-direction, sticky
// Size: 539x208 px · Cycle: 4.8 s
// Shared code it depends on: https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/lp.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#1-198
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// The code of this effect starts here. CELLS, constants and helpers live in the shared code it depends on.

// @element 53
// 53 滚动隐藏页头（往下滚时页头往上收起，往上滚时滑回来）
const HH_K = [[0.4, 0], [1.5, 200], [2.1, 200], [2.6, 130], [3.3, 130], [4.3, 0]];
const HH_PASS = (() => { let a = 0.4, b = 1.5; for (let i = 0; i < 40; i++) { const m = (a + b) / 2; if (steps(HH_K, m) < 34) a = m; else b = m; } return b; })();
const hhHidden = t => { t = mod(t, CYC); return t >= HH_PASS && t < 2.1; };
CELLS[52] = (k, w, h, u) => {
  frame(k);
  const y = steps(HH_K, u), hid = trans(hhHidden, u, 0.3, eOut), hy = -34 * hid;
  const v = steps(HH_K, u + 0.02) - steps(HH_K, u - 0.02);
  view(k, () => {
    k.save(); k.translate(0, 34 - y);
    pic(k, 24, 16, 214, 112, 1, 8);
    txt(k, '轻装上阵，专注内容', 262, 50, `800 22px ${JP}`, C.ink); bars(k, 262, 66, 210, 3); btn(k, 262, 106, 100, 26, '了解更多');
    k.fillStyle = C.soft; k.fillRect(0, 150, VW, 236);
    txt(k, '特色', VW / 2, 184, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 200, 136, 120, 8); k.fill(); pic(k, x + 8, 208, 120, 58, i, 5); bars(k, x + 10, 278, 110, 3, C.faint, 6, 11); }
    bars(k, 34, 352, 440, 3, C.faint, 6, 11);
    k.restore();
    const sh = (1 - hid) * clamp(y / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    if (hid < 1) header(k, hy);
    sbar(k, y / 250, 0.42);
    // 滚动方向提示
    const da = clamp(Math.abs(v) / 3);
    if (da > 0) { k.globalAlpha = da; k.fillStyle = 'rgba(28,35,51,.78)'; rr(k, VW - 118, VH - 32, 96, 22, 11); k.fill(); txt(k, v > 0 ? '↓ 向下滚动' : '↑ 向上滚动', VW - 70, VH - 20.5, `600 11px ${JP}`, '#fff', 'center', 'middle'); k.globalAlpha = 1; }
  });
};
