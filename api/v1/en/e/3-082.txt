// Bicolor lighting / バイカラー
// Pack: photo (Photo looks, lighting & styles) · element 46 · Category: Light sources & color
// What happens: The surroundings darken as a magenta light comes on at the left and a cyan one at the right; each side of the person takes one colour, blending to pale violet down the middle, and both colours spill onto the wall.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure (head centre x 270, y 78, radius 29 px); the background turns to the dark scene.
//   - Colour on the person: a horizontal gradient over x 206–334 — magenta #FF3FA4 for 0–42%, pale violet #A98CFF at 50%, cyan #33D6FF for 58–100% — darker towards the bottom.
//   - Wall spill: radial light of radius 200 px, magenta at (60, 80) and cyan at (480, 80), added at 0.32 opacity.
//   - The plan shows two lights at 90° left and right (magenta and cyan); top right reads "MAGENTA · CYAN".
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, ease-in-out).
// When to use it: For music videos, games and night city scenes that want a strong, neon-like colour mood.
// Tags: bicolor lighting, two-tone light, magenta, cyan, neon, light
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/photo.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#3-082
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 46
// 双色光（バイカラー）: 周りが暗くなり、左にマゼンタ、右にシアンの灯がともる。人物の左右が別の色に染まり、
// 鼻筋のあたりで2色が混ざって淡い紫になる。配置図にも色の違う2灯
CELLS[45] = cell((k, u) => {
  const a = env(u), MG = '#FF3FA4', CY = '#33D6FF';
  stagePortrait(k, pal(BASE, DARK, a));
  bust(k, PX, PY, PR, BASE.person, kk => {
    if (a < 0.002) return;
    const g = kk.createLinearGradient(PX - 2.2 * PR, 0, PX + 2.2 * PR, 0);
    g.addColorStop(0, MG); g.addColorStop(0.42, MG); g.addColorStop(0.5, '#A98CFF'); g.addColorStop(0.58, CY); g.addColorStop(1, CY);
    kk.globalAlpha = a; kk.fillStyle = g; kk.fillRect(0, 0, CW, CH);
    // 胴の陰り: 下ほど暗く
    const v = kk.createLinearGradient(0, PY, 0, CH); v.addColorStop(0, 'rgba(0,0,0,0)'); v.addColorStop(1, 'rgba(10,10,20,0.45)');
    kk.fillStyle = v; kk.fillRect(0, 0, CW, CH);
  });
  // 壁に色のこぼれ
  if (a > 0.002) {
    k.save(); k.globalCompositeOperation = 'lighter';
    for (const [x, c] of [[60, '255,63,164'], [480, '51,214,255']]) {
      const g = k.createRadialGradient(x, 80, 6, x, 80, 200); g.addColorStop(0, `rgba(${c},${(0.32 * a).toFixed(3)})`); g.addColorStop(1, `rgba(${c},0)`);
      k.fillStyle = g; k.fillRect(0, 0, CW, CH);
    }
    k.restore();
  }
  plan(k, [[Math.PI / 2, a, MG], [-Math.PI / 2, a, CY]]);
  hud(k, 'MAGENTA · CYAN');
});
