// Neural net / ニューラルネット
// Pack: scenes (Styled scenes) · element 12 · Category: Tech moods
// What happens: Over deep navy, orange signal lights run rightward along the strongest weights of a layered graph while nodes brighten and dim with activation.
// Motion spec:
//   - A 6-layer network with 4-7-9-9-7-3 nodes, 72 px between layers and 19 px between nodes.
//   - Links with weight above 0.62 are highlighted, carrying orange signals left to right 3 times per 6.4 s.
//   - Nodes have a 5 px radius; activation pulses 2 times per 6.4 s, phase-shifted layer by layer.
//   - Radial background #0E1230 → #04050B; node glows have a 13 px radius.
// When to use it: Use it as a hero background for AI products or research, to open talks explaining machine learning, or as a transition in tech videos.
// Tags: neural-network, ai, nodes, deep-learning, signals, graph, tech
// Size: 539x208 px · Cycle: 6.4 s
// Needs: kit scenes -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/scenes.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#4-012
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 12
// 12 神经网络：多层网络的连接图，信号光点沿权重线从左往右传递，节点随激活明暗起伏
const NN_LAYERS = [4, 7, 9, 9, 7, 3];
const NN = NN_LAYERS.map((n, li) => Array.from({ length: n }, (_, i) => [90 + li * 72, 98 + (i - (n - 1) / 2) * 19]));
const NN_W = []; // 每条连线的权重（常量）
NN.forEach((L, l) => { if (l < NN.length - 1) L.forEach((a, ai) => NN[l + 1].forEach((b, bi) => NN_W.push([l, a, b, hash(l * 1000 + ai * 37 + bi)]))); });
CELLS[11] = (k, w, h, u) => {
  const bg = k.createRadialGradient(269.5, 104, 0, 269.5, 104, 300); bg.addColorStop(0, '#0E1230'); bg.addColorStop(1, '#04050B'); k.fillStyle = bg; k.fillRect(0, 0, CW, CH);
  const ph = u / CYC;                                // 信号每周期传递 3 轮（原作周期 2.2/0.8≈2.75 秒）
  k.save(); k.globalCompositeOperation = 'lighter';
  for (const [l, a, b, wt] of NN_W) {
    const hot = wt > 0.62;
    k.strokeStyle = hot ? `rgba(120,160,255,${0.12 + wt * 0.2})` : 'rgba(90,120,220,.06)'; k.lineWidth = hot ? 0.8 : 0.5;
    k.beginPath(); k.moveTo(a[0], a[1]); k.lineTo(b[0], b[1]); k.stroke();
    const p = ((ph * 3 - l * 0.12 + wt) % 1 + 1) % 1 * 2.2;
    if (hot && p < 1) { const x = a[0] + (b[0] - a[0]) * p, y = a[1] + (b[1] - a[1]) * p; glow(k, x, y, 7, '255,190,110', 1); }
  }
  k.restore();
  NN.forEach((L, li) => L.forEach(([x, y], i) => {
    const act = 0.5 + 0.5 * Math.sin(TAU * ph * 2 - li * 1.1 + i * 1.3);   // 激活每周期起伏 2 次
    glow(k, x, y, 13, '255,140,60', act * 0.5);
    k.fillStyle = '#0A0D20'; k.strokeStyle = `rgba(170,195,255,${0.6 + act * 0.4})`; k.lineWidth = 1.2;
    k.beginPath(); k.arc(x, y, 5, 0, TAU); k.fill(); k.stroke();
    k.fillStyle = `rgba(255,200,130,${0.3 + act * 0.7})`; k.beginPath(); k.arc(x, y, 2.6 * act + 0.8, 0, TAU); k.fill();
  }));
  ['INPUT', 'H1', 'H2', 'H3', 'H4', 'OUT'].forEach((t, i) => txt(k, t, 90 + i * 72, 196, `500 7px ${MONO}`, 'rgba(170,190,255,.6)', 'center'));
  grain(k, 0.05);
};
