// Crumble / クランブル
// Pack: motion2 (Motion graphics moves, part 2) · element 36 · Category: Text & shapes
// What happens: Four characters tilt slightly in turn, then fall spinning and accelerating while shedding debris, vanish, and come back down from above into place.
// Motion spec:
//   - 4 characters (64 px) starting at 0.3 s, 0.16 s apart; one character's cycle is 3.6 s.
//   - 22–27%: tilts to a random angle within ±12°, held until 30%.
//   - 30–62%: falls 109 px with a cubic ease-in, rotating to within ±90° while fading; 5 debris bits fall with it.
//   - 80–92%: comes down from 32 px above with ease-out while fading in, back in place.
// When to use it: Use it for captions about endings, collapse or failure, or an outro.
// Tags: crumble, fall, collapse, debris, outro, text
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-075
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 36
// 改写自 KittenJam Motion Library #39（Hannah）· 崩落：每个字（间隔 0.16 秒）先歪斜 ±12° 以内，再加速下坠 109px、旋转 ±90° 以内并淡出，带碎屑；随后从上方 32px 处缓出落回原位重组。每字一轮 3.6 秒。
const CRUM_R = (() => { const r = rand(3939); return Array.from({ length: 4 }, () => ({ r0: (r() * 2 - 1) * 12, r1: (r() * 2 - 1) * 90, bits: Array.from({ length: 5 }, () => [(r() - 0.5) * 50, (r() - 0.5) * 40, 1.2 + r() * 0.8, r() * 6, 3 + r() * 4]) })); })();
CELLS[35] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '土崩瓦解', F = `900 64px ${JP}`, CY = 92, D2R = Math.PI / 180;
  k.fillStyle = C.navy2; rr(k, 70, 132, w - 140, 4, 2); k.fill();
  chars(k, S, F, 18).forEach((c, i) => {
    const t = (u - 0.3 - 0.16 * i) / 3.6, R = CRUM_R[i % CRUM_R.length];   // 取模：脚本被按非 UTF-8 解码时字符数会变多
    let y = 0, rot = 0, op = 1;
    if (t > 0.22 && t < 0.27) rot = R.r0 * D2R * P(t, 0.22, 0.27);
    else if (t >= 0.27 && t < 0.3) rot = R.r0 * D2R;
    else if (t >= 0.3 && t < 0.62) { const q = P(t, 0.3, 0.62), e = eIn(q); y = 109 * e; rot = lerp(R.r0, R.r1, e) * D2R; op = 1 - q; }
    else if (t >= 0.62 && t < 0.8) op = 0;
    else if (t >= 0.8 && t < 0.92) { const q = eOut(P(t, 0.8, 0.92)); y = -32 * (1 - q); op = q; }
    // 碎屑跟随下坠
    if (t >= 0.3 && t < 0.62) {
      const q = P(t, 0.3, 0.62);
      R.bits.forEach(([bx, by, sp, a, s]) => {
        k.save(); k.globalAlpha = 0.8 * (1 - q); k.translate(w / 2 + c.x + bx * (0.4 + q), CY + by + 140 * sp * eIn(q)); k.rotate(a + q * 6);
        k.fillStyle = i % 2 ? C.yel : '#fff'; k.fillRect(-s / 2, -s / 2, s, s); k.restore();
      });
    }
    if (op <= 0) return;
    k.save(); k.globalAlpha = op; k.translate(w / 2 + c.x, CY + y); k.rotate(rot);
    txt(k, c.ch, 0, 0, F, i % 2 ? C.yel : '#fff', 'center', 'middle');
    k.restore();
  });
};
