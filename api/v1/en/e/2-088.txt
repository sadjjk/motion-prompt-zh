// Swing letters / 振り子の揺れ
// Pack: motion2 (Motion graphics moves, part 2) · element 49 · Category: Text & shapes
// What happens: Four characters hang on threads from a rail and swing lazily about yellow pivots above them, each slightly out of phase with its neighbour.
// Motion spec:
//   - Each character hangs from a pivot above and swings ±14°.
//   - It makes 3 swings in 4.7 s (about 1.57 s each) on a cosine curve.
//   - Neighbouring characters are offset by 0.27 s.
//   - Every second character is coral, the rest white.
// When to use it: Use it for relaxed logos, character intros or idle screens.
// Tags: swing, pendulum, sway, hanging, cute, idle
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-088
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 49
// 改写自 KittenJam Motion Library #26（Hannah）· 钟摆摇晃：每个字用细线挂在横杆下，以字上方的小圆点为支点 ±14° 左右摆，4.7 秒摆 3 个来回（ease-in-out 的余弦），相邻字相位错开 0.27 秒，偶数位字用珊瑚色。
const SWNG_T = CYC / 3;
CELLS[48] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const S = '悠然自得', F = `900 56px ${JP}`, RAIL = 34, PIV = 50;
  k.fillStyle = C.navy3; rr(k, 70, RAIL - 2, w - 140, 4, 2); k.fill();
  chars(k, S, F, 34).forEach((c, i) => {
    const a = 14 * Math.PI / 180 * Math.cos(Math.PI * 2 * (u + 0.27 * i) / SWNG_T), x = w / 2 + c.x;
    k.strokeStyle = C.dim; k.lineWidth = 1.5; k.beginPath(); k.moveTo(x, RAIL); k.lineTo(x, PIV); k.stroke();
    k.save(); k.translate(x, PIV); k.rotate(a);
    k.beginPath(); k.moveTo(0, 0); k.lineTo(0, 30); k.stroke();
    txt(k, c.ch, 0, 62, F, i % 2 ? C.coral : '#fff', 'center', 'middle');
    k.restore();
    k.fillStyle = C.yel; k.beginPath(); k.arc(x, PIV, 4.5, 0, 7); k.fill();
  });
};
