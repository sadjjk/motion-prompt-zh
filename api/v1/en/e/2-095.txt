// Multilingual greeting / 多言語あいさつ
// Pack: motion2 (Motion graphics moves, part 2) · element 56 · Category: Text & shapes
// What happens: Inside a pill-shaped frame 你好, Hello, こんにちは and Bonjour blur out upward and blur in from below, while the frame width and the dot after it glide to fit each word.
// Motion spec:
//   - 4 words switch every 1.175 s (one round in 4.7 s).
//   - The old word moves up 18 px and blurs to 6 px as it fades over 0.3 s.
//   - The new word comes in from 18 px below over 0.15–0.5 s, blur 6 px → 0.
//   - Over 0–0.5 s the frame width eases (cubic in-out) to the new word plus 30 px each side, carrying the dot.
// When to use it: Use it in heroes for global brands, travel or language services, or on welcome screens.
// Tags: greeting, multilingual, hello, rotate-words, blur, hero
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://sadjjk.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://sadjjk.github.io/motion-prompt-zh/?lang=en#2-095
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 56
// 改写自 KittenJam Motion Library #301（Hannah）· 多语问候切换：胶囊框里每 1.175 秒换一种语言的问候（你好 → Hello → こんにちは → Bonjour）：旧词 0.3 秒上移 18px、模糊到 6px 淡出，新词 0.15–0.5 秒从下方模糊淡入；框宽以三次缓入缓出平滑变为新词宽度，后面的圆点跟着移动。
const GRT_W = ['你好', 'Hello', 'こんにちは', 'Bonjour'];
CELLS[55] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const F = `800 44px ${JP}`, N = GRT_W.length, T = CYC / N, n = Math.min(N - 1, Math.floor(u / T)), lt = u - n * T;
  const cur = GRT_W[n], prev = GRT_W[mod(n - 1, N)], PAD = 30, H = 76, CY = 92;
  const pw = lerp(tw(k, prev, F) + PAD * 2, tw(k, cur, F) + PAD * 2, eIO(P(lt, 0, 0.5)));
  const x0 = w / 2 - (pw + 30) / 2;
  rr(k, x0, CY - H / 2, pw, H, H / 2); k.fillStyle = C.navy2; k.fill(); k.strokeStyle = C.navy3; k.lineWidth = 2; k.stroke();
  k.save(); rr(k, x0, CY - H / 2, pw, H, H / 2); k.clip();
  const qo = P(lt, 0, 0.3), qi = P(lt, 0.15, 0.5);
  if (qo < 1) { k.globalAlpha = 1 - qo; k.filter = `blur(${(6 * qo).toFixed(2)}px)`; txt(k, prev, x0 + pw / 2, CY - 18 * eOut(qo), F, '#fff', 'center', 'middle'); }
  if (qi > 0) { k.globalAlpha = qi; k.filter = `blur(${(6 * (1 - qi)).toFixed(2)}px)`; txt(k, cur, x0 + pw / 2, CY + 18 * (1 - eOut(qi)), F, '#fff', 'center', 'middle'); }
  k.restore();
  k.fillStyle = C.coral; k.beginPath(); k.arc(x0 + pw + 18, CY + 16, 7, 0, 7); k.fill();
  txt(k, '来自世界各地的问候', w / 2, 172, `600 14px ${JP}`, C.dim, 'center', 'alphabetic');
};
